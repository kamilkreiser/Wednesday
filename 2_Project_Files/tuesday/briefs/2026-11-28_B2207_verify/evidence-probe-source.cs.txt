    private sealed class Probe : IOxpd1AuthenticationAgentHandler
    {
        internal Func<Oxpd1AgentRequest, CancellationToken, Task<Oxpd1PrePromptAnswer>>? Pre { get; init; }

        internal List<Oxpd1AgentRequest> Seen { get; } = [];

        public Task<Oxpd1PrePromptAnswer> PrePromptAsync(Oxpd1AgentRequest request, CancellationToken cancellationToken)
        {
            lock (Seen)
            {
                Seen.Add(request);
            }

            return Pre is null ? Task.FromResult(Oxpd1PrePromptAnswer.Continued()) : Pre(request, cancellationToken);
        }

        public Task<Oxpd1PostPromptAnswer> PostPromptAsync(Oxpd1AgentRequest request, CancellationToken cancellationToken)
        {
            lock (Seen)
            {
                Seen.Add(request);
            }

            return Task.FromResult(Oxpd1PostPromptAnswer.Canceled());
        }

        public Task SignedOutAsync(Oxpd1AgentRequest request, CancellationToken cancellationToken)
        {
            lock (Seen)
            {
                Seen.Add(request);
            }

            return Task.CompletedTask;
        }
    }

    private static async Task<string> Run(string label, byte[] bytes, string? contentType = SoapType, Probe? handler = null, Stream? stream = null)
    {
        handler ??= new Probe();
        Oxpd1AuthenticationAgentEndpoint endpoint = new(handler);
        string outcome;
        try
        {
            using Stream body = stream ?? new MemoryStream(bytes);
            Oxpd1AgentReply reply = await endpoint.AnswerAsync(body, contentType, TestContext.Current.CancellationToken);
            string text = Encoding.UTF8.GetString(reply.Body);
            Interlocked.Increment(ref replies);
            lock (Sync)
            {
                File.AppendAllText(Out + ".bodies", label + "\t" + text.Replace("\n", "\\n", StringComparison.Ordinal) + "\n");
            }

            if (text.Contains("Succeeded", StringComparison.OrdinalIgnoreCase))
            {
                Interlocked.Increment(ref succeededSeen);
            }

            string reason = string.Empty;
            string relates = "none";
            try
            {
                XDocument d = XDocument.Parse(text);
                reason = d.Descendants(XName.Get("Text", Soap)).FirstOrDefault()?.Value ?? string.Empty;
                XElement? rel = d.Root!.Element(XName.Get("Header", Soap))?.Elements().FirstOrDefault(e => e.Name.LocalName == "RelatesTo");
                relates = rel is null ? "none" : rel.Name.NamespaceName + (rel.Value == Mid ? " = the request's MessageID" : " (other value, length " + rel.Value.Length + ")");
            }
            catch (XmlException)
            {
                reason = "REPLY NOT XML";
            }

            bool leak = text.Contains(Ctx, StringComparison.OrdinalIgnoreCase) || text.Contains(Srv, StringComparison.OrdinalIgnoreCase) || text.Contains(Detail, StringComparison.OrdinalIgnoreCase);
            bool tsLeak = reply.ToString().Contains(Ctx, StringComparison.OrdinalIgnoreCase) || reply.ToString().Contains(Mid, StringComparison.OrdinalIgnoreCase);
            string seen;
            lock (handler.Seen)
            {
                seen = handler.Seen.Count.ToString(System.Globalization.CultureInfo.InvariantCulture);
                if (handler.Seen.Count > 0)
                {
                    Oxpd1AgentRequest r = handler.Seen[0];
                    seen += " (context equal " + (r.AuthenticationContextId == Ctx) + ", language " + (r.LanguageCode ?? "null") + ", server equal " + (r.ServerContextId == Srv) + (r.ServerContextId is null ? " (null)" : " (length " + r.ServerContextId.Length + ")") + ", ToString '" + r + "')";
                }
            }

            outcome = "status " + reply.Status + " fault " + (reply.FaultCode ?? "none") + " | '" + reason + "' | RelatesTo " + relates + " | handler saw " + seen + " | body holds context/server/detail " + leak + " | reply.ToString '" + reply + "' holds id " + tsLeak;
        }
        catch (Exception e)
        {
            int n;
            lock (handler.Seen)
            {
                n = handler.Seen.Count;
            }

            outcome = "THREW " + e.GetType().Name + " | message holds detail " + e.Message.Contains(Detail, StringComparison.Ordinal) + " | handler saw " + n;
        }

        Note(label + ": " + outcome);
        return outcome;
    }

    private static byte[] U8(string s) => Encoding.UTF8.GetBytes(s);

    private sealed class ChunkStream(byte[] data, int chunk, bool endless) : Stream
    {
        private int pos;

        internal long Served { get; private set; }

        internal int Reads { get; private set; }

        public override bool CanRead => true;

        public override bool CanSeek => false;

        public override bool CanWrite => false;

        public override long Length => throw new NotSupportedException();

        public override long Position { get => throw new NotSupportedException(); set => throw new NotSupportedException(); }

        public override void Flush()
        {
        }

        public override int Read(byte[] buffer, int offset, int count) => Fill(buffer.AsSpan(offset, count));

        public override ValueTask<int> ReadAsync(Memory<byte> buffer, CancellationToken cancellationToken = default) => ValueTask.FromResult(Fill(buffer.Span));

        private int Fill(Span<byte> dest)
        {
            Reads++;
            int n = Math.Min(chunk, dest.Length);
            if (!endless)
            {
                n = Math.Min(n, data.Length - pos);
            }

            for (int i = 0; i < n; i++)
            {
                dest[i] = endless ? (byte)' ' : data[pos + i];
            }

            pos += n;
            Served += n;
            return n;
        }

        public override long Seek(long offset, SeekOrigin origin) => throw new NotSupportedException();

        public override void SetLength(long value) => throw new NotSupportedException();

        public override void Write(byte[] buffer, int offset, int count) => throw new NotSupportedException();
    }

    private static byte[] Padded(int total)
    {
        string head = Env(Call()).Replace("<s:Body>", "<s:Body>" + "\u0001PAD", StringComparison.Ordinal);
        int pad = total - U8(head).Length + 4;
        return U8(head.Replace("\u0001PAD", new string(' ', pad), StringComparison.Ordinal));
    }

    private static string Nested(int levels) => string.Concat(Enumerable.Repeat("<x:n>", levels)) + string.Concat(Enumerable.Repeat("</x:n>", levels));

            Note("type " + t.Name + ": public static methods [" + pub + "], constructors [" + ctors + "], sealed " + t.IsSealed);

        string asm = File.ReadAllText(typeof(Oxpd1AuthenticationAgentEndpoint).Assembly.Location, Encoding.Latin1);

        Oxpd1AuthenticationAgentEndpoint endpoint = new(new Probe());

    [Fact]
    public async Task P5_two_and_sixty_four_concurrent_requests_on_one_endpoint()
    {
        Note("==== P5 concurrency");
        TaskCompletionSource gate = new(TaskCreationOptions.RunContinuationsAsynchronously);
        int arrived = 0;
        Probe handler = new()
        {
            Pre = async (r, ct) =>
            {
                if (Interlocked.Increment(ref arrived) == 2)
                {
                    gate.SetResult();
                }

                await gate.Task.WaitAsync(TimeSpan.FromSeconds(10), ct);
                return r.AuthenticationContextId.EndsWith('1') ? Oxpd1PrePromptAnswer.Continued() : Oxpd1PrePromptAnswer.Failed("second");
            },
        };
        Oxpd1AuthenticationAgentEndpoint endpoint = new(handler);
        string c1 = new string('A', 31) + "1";
        string c2 = new string('B', 31) + "2";
        Task<Oxpd1AgentReply> t1 = Answer(endpoint, Env(Call(context: c1), messageId: "urn:example:first"));
        Task<Oxpd1AgentReply> t2 = Answer(endpoint, Env(Call(context: c2, language: "fr-FR"), messageId: "urn:example:second"));
        Oxpd1AgentReply[] both = await Task.WhenAll(t1, t2);
        foreach ((Oxpd1AgentReply rep, string want, string code) in new[] { (both[0], "urn:example:first", "Continued"), (both[1], "urn:example:second", "Failed") })
        {
            XDocument d = XDocument.Parse(Encoding.UTF8.GetString(rep.Body));
            string rel = d.Root!.Element(XName.Get("Header", Soap))!.Element(XName.Get("RelatesTo", Wsa))!.Value;
            string got = d.Descendants(XName.Get("resultCode", Ns)).Single().Value;
            Note("two interleaved (both inside the handler at once): RelatesTo own " + (rel == want) + ", code own " + (got == code) + ", status " + rep.Status);
        }

        Note("two interleaved: handler saw " + handler.Seen.Count + ", languages " + string.Join(",", handler.Seen.Select(s => s.LanguageCode).Order()));

        Probe many = new()
        {
            Pre = async (r, ct) =>
            {
                await Task.Delay(Random.Shared.Next(0, 20), ct);
                return r.ServerContextId == r.AuthenticationContextId[..8] ? Oxpd1PrePromptAnswer.Continued() : Oxpd1PrePromptAnswer.Canceled();
            },
        };
        Oxpd1AuthenticationAgentEndpoint shared = new(many);
        List<Task<(int I, Oxpd1AgentReply R)>> tasks = [];
        for (int i = 0; i < 64; i++)
        {
            int k = i;
            string ctx = "C" + k.ToString("D31", System.Globalization.CultureInfo.InvariantCulture);
            string env = Env(Call(context: ctx, server: ctx[..8]), messageId: "urn:example:m" + k.ToString(System.Globalization.CultureInfo.InvariantCulture));
            tasks.Add(Task.Run(async () => (k, await Answer(shared, env)), TestContext.Current.CancellationToken));
        }

        int ownRel = 0;
        int continued = 0;
        foreach ((int i, Oxpd1AgentReply r) in await Task.WhenAll(tasks))
        {
            XDocument d = XDocument.Parse(Encoding.UTF8.GetString(r.Body));
            if (d.Root!.Element(XName.Get("Header", Soap))!.Element(XName.Get("RelatesTo", Wsa))!.Value == "urn:example:m" + i.ToString(System.Globalization.CultureInfo.InvariantCulture))
            {
                ownRel++;
            }

            if (d.Descendants(XName.Get("resultCode", Ns)).Single().Value == "Continued")
            {
                continued++;
            }
        }

        Note("64 concurrent on one endpoint: RelatesTo own " + ownRel + " of 64, each request's own values reached the handler (Continued) " + continued + " of 64, handler saw " + many.Seen.Count);
        FieldInfo[] fields = typeof(Oxpd1AuthenticationAgentEndpoint).GetFields(BindingFlags.Instance | BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic);
        Note("endpoint fields: " + string.Join(", ", fields.Select(f => (f.IsStatic ? "static " : string.Empty) + (f.IsInitOnly ? "readonly " : f.IsLiteral ? "const " : "MUTABLE ") + f.FieldType.Name + " " + f.Name)));
        foreach (Type t in typeof(Oxpd1AuthenticationAgentEndpoint).Assembly.GetTypes())
        {
            foreach (FieldInfo f in t.GetFields(BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.DeclaredOnly))
            {
                if (!f.IsLiteral)
                {
                    Note("static field in the assembly: " + t.Name + "." + f.Name + " " + f.FieldType.Name + (f.IsInitOnly ? " readonly" : " MUTABLE"));
                }
            }
        }

        Note("TOTAL replies " + Volatile.Read(ref replies) + " (of the probe's Run calls), any reply body holding 'Succeeded': " + Volatile.Read(ref succeededSeen));
    }

    private static async Task<Oxpd1AgentReply> Answer(Oxpd1AuthenticationAgentEndpoint e, string text)
    {
        using MemoryStream s = new(U8(text));
        return await e.AnswerAsync(s, SoapType, TestContext.Current.CancellationToken);
    }
