function rawSend(base, method, target, headers) {
  const u = new URL(base);
  return new Promise((resolve) => {
    const sock = net.connect(Number(u.port), u.hostname);
    let data = '';
    const hdr = { ...headers, Host: u.host, 'Content-Length': '2', Connection: 'close' };
    sock.on('connect', () => {
      sock.write(`${method} ${target} HTTP/1.1\r\n` + Object.entries(hdr).map(([k, v]) => `${k}: ${v}`).join('\r\n') + '\r\n\r\n{}');
    });
    sock.on('data', (d) => { data += d.toString('latin1'); });
    sock.on('end', () => { const m = /^HTTP\/1\.1 (\d{3})/.exec(data); resolve(m ? Number(m[1]) : 'NOSTATUS'); });
    sock.on('error', (e) => resolve('ERR ' + e.message));
  });
}
async function drive(h, session) {
  const ctx = await h.as(session);
  const out = {};
  for (const [method, p] of CASES) {
    let status, error;
    try {
      const res = await fetch(h.base + p, { method, redirect: 'manual', headers: ctx.json, body: method === 'GET' ? undefined : '{}' });
      status = res.status;
      try { const b = await res.json(); error = b && (b.error || null); } catch (_e) { error = null; }
    } catch (e) { status = 'ERR ' + e.message; }
    out[`${method} ${p}`] = { status, error: typeof error === 'string' ? error.slice(0, 60) : error };
  }
  const port = new URL(h.base).port;
  for (const [method, t] of RAW(port)) {
    out[`RAW ${method} ${t.replace(port, 'PORT')}`] = { status: await rawSend(h.base, method, t, ctx.json), error: null };
  }
  return out;
}
let result = {};

  try {
    result.viewer = await drive(h, h.seeded.viewerSession);
    result.anonymous = await drive(h, undefined);
    result.administrator = await drive(h, h.seeded.adminSession);
  } finally { await h.stop(); }
  if (OUT) fs.writeFileSync(OUT, JSON.stringify(result, null, 1));
}, 300000);
afterAll(() => { for (const d of made) fs.rmSync(d, { recursive: true, force: true }); });
