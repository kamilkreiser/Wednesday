const isP = (s) => s.startsWith(':');
const fill = (p, id) => p.split('/').map((s) => (isP(s) ? (id || S) : s)).join('/');
const mapLit = (p, f, id) => p.split('/').map((s) => (isP(s) ? (id || S) : f(s))).join('/');
const alt = (s) => [...s].map((c, i) => (i % 2 ? c.toUpperCase() : c.toLowerCase())).join('');
const alt2 = (s) => [...s].map((c, i) => (i % 2 ? c.toLowerCase() : c.toUpperCase())).join('');
const lastUp = (p, id) => { const a = fill(p, id).split('/'); for (let i = a.length - 1; i > 0; i--) if (!isP(p.split('/')[i])) { a[i] = a[i].toUpperCase(); break; } return a.join('/'); };
const mountUp = (p, id) => { const a = fill(p, id).split('/'); a[1] = a[1].toUpperCase(); a[2] = a[2].toUpperCase(); return a.join('/'); };
function forms(p, id) {
  const c = fill(p, id);
  return [
    ['canon', c], ['upper', mapLit(p, (s) => s.toUpperCase(), id)], ['title', mapLit(p, (s) => s.charAt(0).toUpperCase() + s.slice(1), id)],
    ['alt', mapLit(p, alt, id)], ['alt2', mapLit(p, alt2, id)], ['lastseg-up', lastUp(p, id)], ['mount-up', mountUp(p, id)],
    ['slash', c + '/'], ['upper-slash', mapLit(p, (s) => s.toUpperCase(), id) + '/'], ['query', c + '?x=1'],
    ['upper-slash-query', mapLit(p, (s) => s.toUpperCase(), id) + '/?a=b&c'],
  ];
}

function rawSend(base, method, target, headers, body = '{}') {
  const u = new URL(base);
  return new Promise((resolve) => {
    const sock = net.connect(Number(u.port), u.hostname);
    let data = Buffer.alloc(0);
    const b = Buffer.from(body);
    const hdr = { ...headers, Host: u.host, 'Content-Length': String(b.length), Connection: 'close' };
    sock.on('connect', () => {
      sock.write(Buffer.concat([Buffer.from(`${method} `), Buffer.from(target, 'utf8'), Buffer.from(' HTTP/1.1\r\n' + Object.entries(hdr).map(([k, v]) => `${k}: ${v}`).join('\r\n') + '\r\n\r\n'), b]));
    });
    sock.setTimeout(10000, () => sock.destroy());
    sock.on('data', (d) => { data = Buffer.concat([data, d]); });
    sock.on('close', () => { const m = /^HTTP\/1\.[01] (\d{3})/.exec(data.toString('latin1')); resolve(m ? Number(m[1]) : 'NOSTATUS'); });
    sock.on('error', (e) => resolve('ERR ' + e.code));
  });
}
async function send(h, ctx, method, p, bodyObj, extra = {}) {
  try {
    const res = await fetch(h.base + p, { method, redirect: 'manual', headers: { ...ctx.json, ...extra }, body: (method === 'GET' || method === 'HEAD') ? undefined : JSON.stringify(bodyObj || {}) });
    let error = null; let json = null;
    try { json = await res.json(); error = json && typeof json.error === 'string' ? json.error.slice(0, 50) : null; } catch (_e) { /* non-JSON */ }
    return { status: res.status, error, json };
  } catch (e) { return { status: 'ERR ' + e.message, error: null }; }
}
async function readItem(h, A, fid) {
  const r = await send(h, A, 'GET', '/api/feedback');
  const items = (r.json && r.json.items) || [];
  const it = items.find((i) => String(i.id) === String(fid));
  return { listStatus: r.status, present: !!it, admin_notes: it ? it.admin_notes : null, itemStatus: it ? it.status : null };
}
