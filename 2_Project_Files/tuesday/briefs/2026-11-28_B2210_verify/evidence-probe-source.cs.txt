// Each probe writes one line per shape to OUT; no id or credential VALUE is written (booleans, lengths, kinds only).

    private const string OUT = "/Volumes/Datasec_DEV/TUESDAY/2_Project_Files/fleet/state/s338/g2210/probe-out/probe.txt";
    private const string Setting = "SYNTHETIC_OXPD1_SETTING";
    private const string Password = "synthetic-admin-0001";
    private const string S = "http://www.w3.org/2003/05/soap-envelope";
    private const string AuthNs = "http://www.hp.com/schemas/imaging/OXPd/service/authentication/2013/03/01";
    private const string AccNs = "http://www.hp.com/schemas/imaging/OXPd/service/accessories/2010/04/14";
    private const string Id32 = "SYNTHETICcontext0000000000000001";
    private const string AccessoryId = "SYNTHETICaccessory00000000000001";
    private static readonly Guid AgentId = Guid.Parse("5a1e0000-0000-4000-8000-00000000c001");
    private static readonly Uri Callback = new("https://lab-host.example:8443/hid");
    private static readonly object Gate = new();

    private static CancellationToken Guard => TestContext.Current.CancellationToken;

    private static void W(string line)
    {
        lock (Gate)
        {
            Directory.CreateDirectory(Path.GetDirectoryName(OUT)!);
            File.AppendAllText(OUT, line + "\n");
        }
    }

    private static (Oxpd1Client Client, Func<int> Reads) Client(StubDevice stub, string? settingValue = Password, TimeSpan? timeout = null, string? recordFile = null)
    {
        int reads = 0;
        Oxpd1ClientOptions options = new("127.0.0.1", [TestCertificates.Sha256(stub.Certificate)], Setting)
        {
            SoapPort = stub.Port,
            TreePort = stub.Port,
            Timeout = timeout ?? TimeSpan.FromSeconds(10),
        };
        Oxpd1Client client = new(options, name =>
        {
            Interlocked.Increment(ref reads);
            return name == Setting ? settingValue : null;
        }, recordFile);
        return (client, () => Volatile.Read(ref reads));
    }

    private static string Envelope(string inner) => "<s:Envelope xmlns:s=\"" + S + "\"><s:Body>" + inner + "</s:Body></s:Envelope>";

    private static string EmptyAnswer(string ns, string operation) => Envelope("<" + operation + "Response xmlns=\"" + ns + "\"/>");

    private static string ReserveAnswerRaw(string resultXml) =>
        Envelope("<ReserveAuthenticationContextResponse xmlns=\"" + AuthNs + "\">" + resultXml + "</ReserveAuthenticationContextResponse>");

    private static string ReserveAnswer(string inside) => ReserveAnswerRaw("<ReserveAuthenticationContextResult>" + inside + "</ReserveAuthenticationContextResult>");

    private static string Kind(Oxpd1ContextKind k) => k.ToString();

    private static Oxpd1ContextCall[] Six() =>
    [
        Oxpd1ContextCall.ReserveAuthenticationContext(AgentId),
        Oxpd1ContextCall.ReleaseAuthenticationContext(Id32),
        Oxpd1ContextCall.HidOpen(AccessoryId),
        Oxpd1ContextCall.HidStartReading(AccessoryId, Callback, "synthetic-server-0001"),
        Oxpd1ContextCall.HidStopReading(AccessoryId),
        Oxpd1ContextCall.HidClose(AccessoryId),
    ];

    private static string AnswerFor(Oxpd1ContextKind k) => k == Oxpd1ContextKind.ReserveAuthenticationContext
        ? ReserveAnswer(Id32)
        : EmptyAnswer(k == Oxpd1ContextKind.ReleaseAuthenticationContext ? AuthNs : AccNs, k.ToString());

    // (a) the credential each of the six carries, and the password half of the guest header.
    [Fact]
    public async Task PA1_credential_of_each_executed_call()
    {
        foreach (Oxpd1ContextCall call in Six())
        {
            await using StubDevice stub = await StubDevice.StartAsync(behaviour: new StubBehaviour { ForceBody = AnswerFor(call.Kind) });
            (Oxpd1Client client, Func<int> reads) = Client(stub);
            using (client)
            {
                Oxpd1ContextResult r = await new Oxpd1ContextCaller(client).CallAsync(call with { Execute = true }, Guard);
                SeenRequest seen = Assert.Single(stub.Seen);
                string auth = seen.Authorization ?? string.Empty;
                string decoded = auth.StartsWith("Basic ", StringComparison.Ordinal) ? Encoding.UTF8.GetString(Convert.FromBase64String(auth[6..])) : "";
                int colon = decoded.IndexOf(':');
                string user = colon < 0 ? "(none)" : decoded[..colon];
                bool pwIsAdminSetting = colon >= 0 && decoded[(colon + 1)..] == Password;
                W($"PA1 {Kind(call.Kind)} | outcome {r.Outcome} applied {r.Applied} | requests 1 | scheme {(auth.StartsWith("Basic ", StringComparison.Ordinal) ? "Basic" : "other")} | user {user} | password half equals the client's AdminPasswordSetting value: {pwIsAdminSetting} | setting reads {reads()} | path {seen.Path}");
            }
        }
    }

    // (a) no request leaves without Execute: the six as made, each undo as returned, and an undo of an undo.
    [Fact]
    public async Task PA2_nothing_leaves_without_execute()
    {
        await using StubDevice stub = await StubDevice.StartAsync(behaviour: new StubBehaviour { ForceBody = ReserveAnswer(Id32) });
        (Oxpd1Client client, Func<int> reads) = Client(stub);
        using (client)
        {
            Oxpd1ContextCaller caller = new(client);
            foreach (Oxpd1ContextCall call in Six())
            {
                Oxpd1ContextResult r = await caller.CallAsync(call, Guard);
                W($"PA2 dry {Kind(call.Kind)} | DryRun {r.DryRun} | seen {stub.Seen.Count} | reads {reads()}");
            }

            Oxpd1ContextResult reserve = await caller.CallAsync(Six()[0] with { Execute = true }, Guard);
            int seenAfter = stub.Seen.Count;
            int readsAfter = reads();
            Oxpd1ContextResult undoDry = await caller.CallAsync(reserve.Undo!, Guard);
            W($"PA2 reserve executed then its undo called as returned | undo Execute {reserve.Undo!.Execute} | undo result DryRun {undoDry.DryRun} | seen before {seenAfter} after {stub.Seen.Count} | reads before {readsAfter} after {reads()}");

            stub.Behaviour.ForceBody = EmptyAnswer(AccNs, "HidOpen");
            Oxpd1ContextResult open = await caller.CallAsync(Six()[2] with { Execute = true }, Guard);
            int s2 = stub.Seen.Count;
            Oxpd1ContextResult closeDry = await caller.CallAsync(open.Undo!, Guard);
            W($"PA2 open executed then its undo called as returned | undo Execute {open.Undo!.Execute} | DryRun {closeDry.DryRun} | seen before {s2} after {stub.Seen.Count}");
        }
    }

    // (a) every public member of the lane's public types (anything else that could send).
    [Fact]
    public void PA4_public_surface()
    {
        foreach (Type t in typeof(Oxpd1ContextCaller).Assembly.GetExportedTypes().Where(t => t.Name.StartsWith("Oxpd1Context", StringComparison.Ordinal)).OrderBy(t => t.Name))
        {
            IEnumerable<string> members = t.GetMembers(BindingFlags.Public | BindingFlags.Instance | BindingFlags.Static | BindingFlags.DeclaredOnly)
                .Select(m => m.MemberType + " " + m.Name).Distinct().OrderBy(x => x, StringComparer.Ordinal);
            W($"PA4 {t.Name} ({(t.IsEnum ? "enum" : t.IsSealed ? "sealed" : "open")}) | " + string.Join(", ", members));
            IEnumerable<string> ctors = t.GetConstructors(BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Instance)
                .Select(c => (c.IsPublic ? "public" : c.IsAssembly ? "internal" : c.IsPrivate ? "private" : c.IsFamily ? "protected" : "other") + "(" + string.Join(",", c.GetParameters().Select(p => p.ParameterType.Name)) + ")");
            W($"PA4 {t.Name} constructors | " + string.Join(", ", ctors));
        }

        FieldInfo[] fields = typeof(Oxpd1ContextCaller).GetFields(BindingFlags.Instance | BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic);
        W("PA4 Oxpd1ContextCaller fields | " + string.Join(", ", fields.Select(f => (f.IsStatic ? "static " : "") + (f.IsInitOnly ? "readonly " : "") + f.FieldType.Name + " " + f.Name)));
    }

    // (a) two concurrent executed reserves over one caller: each its own result.
    [Fact]
    public async Task PA5_two_concurrent_calls()
    {
        await using StubDevice stub = await StubDevice.StartAsync(behaviour: new StubBehaviour { ForceBody = ReserveAnswer(Id32), Delay = TimeSpan.FromMilliseconds(300) });
        (Oxpd1Client client, _) = Client(stub);
