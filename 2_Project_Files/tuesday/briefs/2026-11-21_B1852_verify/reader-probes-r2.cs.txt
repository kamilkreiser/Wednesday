internal static class ReaderReport
{
    private static readonly object Sync = new();
    internal const string FilePath = "/private/tmp/b1852-gate/out/probes.txt";

    internal static void Line(string text)
    {
        lock (Sync)
        {
            Directory.CreateDirectory(Path.GetDirectoryName(FilePath)!);
            File.AppendAllText(FilePath, DateTime.UtcNow.ToString("HH:mm:ss.fff") + " " + text + "\n");
        }

internal sealed class Texts
{
    private readonly List<string> _all = new();

    internal void Add(object? value)
    {
        if (value is null)
        {
            return;
        }

        lock (_all)
        {
            _all.Add(value.ToString() ?? string.Empty);
            if (value is Exception e)
            {
                _all.Add(e.Message);
            }
        }
    }

    internal static List<byte[]> Forms(byte[] needle)
    {
        var forms = RawScan.FormsOf(needle);
        forms.Add(Encoding.Unicode.GetBytes(Encoding.Latin1.GetString(needle)));
        forms.Add(Encoding.BigEndianUnicode.GetBytes(Encoding.Latin1.GetString(needle)));
        // base64 of the needle at each alignment inside a longer stream: the full groups only.
        for (var skip = 0; skip < 3 && needle.Length - skip >= 6; skip++)
        {
            var part = needle.AsSpan(skip).ToArray();
            var b64 = Convert.ToBase64String(part);
            var whole = (part.Length / 3) * 4;
            var core = b64.Substring(4, whole - 8);
            if (core.Length >= 8)
            {
                forms.Add(Encoding.ASCII.GetBytes(core));
                forms.Add(Encoding.ASCII.GetBytes(core.Replace('+', '-').Replace('/', '_')));
            }
        }

        return forms;
    }

    internal int Check(IEnumerable<byte[]> needles, string cell)
    {
        List<string> copy;
        lock (_all)
        {
            copy = _all.ToList();
        }

        var i = 0;
        foreach (var needle in needles)
        {
            foreach (var form in Forms(needle))
            {
                var asText = Encoding.Latin1.GetString(form);
                foreach (var t in copy)
                {
                    Assert.False(t.Contains(asText, StringComparison.OrdinalIgnoreCase), cell + ": needle " + i + " found in a text");
                }
            }

            i++;
        }

        return copy.Count;
    }

    internal static bool FolderHolds(string folder, byte[] needle)
    {
        var forms = Forms(needle);
        foreach (var file in Directory.EnumerateFiles(folder, "*", SearchOption.AllDirectories))
        {
            var content = File.ReadAllBytes(file);
            foreach (var form in forms)
            {
                if (content.AsSpan().IndexOf(form) >= 0)
                {
                    return true;
                }
            }
        }

        return false;
    }

    internal static void StoreClean(StoreRoot root, string cell, IEnumerable<byte[]> needles)
    {
        var marker = Encoding.ASCII.GetBytes("reader-control-" + Guid.NewGuid().ToString("N"));
        var control = Path.Combine(root.Root, "reader-control.txt");
        Directory.CreateDirectory(root.Root);
        File.WriteAllBytes(control, marker);
        try
        {
            Assert.True(FolderHolds(root.Root, marker), cell + ": control not found");
            var i = 0;
            foreach (var n in needles)
            {
                Assert.False(FolderHolds(root.Root, n), cell + ": needle " + i + " in clear under the store root");
                i++;
            }
        }
        finally
        {
            File.Delete(control);
        }
    }
}

internal sealed class BlockingRefresher : ITokenRefresher
{
    internal readonly ManualResetEventSlim Gate = new(false);
    internal int Calls;
    internal Func<TokenGrant>? Answer;

    public Task<TokenGrant> RefreshAsync(SessionKey session, ReadOnlyMemory<byte> renewalState, string subject, CancellationToken cancellationToken)
    {
        Interlocked.Increment(ref Calls);
        Gate.Wait(TimeSpan.FromSeconds(20));
        return Task.FromResult(Answer!());
    }
}

internal sealed class ScriptRefresher : ITokenRefresher
{
    internal Func<SessionKey, ReadOnlyMemory<byte>, string, CancellationToken, Task<TokenGrant>> Body = (_, _, _, _) => Task.FromResult<TokenGrant>(null!);
    internal int Calls;
    internal readonly List<ReadOnlyMemory<byte>> Renewals = new();

    public Task<TokenGrant> RefreshAsync(SessionKey session, ReadOnlyMemory<byte> renewalState, string subject, CancellationToken cancellationToken)
    {
        Interlocked.Increment(ref Calls);
        lock (Renewals)
        {
            Renewals.Add(renewalState);
        }

        return Body(session, renewalState, subject, cancellationToken);
    }
}

internal sealed class KeepingStore : ISessionRecordStore
{
    private readonly ISessionRecordStore _inner;
    internal readonly List<byte[]> Loaded = new();
    internal bool FailList;

    internal KeepingStore(ISessionRecordStore inner)
    {
        _inner = inner;
    }

    public void Save(string id, ReadOnlySpan<byte> record) => _inner.Save(id, record);

    public bool TryLoad(string id, out byte[]? record)
    {
        var r = _inner.TryLoad(id, out record);
        if (record is not null)
        {
            Loaded.Add(record);
        }

        return r;
    }

    public bool Delete(string id) => _inner.Delete(id);

    public IReadOnlyList<string> ListIds() => FailList ? throw new IOException("planted listing failure") : _inner.ListIds();
}

    private static async Task<(AccessToken? Token, TokenBrokerException? Refusal)> Try(Func<Task<AccessToken>> call, Texts texts)
    {
        try
        {
            var t = await call();
            texts.Add(t);
            return (t, null);
        }
        catch (TokenBrokerException e)
        {
            texts.Add(e);
            Assert.Null(e.InnerException);
            return (null, e);
        }
    }

    private static TokenBrokerException? TrySync(Action call, Texts texts)
    {
        try
        {
            call();
            return null;
        }
        catch (TokenBrokerException e)
        {
            texts.Add(e);
            Assert.Null(e.InnerException);
            return e;
        }
    }

    private static List<string> NearSubjects()

    [Fact]
    public async Task R_hanging_late_and_boundary_refreshers_are_bounded_on_the_broker_clock()
    {
        var texts = new Texts();
        // answers just before the bound: accepted
        using (var kit = new Kit())
        {
            kit.SignIn(1);
            var release = new TaskCompletionSource();
            using var answer = new TokenGrant(Data.Access(31), Data.Renewal(31), Data.User(1), kit.Now + Data.Life);
            kit.Refresher.Answer = FakeRefresher.IgnoresCancellation(release.Task, new TokenGrant(Data.Access(31), Data.Renewal(31), Data.User(1), kit.Now + Data.Life));
            var call = kit.Broker.GetNewTokenAsync(Data.Key(1), Ct);
            await Expect.Eventually(() => kit.Clock.LiveTimers == 1, "bound");
            kit.Clock.Advance(TimeSpan.FromSeconds(30) - TimeSpan.FromTicks(1));
            Assert.False(call.IsCompleted);
            release.SetResult();
            using var got = await Expect.Guarded(call, "bound");
            Assert.True(got.AsSpan().SequenceEqual(Data.Access(31)), "bound: an answer one tick before the bound was not accepted");
        }

        // at exactly the bound: timed out; the late answer is disposed and never stored
        using (var kit = new Kit())
        {
            kit.SignIn(1);
            var release = new TaskCompletionSource();
            var late = new TokenGrant(Data.Access(32), Data.Renewal(32), Data.User(1), kit.Now + Data.Life);
            kit.Refresher.Answer = FakeRefresher.IgnoresCancellation(release.Task, late);
            var call = kit.Broker.GetNewTokenAsync(Data.Key(1), Ct);
            await Expect.Eventually(() => kit.Clock.LiveTimers == 1, "exact");
            kit.Clock.Advance(TimeSpan.FromSeconds(30));
            await Expect.GuardedRefusal(TokenBrokerError.RefreshTimedOut, call, "exact");
            Assert.True(kit.Refresher.TokenOfCall(0).IsCancellationRequested);
            release.SetResult();
            await Expect.Eventually(() => Expect.AllZero(late.AccessBytes), "exact late disposed");
            kit.Restart();
            using var held = await kit.Broker.GetTokenAsync(Data.Key(1), Ct);
            Assert.True(held.AsSpan().SequenceEqual(Data.Access(1)), "exact: the late answer was stored");
            Texts.StoreClean(kit.Root, "late", new[] { Data.Access(32), Data.Renewal(32) });
        }

        // a re-entrant refresher that awaits the broker's own refresh of the same session: bounded
        using (var kit = new Kit())
        {
            kit.SignIn(1);
            var inner = 0;
            kit.Refresher.Answer = async ct =>
            {
                Interlocked.Increment(ref inner);
                using var t = await kit.Broker.GetNewTokenAsync(Data.Key(1), CancellationToken.None);
                return new TokenGrant(Data.Access(33), Data.Renewal(33), Data.User(1), kit.Now + Data.Life);
            };
            var call = kit.Broker.GetNewTokenAsync(Data.Key(1), Ct);
            await Expect.Eventually(() => kit.Clock.LiveTimers == 1, "reentrant");
            kit.Clock.Advance(TimeSpan.FromSeconds(31));
            await Expect.GuardedRefusal(TokenBrokerError.RefreshTimedOut, call, "reentrant");
            Assert.Equal(1, kit.Refresher.Calls);
            using var held = await kit.Broker.GetTokenAsync(Data.Key(1), Ct);
            Assert.True(held.AsSpan().SequenceEqual(Data.Access(1)), "reentrant: held grant changed");
        }

        ReaderReport.Line("H8: an answer 1 tick before the bound accepted; at the bound RefreshTimedOut, refresher token cancelled, late answer zeroed and not stored (restart reads the old token); a re-entrant refresher bounded (1 call)");

    private static string Hex() => Convert.ToHexStringLower(RandomNumberGenerator.GetBytes(16));

    private static byte[] Record(IEnumerable<(string Name, string Raw)> members)
    {
        return Encoding.UTF8.GetBytes("{" + string.Join(",", members.Select(m => "\"" + m.Name + "\":" + m.Raw)) + "}");
    }

    private static List<(string Name, string Raw)> Template(string id, int n, DateTimeOffset now, SessionPlatform platform = SessionPlatform.Oxpd2)
    {
        var hash = platform == SessionPlatform.Oxpd1 ? "\"" + Convert.ToBase64String(SHA256.HashData(Encoding.ASCII.GetBytes("handle-" + n))) + "\"" : "null";
        return new List<(string, string)>
        {
            ("format", "1"),
            ("id", "\"" + id + "\""),
            ("serial", "\"" + Data.Serial(n) + "\""),
            ("platform", ((int)platform).ToString()),
            ("platformSession", "\"" + Data.PlatformSession(n) + "\""),
            ("handleSha256", hash),
            ("user", "\"" + Data.User(n) + "\""),
            ("idleMilliseconds", "1800000"),
            ("access", "\"" + Convert.ToBase64String(Data.Access(n)) + "\""),
            ("renewal", "\"" + Convert.ToBase64String(Data.Renewal(n)) + "\""),
            ("subject", "\"" + Data.User(n) + "\""),
            ("expiresAt", (now + Data.Life).ToUnixTimeMilliseconds().ToString()),
            ("lastActivity", (now - TimeSpan.FromMinutes(1)).ToUnixTimeMilliseconds().ToString()),
            ("written", (now - TimeSpan.FromMinutes(1)).ToUnixTimeMilliseconds().ToString()),
        };
    }

    private static List<(string Name, string Raw)> With(List<(string Name, string Raw)> t, string name, string? raw)
    {
        var copy = t.ToList();
        var i = copy.FindIndex(m => m.Name == name);
        if (raw is null)
        {
            copy.RemoveAt(i);
        }
        else
        {
            copy[i] = (name, raw);
        }

        return copy;
    }

            try

                store = kit.Root.Open();

                }
                else

                    startedOther++;

                {
                    Assert.True(File.ReadAllBytes(f).AsSpan().SequenceEqual(c), "flip " + i + ": a record file changed on a refused start");

            }
        }

            kit.Faulty!.FailSave = true;
            var r = TrySync(() => kit.SignIn(1), texts);
            Assert.Equal(TokenBrokerError.StoreFailed, r!.Reason);
            Assert.Null(kit.Broker.StateOf(Data.Key(1)));
            Assert.Empty(kit.Events);
        }

        // write at an accepted refresh and at activity
        using (var kit = new Kit(faulty: true))
        {
            kit.SignIn(1);
            kit.Faulty!.FailSave = true;
            kit.Refresher.Answer = FakeRefresher.Gives(() => new TokenGrant(Data.Access(71), Data.Renewal(71), Data.User(1), kit.Now + Data.Life));
            using (var t = await kit.Broker.GetNewTokenAsync(Data.Key(1), Ct))
            {
                Assert.True(t.AsSpan().SequenceEqual(Data.Access(71)), "write fail at refresh");
            }

            using (var t = await kit.Broker.GetTokenAsync(Data.Key(1), Ct))
            {
                Assert.True(t.AsSpan().SequenceEqual(Data.Access(71)), "write fail at activity");
            }

            kit.Faulty.FailSave = false;
            kit.Broker.Touch(Data.Key(1));
            kit.Restart();
            using var after = await kit.Broker.GetTokenAsync(Data.Key(1), Ct);
            Assert.True(after.AsSpan().SequenceEqual(Data.Access(71)), "the healed write did not carry the new grant");
        }

        // delete at sign-out, at a new sign-in, at expiry by a read and by Sweep
        string[] where = { "sign-out", "new sign-in", "expiry read", "expiry sweep", "subject at refresh" };
        foreach (var w in where)
        {
            using var kit = new Kit(faulty: true);
            kit.SignIn(1, idle: TimeSpan.FromSeconds(10));
            kit.Faulty!.FailDelete = true;
            TokenBrokerException? r = null;
            switch (w)
            {
                case "sign-out":
                    r = TrySync(() => kit.Broker.SignOut(Data.Key(1)), texts);
                    break;
                case "new sign-in":
                    r = TrySync(() => kit.SignIn(1), texts);
                    break;
                case "expiry read":
                    kit.Clock.Advance(TimeSpan.FromSeconds(11));
                    r = (await Try(() => kit.Broker.GetTokenAsync(Data.Key(1), Ct), texts)).Refusal;
                    break;
                case "expiry sweep":
                    kit.Clock.Advance(TimeSpan.FromSeconds(11));
                    r = TrySync(() => kit.Broker.Sweep(), texts);
                    break;
                default:
                    kit.Refresher.Answer = FakeRefresher.Gives(() => new TokenGrant(Data.Access(72), Data.Renewal(72), Data.User(2), kit.Now + Data.Life));
                    r = (await Try(() => kit.Broker.GetNewTokenAsync(Data.Key(1), Ct), texts)).Refusal;
                    break;
            }

            var live = kit.Broker.StateOf(Data.Key(1)) is not null;
            var recordsWhileFailing = kit.SessionNames().Count;
            kit.Faulty.FailDelete = false;
            kit.Broker.Sweep();
            ReaderReport.Line("H17 delete failure at " + w + ": the call gave " + (r?.Reason.ToString() ?? "no refusal") + "; a session live for device 1 afterwards " + live + "; records while failing " + recordsWhileFailing + "; after healing and Sweep " + kit.SessionNames().Count);
            Assert.NotNull(r);
            Assert.Empty(kit.SessionNames());
        }

        // read at restore
        using (var kit = new Kit())
        {
            kit.SignIn(1);
            kit.SignIn(2);
            kit.CloseBroker();
            var failing = new FakeRecordStore(new SecretStoreSessionRecordStore(kit.Store)) { FailLoad = true };
            using var b2 = new TokenBroker(new TokenBrokerOptions(), failing, kit.Refresher, kit.Clock);
            var lr = b2.LastRestore;
            ReaderReport.Line("H17 read failure at restore: LastRestore restored " + lr.Restored + " unreadable " + lr.Unreadable + " deleted " + lr.Deleted + "; session records left " + kit.SessionNames().Count);
            Assert.Equal(2, lr.Unreadable);
            Assert.Equal(2, lr.Deleted);
            Assert.Empty(kit.SessionNames());
        }

        // listing at restore
        var root = new StoreRoot();
        try
        {
            using var store = root.Open();
            var keeping = new KeepingStore(new SecretStoreSessionRecordStore(store)) { FailList = true };
            var e = Assert.Throws<TokenBrokerException>(() => new TokenBroker(new TokenBrokerOptions(), keeping, new FakeRefresher(), new TestClock()));
            Assert.Equal(TokenBrokerError.StoreFailed, e.Reason);
            Assert.Null(e.InnerException);
        }
        finally
        {
            root.Dispose();
        }

        texts.Check(new[] { Data.Access(1), Data.Access(71), Encoding.UTF8.GetBytes(Data.User(1)) }, "store failures");
    }

    // ---------- item 3: raw store files at each stage; disposal clears ----------
    [Fact]
    public async Task R_raw_store_files_hold_no_clear_secret_at_any_stage_and_disposal_clears_every_array()
    {
        using var kit = new Kit();
        var a = kit.SignIn(1, SessionPlatform.Oxpd1);
        var handle = a.Handle!.Value.Value;
        var needles = new List<byte[]>
        {
            Data.Access(1), Data.Renewal(1), Encoding.UTF8.GetBytes(Data.User(1)), Encoding.UTF8.GetBytes(Data.PlatformSession(1)),
            Encoding.UTF8.GetBytes(Data.Serial(1)), Encoding.ASCII.GetBytes(handle), Base64Url.DecodeFromChars(handle),
            SHA256.HashData(Encoding.ASCII.GetBytes(handle)),
        };
        Texts.StoreClean(kit.Root, "after sign-in", needles);
        var state1 = kit.Broker.StateOf(Data.Key(1, SessionPlatform.Oxpd1))!;
        var held1 = state1.Grant;
        kit.Refresher.Answer = FakeRefresher.Gives(() => new TokenGrant(Data.Access(81), Data.Renewal(81), Data.User(1), kit.Now + Data.Life));
        using (await kit.Broker.GetNewTokenAsync(Data.Key(1, SessionPlatform.Oxpd1), Ct))
        {
        }

        Assert.True(Expect.AllZero(held1.Access) && Expect.AllZero(held1.Renewal), "the replaced grant was not cleared");
        needles.Add(Data.Access(81));
        needles.Add(Data.Renewal(81));
        Texts.StoreClean(kit.Root, "after refresh", needles);
        var held2 = kit.Broker.StateOf(Data.Key(1, SessionPlatform.Oxpd1))!.Grant;
        Assert.True(kit.Broker.SignOut(Data.Key(1, SessionPlatform.Oxpd1)));
        Assert.True(Expect.AllZero(held2.Access) && Expect.AllZero(held2.Renewal), "the ended grant was not cleared");
        Texts.StoreClean(kit.Root, "after sign-out", needles);
        kit.SignIn(2, idle: TimeSpan.FromSeconds(5));
        kit.Clock.Advance(TimeSpan.FromSeconds(6));
        kit.Broker.Sweep();
        kit.SignIn(3);
        kit.SignIn(3);
        kit.SignIn(4);
        needles.AddRange(new[] { Data.Access(2), Data.Access(3), Data.Access(4), Data.Renewal(4), Encoding.UTF8.GetBytes(Data.User(4)) });
        Texts.StoreClean(kit.Root, "after expiry and new sign-in", needles);
        var held4 = kit.Broker.StateOf(Data.Key(4))!.Grant;
        var held3 = kit.Broker.StateOf(Data.Key(3))!.Grant;
        kit.CloseBroker();
        Assert.True(Expect.AllZero(held4.Access) && Expect.AllZero(held3.Access), "broker disposal did not clear its grants");
        kit.ReopenBroker();
        Texts.StoreClean(kit.Root, "after restart", needles);
        // grant and access token disposal
        var g = new TokenGrant(Data.Access(5), Data.Renewal(5), Data.User(5), kit.Now + Data.Life);
        var ga = g.AccessBytes;
        var gr = g.RenewalBytes;
        g.Dispose();
        Assert.True(Expect.AllZero(ga) && Expect.AllZero(gr), "grant disposal");
        var t = await kit.Broker.GetTokenAsync(Data.Key(4), Ct);
        var tb = t.Bytes;
        t.Dispose();
        Assert.True(Expect.AllZero(tb), "access token disposal");
        // the renewal copy given to the refresher, the refresher's answer, and the arrays the store hands back at restore
        var root = new StoreRoot();
        try
        {
            var clock = new TestClock();
            using var store = root.Open();
            var keeping = new KeepingStore(new SecretStoreSessionRecordStore(store));
            var script = new ScriptRefresher();
            TokenGrant? answer = null;
            script.Body = (_, _, _, _) => Task.FromResult(answer = new TokenGrant(Data.Access(91), Data.Renewal(91), Data.User(1), clock.GetUtcNow() + Data.Life));
            using (var broker = new TokenBroker(new TokenBrokerOptions(), keeping, script, clock))
            {
                using (var gg = new TokenGrant(Data.Access(1), Data.Renewal(1), Data.User(1), clock.GetUtcNow() + Data.Life))
                {
                    broker.SignIn(Data.Key(1), Data.User(1), Data.Idle, gg);
                }

                using (await broker.GetNewTokenAsync(Data.Key(1), Ct))
                {
                }

                Assert.True(script.Renewals.All(m => m.ToArray().All(x => x == 0)), "the renewal copy given to the refresher was not cleared");
                Assert.True(Expect.AllZero(answer!.AccessBytes) && Expect.AllZero(answer.RenewalBytes), "the refresher's answer was not cleared");
            }

            using (new TokenBroker(new TokenBrokerOptions(), keeping, script, clock))
            {
                Assert.NotEmpty(keeping.Loaded);
                Assert.True(keeping.Loaded.All(x => x.All(b => b == 0)), "the record bytes read at restore were not cleared");
            }
        }
        finally
        {
            root.Dispose();
        }

        ReaderReport.Line("item 3: raw store files clean after sign-in, refresh, sign-out, expiry, new sign-in, restart (" + needles.Count + " needles x " + Texts.Forms(Data.Access(1)).Count + " forms each incl. UTF-16 and base64 at 3 alignments); zero after disposal: replaced grant, ended grant, broker grants, TokenGrant, AccessToken, refresher renewal copy, refresher answer, record bytes at restore");
    }

    // ---------- D12: the record buffer growth ----------
    [Theory]
    [InlineData("ascii", 1)]
    [InlineData("ascii", 64)]
    [InlineData("plus", 40)]
    [InlineData("plus", 60)]
    [InlineData("accented", 40)]
    public void R_the_record_writer_grows_its_buffer_only_when_the_estimate_is_short(string kind, int length)
    {
        using var kit = new Kit();
        var user = kind switch
        {
            "ascii" => new string('u', length),
            "plus" => new string('+', length),
            _ => new string('é', length),
        };
        using (var g = new TokenGrant(Data.Access(1), Data.Renewal(1), user, kit.Now + Data.Life))
        {
            kit.Broker.SignIn(Data.Key(1), user, Data.Idle, g);
        }

        var state = kit.Broker.StateOf(Data.Key(1))!;
        var capacity = 1024 + (((state.Grant.Access.Length + state.Grant.Renewal.Length) / 3) + 2) * 4 + (user.Length * 2) + (user.Length * 2);
        SessionRecord.Write(state, kit.Now);
        var before = GC.GetAllocatedBytesForCurrentThread();
        var bytes = SessionRecord.Write(state, kit.Now);
        var allocated = GC.GetAllocatedBytesForCurrentThread() - before;
        var growth = allocated - capacity - bytes.Length;
        SessionRecord.Write(state, kit.Now, out var report);
        ReaderReport.Line("D12 writer report: user " + kind + " x" + length + ": growths " + report.Growths + ", every retired buffer cleared " + report.EveryRetiredCleared);
        Assert.True(report.EveryRetiredCleared, "D12: a retired buffer was not cleared");
        ReaderReport.Line("D12 record writer: user " + kind + " x" + length + ": estimate " + capacity + " B, record " + bytes.Length + " B, allocated " + allocated + " B, beyond estimate+record " + growth + " B" + (growth > 2048 ? " => the buffer GREW (an uncleared copy of the bytes written so far is left to the collector)" : string.Empty));
    }

    // ---------- seeded model-based fuzz ----------
    [Theory]
    [InlineData(1)]
    [InlineData(2)]
    [InlineData(3)]
    [InlineData(4)]
    [InlineData(5)]
    [InlineData(6)]
    [InlineData(7)]
    [InlineData(8)]
    [InlineData(9)]
    [InlineData(10)]
    [InlineData(11)]
    [InlineData(12)]
    [InlineData(13)]
    [InlineData(14)]
    [InlineData(15)]
    [InlineData(16)]
    [InlineData(17)]
    [InlineData(18)]
    [InlineData(19)]
    [InlineData(20)]
    public async Task R_seeded_operations_never_hand_out_a_token_of_another_subject_or_leave_a_record_of_an_ended_session(int seed)
    {
        var rng = new Random(seed * 7919);
        using var kit = new Kit();
        var live = new Dictionary<string, (SessionKey Key, int User, string Id, SessionHandle? Handle, long ActMono, DateTimeOffset ActWall, TimeSpan Idle)>(StringComparer.Ordinal);
        var ended = new List<SessionHandle>();
        var counter = 0;
        var tokensOut = 0;
        var refusals = new Dictionary<TokenBrokerError, int>();
        var texts = new Texts();
        var expiry = new Dictionary<string, (DateTimeOffset Exp, long RecvMono, DateTimeOffset RecvWall)>(StringComparer.Ordinal);
        var restoredIds = new HashSet<string>(StringComparer.Ordinal);
        var lengthened = 0;
        long maxLater = 0;
        var restarts = 0;
        long Mono() => kit.Clock.GetTimestamp();
        void SyncEvents(List<SessionEventArgs> seen)
        {
            foreach (var e in seen)
            {
                if (e.Kind == SessionEventKind.SignedOut)
                {
                    var hit = live.Where(p => p.Value.Id == e.SessionId).Select(p => p.Key).ToList();
                    foreach (var h in hit)
                    {
                        if (live[h].Handle is { } hh)
                        {
                            ended.Add(hh);
                        }

                        live.Remove(h);
                    }
                }
            }
        }

        byte[] Tok(int user, int n) => Encoding.UTF8.GetBytes("tk|" + Data.User(user) + "|" + n.ToString("D6"));
        void CheckToken(AccessToken t, string serial)
        {
            var text = Encoding.UTF8.GetString(t.AsSpan());
            Assert.True(live.ContainsKey(serial), "seed " + seed + ": a token for a device with no live session");
            var expected = "tk|" + Data.User(live[serial].User) + "|";
            Assert.True(text.StartsWith(expected, StringComparison.Ordinal), "seed " + seed + ": a token of another subject was handed out");
            var (exp, recvMono, recvWall) = expiry[text];
            Assert.True(exp > kit.Now, "seed " + seed + ": an expired token (wall) was handed out");
            var s = live[serial];
            if (!restoredIds.Contains(s.Id))
            {
                Assert.True((exp - recvWall) - kit.Clock.GetElapsedTime(recvMono) > TimeSpan.Zero, "seed " + seed + ": an expired token (monotonic) was handed out");
            }

            Assert.True(kit.Clock.GetElapsedTime(s.ActMono) < s.Idle && kit.Now - s.ActWall < s.Idle, "seed " + seed + ": a token of an idle session was handed out");
            tokensOut++;
        }

        for (var step = 0; step < 300; step++)
        {
            var mark = kit.Events.Count;
            var d = rng.Next(5);
            var serial = Data.Serial(d);
            var op = rng.Next(10);
            if (op == 0 || !live.ContainsKey(serial) && op < 4)
            {
                var user = rng.Next(4);
                var platform = rng.Next(2) == 0 ? SessionPlatform.Oxpd1 : SessionPlatform.Oxpd2;
                var key = new SessionKey(serial, platform, "ps-" + rng.Next(3));
                var idle = TimeSpan.FromMinutes(1 + rng.Next(40));
                var statedUser = rng.Next(10) == 0 ? (user + 1) % 4 : user;
                var n = ++counter;
                var exp = kit.Now + TimeSpan.FromMinutes(rng.Next(-5, 90));
                using var g = new TokenGrant(Tok(statedUser, n), Encoding.UTF8.GetBytes("rn|" + n), Data.User(statedUser), exp);
                expiry[Encoding.UTF8.GetString(Tok(statedUser, n))] = (exp, Mono(), kit.Now);
                SignInResult? res = null;
                var r = TrySync(() => res = kit.Broker.SignIn(key, Data.User(user), idle, g), texts);
                SyncEvents(kit.Events.Skip(mark).ToList());
                if (r is null)
                {
                    Assert.Equal(user, statedUser);
                    live[serial] = (key, user, res!.SessionId, res.Handle, Mono(), kit.Now, idle);
                    texts.Add(res);
                }
                else
                {
                    Assert.True(r.Reason == TokenBrokerError.SubjectMismatch && statedUser != user, "seed " + seed + ": sign-in refused " + r.Reason);
                }
            }
            else if (op <= 3)
            {
                var s = live.TryGetValue(serial, out var x) ? x : default;
                var byHandle = s.Handle is not null && rng.Next(2) == 0;
                var wrongKey = rng.Next(8) == 0;
                var key = wrongKey ? s.Key with { PlatformSessionId = "ps-9" } : s.Key;
                var force = op == 3;
                var behaviour = rng.Next(8);
                var n = ++counter;
                var answerUser = behaviour == 1 ? (s.User + 1) % 4 : s.User;
                var answerExp = behaviour == 2 ? kit.Now - TimeSpan.FromSeconds(1) : kit.Now + TimeSpan.FromMinutes(5 + rng.Next(60));
                expiry[Encoding.UTF8.GetString(Tok(answerUser, n))] = (answerExp, Mono(), kit.Now);
                var release = new TaskCompletionSource();
                kit.Refresher.Answer = behaviour switch
                {
                    3 => FakeRefresher.Throws("synthetic"),
                    4 => _ => Task.FromResult<TokenGrant>(null!),
                    5 => FakeRefresher.HangsUntilCancelled(),
                    6 => FakeRefresher.IgnoresCancellation(release.Task, new TokenGrant(Tok(answerUser, n), Encoding.UTF8.GetBytes("rn|" + n), Data.User(answerUser), answerExp)),
                    _ => FakeRefresher.Gives(() => new TokenGrant(Tok(answerUser, n), Encoding.UTF8.GetBytes("rn|" + n), Data.User(answerUser), answerExp)),
                };
                var callsBefore = kit.Refresher.Calls;
                Task<AccessToken> call;
                try
                {
                    call = byHandle
                        ? (force ? kit.Broker.GetNewTokenAsync(s.Handle!.Value, serial, Ct) : kit.Broker.GetTokenAsync(s.Handle!.Value, serial, Ct))
                        : (force ? kit.Broker.GetNewTokenAsync(key.DeviceSerial is null ? Data.Key(d) : key, Ct) : kit.Broker.GetTokenAsync(key.DeviceSerial is null ? Data.Key(d) : key, Ct));
                }
                catch (TokenBrokerException e)
                {
                    call = Task.FromException<AccessToken>(e);
                }

                if (behaviour is 5 or 6)
                {
                    // Round 2 adaptation: the refresher's call now starts on the thread pool, so wait for it to enter
                    // and for its bound to be armed before moving the clock (the same wait the lane's cells added).
                    await Expect.Eventually(() => call.IsCompleted || (kit.Refresher.Calls > callsBefore && kit.Clock.LiveTimers > 0), "seed timer");
                    if (!call.IsCompleted)
                    {
                        kit.Clock.Advance(TimeSpan.FromSeconds(30));
                    }
                }

                var (t, r) = await Try(() => Expect.Guarded(call, "seed " + seed), texts);
                release.TrySetResult();
                SyncEvents(kit.Events.Skip(mark).ToList());
                if (t is not null)
                {
                    CheckToken(t, serial);
                    var cur = live[serial];
                    live[serial] = (cur.Key, cur.User, cur.Id, cur.Handle, Mono(), kit.Now, cur.Idle);
                    t.Dispose();
                }
                else
                {
                    refusals[r!.Reason] = refusals.TryGetValue(r.Reason, out var c) ? c + 1 : 1;
                }
            }
            else if (op == 4)
            {
                if (live.TryGetValue(serial, out var s) && s.Handle is not null && rng.Next(2) == 0)
                {
                    kit.Broker.SignOut(s.Handle.Value, serial);
                }
                else if (ended.Count > 0 && rng.Next(3) == 0)
                {
                    Assert.False(kit.Broker.SignOut(ended[rng.Next(ended.Count)], serial), "seed " + seed + ": an ended handle signed out a session");
                }
                else
                {
                    kit.Broker.SignOut(live.TryGetValue(serial, out var s2) ? s2.Key : Data.Key(d));
                }

                SyncEvents(kit.Events.Skip(mark).ToList());
            }
            else if (op == 5)
            {
                var r = TrySync(() => kit.Broker.Touch(live.TryGetValue(serial, out var s) ? s.Key : Data.Key(d)), texts);
                SyncEvents(kit.Events.Skip(mark).ToList());
                if (r is null)
                {
                    var cur = live[serial];
                    live[serial] = (cur.Key, cur.User, cur.Id, cur.Handle, Mono(), kit.Now, cur.Idle);
                }
            }
            else if (op == 6)
            {
                kit.Broker.Sweep();
                SyncEvents(kit.Events.Skip(mark).ToList());
            }
            else if (op == 7)
            {
                var m = rng.Next(4);
                var by = TimeSpan.FromSeconds(rng.Next(1, 1200));
                if (m == 0)
                {
                    kit.Clock.Advance(by);
                }
                else if (m == 1)
                {
                    kit.Clock.MoveWall(rng.Next(2) == 0 ? by : -by);
                }
                else
                {
                    kit.Clock.AdvanceMonotonic(by);
                }
            }
            else if (op == 8 && rng.Next(4) == 0)
            {
                kit.Restart();
                restarts++;
                foreach (var p in live.ToList())
                {
                    var st = kit.Broker.StateOf(p.Value.Key);
                    if (st is null)
                    {
                        if (p.Value.Handle is { } hh)
                        {
                            ended.Add(hh);
                        }

                        live.Remove(p.Key);
                    }
                    else
                    {
                        Assert.True(st.SessionUser == Data.User(p.Value.User) && st.Id == p.Value.Id, "seed " + seed + ": a restored session changed user or id");
                        var later = st.LastActivityTimestamp - p.Value.ActMono;
                        if (later > 0)
                        {
                            lengthened++;
                            maxLater = Math.Max(maxLater, later);
                        }

                        restoredIds.Add(st.Id);
                        live[p.Key] = (p.Value.Key, p.Value.User, p.Value.Id, p.Value.Handle, st.LastActivityTimestamp, st.LastActivityWall, p.Value.Idle);
                    }
                }
            }

            // the store holds exactly the records of the live sessions (no record of an ended session, none written after its delete)
            var names = kit.SessionNames().Select(nm => nm.Substring(9)).OrderBy(x => x, StringComparer.Ordinal).ToList();
            var ids = live.Values.Select(v => v.Id).OrderBy(x => x, StringComparer.Ordinal).ToList();
            Assert.True(names.SequenceEqual(ids), "seed " + seed + " step " + step + ": store names differ from the live sessions (" + names.Count + " vs " + ids.Count + ")");
        }

        texts.Check(Enumerable.Range(0, 4).Select(u => Encoding.UTF8.GetBytes(Data.User(u))).Append(Encoding.UTF8.GetBytes("tk|")), "seed " + seed);
        ReaderReport.Line("FUZZ seed " + seed + ": 300 steps, tokens handed out " + tokensOut + ", refusals " + string.Join(" ", refusals.OrderBy(p => p.Key).Select(p => p.Key + "=" + p.Value)) + ", refresher calls " + kit.Refresher.Calls + ", live at end " + live.Count + ", restarts " + restarts + ", restored sessions whose idle start moved later " + lengthened + " (max " + TimeSpan.FromTicks(maxLater) + ")");
    }
