    // ---- G04: the download over a REAL SocketsHttpHandler on loopback (Tuesday's point b; redirects) ----
    private sealed class ToLoopback(int port, HttpMessageHandler inner) : DelegatingHandler(inner)
    {
        protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken cancellationToken)
        {
            request.RequestUri = new Uri("http://127.0.0.1:" + port.ToString(System.Globalization.CultureInfo.InvariantCulture) + request.RequestUri!.PathAndQuery);
            return base.SendAsync(request, cancellationToken);
        }
    }

    private sealed class Server : IDisposable
    {
        private readonly TcpListener _listener = new(IPAddress.Loopback, 0);
        internal readonly List<string> Seen = [];
        internal long Sent;

        internal Server() => _listener.Start();

        internal int Port => ((IPEndPoint)_listener.LocalEndpoint).Port;

        internal Task Serve(int connections, Func<int, NetworkStream, Task> script, CancellationToken ct) => Task.Run(async () =>
        {
            for (int n = 0; n < connections; n++)
            {
                using TcpClient client = await _listener.AcceptTcpClientAsync(ct);
                using NetworkStream stream = client.GetStream();
                byte[] buf = new byte[8192];
                StringBuilder head = new();
                while (!head.ToString().Contains("\r\n\r\n", StringComparison.Ordinal))
                {
                    int r = await stream.ReadAsync(buf, ct);
                    if (r == 0)
                    {
                        break;
                    }

                    head.Append(Encoding.ASCII.GetString(buf, 0, r));
                }

                string[] lines = head.ToString().Split("\r\n");
                bool auth = lines.Any(l => l.StartsWith("Authorization:", StringComparison.OrdinalIgnoreCase));
                lock (Seen)
                {
                    Seen.Add(lines[0] + " | Authorization " + (auth ? "PRESENT" : "absent") + " | Host " + (lines.FirstOrDefault(l => l.StartsWith("Host:", StringComparison.OrdinalIgnoreCase)) ?? "-"));
                }

                try
                {
                    await script(n, stream);
                }
                catch (Exception)
                {
                    // the client went away
                }
            }
        }, ct);

        internal async Task Write(NetworkStream s, string text) => await s.WriteAsync(Encoding.ASCII.GetBytes(text));

        internal async Task WriteBytes(NetworkStream s, int count)
        {
            await s.WriteAsync(FileRig.Bytes(count));
            Interlocked.Add(ref Sent, count);
        }

        public void Dispose() => _listener.Stop();
    }

    private static async Task Loop(string label, Func<Server, int, NetworkStream, Task> script, string size, int bound, int connections = 1, Action<HttpClient>? setup = null, bool follow = true, TimeSpan? attempt = null)
    {
        using CancellationTokenSource guard = CancellationTokenSource.CreateLinkedTokenSource(TestContext.Current.CancellationToken);
        guard.CancelAfter(TimeSpan.FromSeconds(60));
        using Server server = new();
        Task serving = server.Serve(connections, (n, s) => script(server, n, s), guard.Token);
        FakeDrive drive = new((index, _) => Synthetic.Json(Item(FileRig.DownloadAddress, size)));
        using HttpClient graphHttp = new(drive);
        GraphOptions options = attempt is null ? new(Synthetic.Base) : new(Synthetic.Base) { AttemptTimeout = attempt.Value };
        using HttpClient download = new(new ToLoopback(server.Port, new SocketsHttpHandler { AllowAutoRedirect = follow, UseProxy = false }));
        setup?.Invoke(download);
        MyDriveFiles files = new(new GraphClient(graphHttp, options), options, download, bound);
        long alloc0 = GC.GetTotalAllocatedBytes(true);
        System.Diagnostics.Stopwatch clock = System.Diagnostics.Stopwatch.StartNew();
        DriveFileBytes? got = null;
        Exception? error = await Record.ExceptionAsync(async () => got = await files.DownloadAsync(FileRig.FileId, new FakeTokens(), guard.Token));
        clock.Stop();
        long alloc = GC.GetTotalAllocatedBytes(true) - alloc0;
        await guard.CancelAsync();
        await Task.WhenAny(serving, Task.Delay(1000));
        string seen;
        lock (server.Seen)
        {
            seen = string.Join(" || ", server.Seen);
        }

        Note($"G04 {label} | {Outcome(error)} | bytes {(got is null ? "-" : got.Bytes.Length.ToString(System.Globalization.CultureInfo.InvariantCulture))} | ms {clock.ElapsedMilliseconds} | server sent {Interlocked.Read(ref server.Sent)} | allocated ~{alloc / 1048576} MiB (process-wide) | requests {seen}");
    }

    private static async Task Endless(Server server, NetworkStream s, bool chunked)
    {
        await server.Write(s, chunked ? "HTTP/1.1 200 OK\r\nTransfer-Encoding: chunked\r\n\r\n" : "HTTP/1.1 200 OK\r\nConnection: close\r\n\r\n");
        for (long sent = 0; sent < 400L * 1048576; sent += 65536)
        {
            if (chunked)
            {
                await server.Write(s, "10000\r\n");
            }

            await server.WriteBytes(s, 65536);
            if (chunked)
            {
                await server.Write(s, "\r\n");
            }
        }
    }

        await Loop("cl-5-body-17-no-size", async (sv, _, s) => { await sv.Write(s, "HTTP/1.1 200 OK\r\nContent-Length: 5\r\n\r\n"); await sv.WriteBytes(s, 17); }, string.Empty, 16);
        await Loop("cl-1000-body-17-close-bound-64", async (sv, _, s) => { await sv.Write(s, "HTTP/1.1 200 OK\r\nContent-Length: 1000\r\n\r\n"); await sv.WriteBytes(s, 17); }, string.Empty, 64);
        await Loop("cl-200MiB-declared-17-then-stall-bound-16", async (sv, _, s) => { await sv.Write(s, "HTTP/1.1 200 OK\r\nContent-Length: 209715200\r\n\r\n"); await sv.WriteBytes(s, 17); await Task.Delay(30000); }, string.Empty, 16);
        await Loop("cl-200MiB-declared-size-200MiB-default-bound", async (sv, _, s) => { await sv.Write(s, "HTTP/1.1 200 OK\r\nContent-Length: 209715200\r\n\r\n"); await sv.WriteBytes(s, 17); }, ",\"size\":209715200", MyDriveFiles.DefaultMaxDownloadBytes);

        await Loop("stall-before-headers-attempt-2s", async (sv, _, s) => await Task.Delay(30000), ",\"size\":5", 16, attempt: TimeSpan.FromSeconds(2));
        await Loop("redirect-302-not-followed", async (sv, _, s) => await sv.Write(s, "HTTP/1.1 302 Found\r\nLocation: /next\r\nContent-Length: 0\r\n\r\n"), ",\"size\":5", 16, follow: false);

            }
            else
            {
                await sv.Write(s, "HTTP/1.1 200 OK\r\nContent-Length: 5\r\n\r\n");

            }
        }, ",\"size\":5", 16, connections: 2, setup: c => c.DefaultRequestHeaders.Authorization = new System.Net.Http.Headers.AuthenticationHeaderValue("Bearer", "gate-synthetic-0001"));
        await Loop("redirect-302-followed-other-host-default-auth", async (sv, n, s) =>
        {
            if (n == 0)
            {
                await sv.Write(s, "HTTP/1.1 302 Found\r\nLocation: http://localhost:" + sv.Port.ToString(System.Globalization.CultureInfo.InvariantCulture) + "/next\r\nContent-Length: 0\r\nConnection: close\r\n\r\n");

            else
            {
                await sv.Write(s, "HTTP/1.1 200 OK\r\nContent-Length: 5\r\n\r\n");

            }
        }, ",\"size\":5", 16, connections: 2, setup: c => c.DefaultRequestHeaders.Authorization = new System.Net.Http.Headers.AuthenticationHeaderValue("Bearer", "gate-synthetic-0001"));
