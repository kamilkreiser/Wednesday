static string Show(string? s)
{
    if (s is null) return "<null>";
    StringBuilder b = new();
    foreach (char c in s)
    {
        if (c >= 0x21 && c < 0x7f && c != '\\') b.Append(c);
        else b.Append("\\u").Append(((int)c).ToString("X4"));
    }
    return b.ToString();
}

string FolderJson(string id, string drive) =>
    "{\"id\":" + System.Text.Json.JsonSerializer.Serialize(id) + ",\"folder\":{\"childCount\":1},\"parentReference\":{\"driveId\":" + System.Text.Json.JsonSerializer.Serialize(drive) + "}}";

async Task<(List<Uri> seen, List<string?> auth, Exception? error, ChannelFolder? folder)> Folder(string teamId, string channelId, string? body = null)
{
    Rec rec = new((_, _) => Json(body ?? FolderJson("SYNTHETICFOLDER0001", "b!synthetic-drive_0001")));
    using HttpClient http = new(rec);
    TeamsBrowser browser = new(new GraphClient(http, new GraphOptions(baseUri)));
    try
    {
        ChannelFolder f = await browser.GetFilesFolderAsync(teamId, channelId, new Tok(), CancellationToken.None);
        return (rec.Seen, rec.Auth, null, f);
    }
    catch (Exception e)
    {
        return (rec.Seen, rec.Auth, e, null);
    }
}

async Task<(List<Uri> seen, Exception? error, IReadOnlyList<ChannelEntry>? list)> Channels(string teamId, string? body = null)
{
    Rec rec = new((_, _) => Json(body ?? "{\"value\":[]}"));
    using HttpClient http = new(rec);
    TeamsBrowser browser = new(new GraphClient(http, new GraphOptions(baseUri)));
    try
    {
        var l = await browser.ListChannelsAsync(teamId, new Tok(), CancellationToken.None);
        return (rec.Seen, null, l);
    }
    catch (Exception e)
    {
        return (rec.Seen, e, null);
    }
}

sealed class Rec(Func<int, Uri, HttpResponseMessage> answer) : HttpMessageHandler
{
    public List<Uri> Seen { get; } = [];
    public List<string?> Auth { get; } = [];
    protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken cancellationToken)
    {
        lock (Seen)
        {
            Seen.Add(request.RequestUri!);
            Auth.Add(request.Headers.Authorization?.Scheme);
        }
        return Task.FromResult(answer(Seen.Count - 1, request.RequestUri!));
    }
}
