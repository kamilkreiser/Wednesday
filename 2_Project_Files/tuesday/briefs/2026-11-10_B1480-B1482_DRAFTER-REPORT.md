`own.py` (rule semantics: `-` wins, `+new` only for paths absent from main's tree, `*` crosses `/`, `+new __tests__/rd<N>-*` expanded per queue key):
```python
import json,os,fnmatch,re,collections,sys
SP=os.environ['SP']
G=json.load(open(SP+'/gather.json')); K=json.load(open(SP+'/kits.json'))
TREE=set(G['tree']); MAIN=G['main']
TP={k:v for k,v in json.load(open('/Volumes/Datasec_DEV/TUESDAY/2_Project_Files/tuesday/briefs/2026-11-10_nexusai_sweep/tickets_paths.json'))}

def expand(owns,keys):
    out=[]
    for l in owns:
        l=l.rstrip()
        m=re.match(r'^\+new __tests__/rd<N>-\*',l)
        if m:
            for k in sorted(keys): out.append(('+new','__tests__/rd%s-*'%k.split('-')[1]))
            continue
        m=re.match(r'^([+-]) (\S+)',l)
        if m: out.append((m.group(1),m.group(2)))
    return out
def claims(path,rules):
    plus=False
    for op,g in rules:
        if op=='-' and fnmatch.fnmatchcase(path,g): return False
    for op,g in rules:
        if op=='+' and fnmatch.fnmatchcase(path,g): plus=True
        if op=='+new' and fnmatch.fnmatchcase(path,g) and path not in TREE: plus=True
    return plus
```
