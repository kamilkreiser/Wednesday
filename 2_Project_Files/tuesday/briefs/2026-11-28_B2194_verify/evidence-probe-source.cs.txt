    [Fact]
    public async Task G01_concurrent_completion()
    {
        using EntraRig rig = new();
        SignInStart start = rig.SignIn.Begin(null);
        string id = GoodId(start);
        rig.Endpoint.Answer = async (_, _, t) =>
        {
            await Task.Delay(50, t);
            return Answers.Json(200, Answers.Token(Synthetic.UserAccess, Synthetic.UserRefresh, id, 3600));
        };
        Task<Exception?>[] calls = [.. Enumerable.Range(0, 16).Select(_ => Task.Run(() => Try(async () => { using var g = await rig.SignIn.CompleteAsync(start.State, Synthetic.Code, CancellationToken.None); })))];
        Exception?[] results = await Task.WhenAll(calls);
        Note($"G01 concurrent-16-same-state: returned {results.Count(r => r is null)}, StateUnknown {results.Count(r => r is EntraException { Reason: EntraError.StateUnknown })}, other {results.Count(r => r is not null && r is not EntraException { Reason: EntraError.StateUnknown })}, token requests {rig.Endpoint.Requests}");

        int before = rig.Endpoint.Requests;
        Exception? bad = await Try(() => rig.SignIn.CompleteAsync(s2.State, "a b", CancellationToken.None));
        Exception? good = await Try(async () => { using var g = await rig.SignIn.CompleteAsync(s2.State, Synthetic.Code, CancellationToken.None); });
        Note($"G01 bad-code-then-good: bad {Outcome(bad)}, then good {Outcome(good)}, requests added {rig.Endpoint.Requests - before}");

        // the state compared ordinally (another case) and a state of 10 000 characters
        SignInStart s3 = rig.SignIn.Begin(null);
        Exception? upper = await Try(() => rig.SignIn.CompleteAsync(s3.State.ToUpperInvariant() == s3.State ? s3.State.ToLowerInvariant() : s3.State.ToUpperInvariant(), Synthetic.Code, CancellationToken.None));
        Exception? huge = await Try(() => rig.SignIn.CompleteAsync(new string('a', 10000), Synthetic.Code, CancellationToken.None));
        Note($"G01 state-other-case: {Outcome(upper)}; state-10000-chars: {Outcome(huge)}; still held {rig.SignIn.VerifierOf(s3.State) is not null}");

    private static int CountOf(string text, string part)
    {
        int n = 0;
        for (int i = text.IndexOf(part, StringComparison.Ordinal); i >= 0; i = text.IndexOf(part, i + 1, StringComparison.Ordinal))
        {
            n++;
        }

        return n;
    }

    private static HttpResponseMessage Declared(long length, string body)
    {
        ByteArrayContent content = new(Encoding.UTF8.GetBytes(body));

    private sealed class EndlessStream : Stream
    {
        private static long _served;

        internal static void Report() => Note($"G04 endless stream bytes served in all: {Interlocked.Read(ref _served)}");

        public override bool CanRead => true;

        public override bool CanSeek => false;

        public override bool CanWrite => false;

        public override long Length => throw new NotSupportedException();

        public override long Position { get => throw new NotSupportedException(); set => throw new NotSupportedException(); }

        public override void Flush()
        {
        }

        public override int Read(byte[] buffer, int offset, int count)
        {
            int n = Math.Min(count, 1000);
            buffer.AsSpan(offset, n).Fill((byte)' ');
            Interlocked.Add(ref _served, n);
            return n;
        }

        public override long Seek(long offset, SeekOrigin origin) => throw new NotSupportedException();

        public override void SetLength(long value) => throw new NotSupportedException();

        public override void Write(byte[] buffer, int offset, int count) => throw new NotSupportedException();
    }

    /// <summary>Sends every request to 127.0.0.1:port over http, keeping the path (the gate's stand-in for a host's handler).</summary>
    private sealed class ToLoopback(int port) : DelegatingHandler
    {
        protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken cancellationToken)
        {
            request.RequestUri = new Uri("http://127.0.0.1:" + port + request.RequestUri!.PathAndQuery);
            return base.SendAsync(request, cancellationToken);
        }
    }

    /// <summary>A one-connection-at-a-time HTTP/1.1 server on 127.0.0.1 inside the test process.</summary>
    private sealed class Loopback : IDisposable
    {
        internal const string OtherOid = "00000000-0000-4000-8000-000000000099";
        private readonly TcpListener _listener = new(IPAddress.Loopback, 0);
        private readonly Func<Loopback, (int Status, string Headers, string Body)> _answer;
        private readonly CancellationTokenSource _stop = new();

        internal Loopback(Func<Loopback, (int, string, string)> answer)
        {
            _answer = answer;
            _listener.Start();
            Port = ((IPEndPoint)_listener.LocalEndpoint).Port;
            _ = Task.Run(RunAsync);
        }

        internal int Port { get; }

        internal string Seen { get; private set; } = "nothing";

        internal string Body { get; private set; } = string.Empty;

        internal string? Nonce { get; set; }

        internal static (int, string, string) TokenAnswer(Loopback self)
        {
            Dictionary<string, string?> claims = IdTokens.Good(self.Nonce);
            claims["oid"] = OtherOid;
            return (200, "Content-Type: application/json\r\n", Answers.Token("synthetic-forged-access-0001", "synthetic-forged-refresh-0001", IdTokens.Make(claims), 3600));
        }

        private async Task RunAsync()
        {
            try
            {
                while (!_stop.IsCancellationRequested)
                {
                    using TcpClient c = await _listener.AcceptTcpClientAsync(_stop.Token);
                    NetworkStream s = c.GetStream();
                    byte[] buf = new byte[65536];
                    int held = 0;
                    int headerEnd = -1;
                    while (headerEnd < 0)
                    {
                        int n = await s.ReadAsync(buf.AsMemory(held), _stop.Token);
                        if (n == 0)
                        {
                            break;
                        }

                        held += n;
                        headerEnd = Encoding.ASCII.GetString(buf, 0, held).IndexOf("\r\n\r\n", StringComparison.Ordinal);
                    }

                    string head = Encoding.ASCII.GetString(buf, 0, Math.Max(headerEnd, 0));
                    int length = 0;
                    foreach (string line in head.Split("\r\n"))
                    {
                        if (line.StartsWith("Content-Length:", StringComparison.OrdinalIgnoreCase))
                        {
                            length = int.Parse(line[15..].Trim(), System.Globalization.CultureInfo.InvariantCulture);
                        }
                    }

                    while (held < headerEnd + 4 + length)
                    {
                        int n = await s.ReadAsync(buf.AsMemory(held), _stop.Token);
                        if (n == 0)
                        {
                            break;
                        }

                        held += n;
                    }

                    string requestLine = head.Split("\r\n")[0];
                    Seen = requestLine.Split(' ')[0] + " " + requestLine.Split(' ')[1] + " body " + length + " bytes" + (head.Contains("Authorization:", StringComparison.OrdinalIgnoreCase) ? " WITH Authorization" : string.Empty);
                    Body = Encoding.ASCII.GetString(buf, headerEnd + 4, Math.Max(0, held - headerEnd - 4));
                    (int status, string headers, string body) = _answer(this);
                    byte[] payload = Encoding.UTF8.GetBytes(body);
                    string response = "HTTP/1.1 " + status + " X\r\n" + headers + "Content-Length: " + payload.Length + "\r\nConnection: close\r\n\r\n";
                    await s.WriteAsync(Encoding.ASCII.GetBytes(response), _stop.Token);
                    await s.WriteAsync(payload, _stop.Token);
                }
            }
            catch (Exception)
            {
                // the listener stops at Dispose
            }
        }

        public void Dispose()
        {
            _stop.Cancel();
            _listener.Stop();
        }
    }

            await source.GetNewTokenAsync(CancellationToken.None);

            Note($"G06 two assertions at one clock instant differ (fresh jti): {a1 != a2 && a1.Length > 0}");
            expired.Dispose();

            Exception? d2 = null;
            rig.Clock.Advance(TimeSpan.FromHours(2));
            d2 = await Try(() => source.GetTokenAsync(CancellationToken.None).AsTask());
            Note($"G06 certificate-disposed-after-construction: GetNewTokenAsync {Outcome(d)}{(d is null ? " (null or token)" : string.Empty)}; GetTokenAsync {Outcome(d2)}; message '{d2?.Message}'; requests {rig.Endpoint.Requests}");

    // ---- G08 (P3, P7): the refresh and the broker ----
    [Fact]
    public async Task G08_refresh_and_broker()
    {
        foreach (string row in new[] { "other-tenant", "other-aud", "no-access", "oid-upper", "id-token-with-nonce" })
        {
            using EntraRig rig = new();
            Dictionary<string, string?> claims = IdTokens.Good(row == "id-token-with-nonce" ? "synthetic-nonce-0003" : null);
            if (row == "other-tenant")
            {
                claims["tid"] = "00000000-0000-4000-8000-000000000009";
            }

            if (row == "other-aud")
            {
                claims["aud"] = "00000000-0000-4000-8000-000000000009";
            }

            if (row == "oid-upper")
            {
                claims["oid"] = Synthetic.ObjectId.ToUpperInvariant();
            }

            string id = IdTokens.Make(claims);
            rig.Endpoint.Answer = (_, _, _) => Answers.JsonTask(200, Answers.Token(row == "no-access" ? null : Synthetic.UserAccess2, Synthetic.UserRefresh2, id, 3600));
            EntraTokenRefresher refresher = new(rig.Client, rig.Clock);
            Exception? e = await Try(async () => { using var g = await refresher.RefreshAsync(Key, Encoding.ASCII.GetBytes(Synthetic.UserRefresh), Synthetic.ObjectId, CancellationToken.None); });
            Note($"G08 refresh {row}: {Outcome(e)}");
        }

        // the broker after a SubjectMismatch: does the session go on with the held token?
        using (EntraRig rig = new())
        {
            Dictionary<string, string?> claims = IdTokens.Good(null);
            claims["oid"] = "00000000-0000-4000-8000-000000000009";
            string id = IdTokens.Make(claims);
            rig.Endpoint.Answer = (_, _, _) => Answers.JsonTask(200, Answers.Token(Synthetic.UserAccess2, Synthetic.UserRefresh2, id, 3600));
            using TokenBroker broker = new(new TokenBrokerOptions(), new FakeRecordStore(), new EntraTokenRefresher(rig.Client, rig.Clock), rig.Clock);
            using (TokenGrant grant = new(Encoding.ASCII.GetBytes(Synthetic.UserAccess), Encoding.ASCII.GetBytes(Synthetic.UserRefresh), Synthetic.ObjectId, rig.Clock.GetUtcNow() + TimeSpan.FromHours(1)))
            {
                broker.SignIn(Key, Synthetic.ObjectId, TimeSpan.FromMinutes(30), grant);
            }

            Exception? n = await Try(async () => { using AccessToken t = await broker.GetNewTokenAsync(Key, CancellationToken.None); });
            string held = "-";
            Exception? g = await Try(async () =>
            {
                using AccessToken t = await broker.GetTokenAsync(Key, CancellationToken.None);
                held = t.AsSpan().SequenceEqual(Encoding.ASCII.GetBytes(Synthetic.UserAccess)) ? "the first user's held token" : t.AsSpan().SequenceEqual(Encoding.ASCII.GetBytes(Synthetic.UserAccess2)) ? "the OTHER user's new token" : "?";
            });
            SessionGraphTokens tokens = new(broker, Key);
            GraphToken? viaSource = await tokens.GetNewTokenAsync(CancellationToken.None);
            Note($"G08 broker-after-SubjectMismatch: GetNewTokenAsync {Outcome(n)}; then GetTokenAsync {Outcome(g)} -> {held}; SessionGraphTokens.GetNewTokenAsync {(viaSource is null ? "null (the 401 stands)" : "a token")}; token requests {rig.Endpoint.Requests}");
        }

        // a session token holding a byte outside printable ASCII
        using (EntraRig rig = new())
        {
            using TokenBroker broker = new(new TokenBrokerOptions(), new FakeRecordStore(), new EntraTokenRefresher(rig.Client, rig.Clock), rig.Clock);
            using (TokenGrant grant = new(Encoding.ASCII.GetBytes("synthetic access-0001"), Encoding.ASCII.GetBytes(Synthetic.UserRefresh), Synthetic.ObjectId, rig.Clock.GetUtcNow() + TimeSpan.FromHours(1)))
            {
                broker.SignIn(Key, Synthetic.ObjectId, TimeSpan.FromMinutes(30), grant);
            }

            Exception? e = await Try(async () => await new SessionGraphTokens(broker, Key).GetTokenAsync(CancellationToken.None));
            Exception? u = await Try(async () => await new SessionGraphTokens(broker, new SessionKey("SERIAL0002", SessionPlatform.Oxpd2, "session0002")).GetTokenAsync(CancellationToken.None));
            Note($"G08 session-token-with-space: {Outcome(e)}; unknown session: {Outcome(u)}");
        }
    }

    // ---- G09 (P8): argument messages and the time-out ----
    [Fact]
    public async Task G09_messages_and_time()
    {
        string[] values = [Synthetic.Secret, Synthetic.Code, Synthetic.Hint, "not-a-id-synthetic-0001", "synthetic-scope-0001 x"];
        List<string> texts = [];
        texts.Add(TrySync(() => _ = new EntraOptions(Synthetic.Authority, "not-a-id-synthetic-0001", Synthetic.Client))?.Message ?? "");
        texts.Add(TrySync(() => new EntraOptions(Synthetic.Authority, Synthetic.Tenant, Synthetic.Client) { ExtraScopes = ["synthetic-scope-0001 x"] }.Validate())?.Message ?? "");
        using EntraRig rig = new();
        texts.Add(TrySync(() => rig.SignIn.Begin(Synthetic.Hint + " "))?.Message ?? "");
        texts.Add((await Try(() => rig.SignIn.CompleteAsync("s", Synthetic.Code + " ", CancellationToken.None)))?.Message ?? "");
        texts.Add(TrySync(() => _ = new AppTokenSource(rig.Client, "synthetic-scope-0001 x"))?.Message ?? "");
        int hits = texts.Sum(t => values.Count(v => t.Contains(v, StringComparison.OrdinalIgnoreCase)));
        Note($"G09 argument messages {texts.Count}, values found in them {hits}");

        // mutable ExtraScopes after Validate
        List<string> extra = ["synthetic.scope.0001"];
        EntraOptions o = new(Synthetic.Authority, Synthetic.Tenant, Synthetic.Client) { RedirectUri = Synthetic.Redirect, ExtraScopes = extra };
        EntraTokenClient c = new(rig.Http, o, rig.Credential, rig.Clock);
        extra.Add("synthetic scope with spaces");
        SignInStart s = new EntraSignIn(c, rig.Clock).Begin(null);
        Note($"G09 extra-scopes-list-changed-after-validate: scope sent '{FormText.QueryOf(s.Address)["scope"]}'");

        // an answer that arrives after the time-out, the caller's cancellation type
        using EntraRig slow = new(requestTimeout: TimeSpan.FromMilliseconds(100));
        SignInStart st = slow.SignIn.Begin(null);
        slow.Endpoint.Answer = async (_, _, _) =>
        {
            await Task.Delay(400);
            return Answers.Json(200, Answers.Token(Synthetic.UserAccess, Synthetic.UserRefresh, GoodId(st), 3600));
        };
        System.Diagnostics.Stopwatch w = System.Diagnostics.Stopwatch.StartNew();
        Exception? late = await Try(async () => { using var g = await slow.SignIn.CompleteAsync(st.State, Synthetic.Code, CancellationToken.None); });
        Note($"G09 answer-after-timeout (handler ignores its token): {Outcome(late)} after {w.ElapsedMilliseconds} ms");
        using CancellationTokenSource pre = new();
        pre.Cancel();
        SignInStart st2 = slow.SignIn.Begin(null);
        int before = slow.Endpoint.Requests;
        Exception? precancel = await Try(() => slow.SignIn.CompleteAsync(st2.State, Synthetic.Code, pre.Token));
        Note($"G09 already-cancelled-token: {Outcome(precancel)}, requests added {slow.Endpoint.Requests - before}, entry held after {slow.SignIn.VerifierOf(st2.State) is not null}");
    }
