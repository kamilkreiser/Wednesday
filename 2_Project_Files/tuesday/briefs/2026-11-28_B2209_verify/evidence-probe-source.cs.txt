/// <summary>Gate B2209 probes (scratch only, never committed). Each probe PRINTS what it saw to a file; none asserts.</summary>
public class GateProbes
{
    private static readonly string Out = Environment.GetEnvironmentVariable("G2209_OUT") ?? "/dev/null";
    private static readonly object Gate = new();

    private static void P(string line)
    {
        lock (Gate)
        {
            File.AppendAllText(Out, line + "\n");
        }
    }

    private static string Auth(string? a) => a is null ? "null" : (a == "Bearer " + Synthetic.Access ? "Bearer <synthetic>" : "OTHER");

    private static string Ex(Exception e) => e switch
    {
        SharePointException s => "SharePointException " + s.Kind + " p" + s.Page + " i" + s.Item,
        GraphCallException g => "GraphCallException " + g.Kind + " " + (g.StatusCode is null ? "-" : ((int)g.StatusCode).ToString(System.Globalization.CultureInfo.InvariantCulture)),
        _ => e.GetType().Name,
    };

    private sealed class RefreshingTokens : IGraphTokenSource
    {
        public ValueTask<GraphToken> GetTokenAsync(CancellationToken cancellationToken) => ValueTask.FromResult(new GraphToken(Synthetic.Access));

        public ValueTask<GraphToken?> GetNewTokenAsync(CancellationToken cancellationToken) => ValueTask.FromResult<GraphToken?>(new GraphToken(Synthetic.Access));
    }

    private static async Task<(string Outcome, List<Recorded> Seen)> Upload(Func<int, Recorded, HttpResponseMessage> answer, string name = Synthetic.FileName, Action<HttpClient>? setup = null, IGraphTokenSource? tokens = null)
    {
        using CancellationTokenSource guard = Guard.TenSeconds();
        using Rig rig = new(answer);
        setup?.Invoke(rig.Http);
        using MemoryStream content = new(Synthetic.Bytes(10));
        string outcome;
        try
        {
            LibraryItem item = await rig.Files.UploadAsync(Synthetic.SiteId, Synthetic.DriveId, null, name, content, 10, tokens ?? rig.Tokens, guard.Token);
            outcome = "RETURNED kind " + item.Kind + " size " + (item.Size?.ToString(System.Globalization.CultureInfo.InvariantCulture) ?? "none") + " key " + (item.ListItemKey is null ? "none" : (item.ListItemKey == Synthetic.ListItemKey ? "synthetic" : "other")) + " nameEqualsSent " + (item.Name == name);
        }
        catch (Exception e)
        {
            outcome = "THREW " + Ex(e);
        }

        return (outcome, rig.Graph.Seen);
    }

    private static Func<int, Recorded, HttpResponseMessage> Session(string sessionJson, string storedJson, HttpStatusCode stored = HttpStatusCode.Created) =>
        (index, _) => index == 0 ? Synthetic.Json(sessionJson) : Synthetic.Json(storedJson, stored);

    [Fact]
    public async Task GC_token_and_conflict_paths()
    {
        string file = Synthetic.File("SYNTHETICFILE0001", Synthetic.FileName, 10, Synthetic.ETag);
        string session = "{\"uploadUrl\":\"" + Synthetic.SessionAddress + "\"}";

        // A 401 on the session call, then a refreshed token: is the conflict body sent again, unchanged?
        (string o1, List<Recorded> s1) = await Upload(
            (index, _) => index == 0 ? Synthetic.Json("{}", HttpStatusCode.Unauthorized) : index == 1 ? Synthetic.Json(session) : Synthetic.Json(file, HttpStatusCode.Created),
            tokens: new RefreshingTokens());
        P("GC 401-then-200 => " + o1 + " | requests " + s1.Count + " | bodies " + string.Join(" ; ", s1.Select(r => r.Method + " " + (r.Body.Length == 0 ? "(empty)" : (Encoding.UTF8.GetString(r.Body) == SharePointFiles.SessionBody ? "SessionBody" : (r.Body.Length == 10 ? "10 bytes" : "OTHER"))) + " auth " + Auth(r.Authorization))));

        // The host's client carries a DEFAULT Authorization header: does the piece carry it?
        (string o2, List<Recorded> s2) = await Upload(Session(session, file), setup: http => http.DefaultRequestHeaders.Authorization = new AuthenticationHeaderValue("Bearer", "synthetic-default-0002"));
        P("GC default-auth-header-on-client => " + o2 + " | " + string.Join(" ; ", s2.Select(r => r.Method + " " + r.Address.Host + " auth " + (r.Authorization is null ? "null" : r.Authorization.Contains("synthetic-default-0002", StringComparison.Ordinal) ? "THE CLIENT DEFAULT" : Auth(r.Authorization)))));

        // A 409 on the session call: is any second request made (a retry with another rule)?
        (string o3, List<Recorded> s3) = await Upload((_, _) => Synthetic.Json("{\"error\":{\"code\":\"nameAlreadyExists\"}}", HttpStatusCode.Conflict));
        P("GC 409 => " + o3 + " | requests " + s3.Count + " | " + string.Join(" ; ", s3.Select(r => r.Method + " " + r.Address.PathAndQuery)));

        // A 307 redirect answer on the session call (the fake's own answer; the handler is the fake, so no follow).
        (string o4, List<Recorded> s4) = await Upload((_, _) => { HttpResponseMessage m = new(HttpStatusCode.TemporaryRedirect); m.Headers.Location = new Uri("https://evil.example.test/x"); return m; });
        P("GC 307 => " + o4 + " | requests " + s4.Count);

        // Names: case variants, ASCII lookalikes, Unicode forms, trailing dot or space.
        string[] names = ["SCAN-0001.PDF", "scan-0001.pdf.", "scan-0001.pdf ", " scan-0001.pdf", "scan-0001..pdf", "caf\u00e9.pdf", "cafe\u0301.pdf", "scan\uff0epdf", "scan-0001.pdf\u200b", "CON", "-", "_", "a.", ".a", "~$a.pdf", "a" + new string('b', 127), "a" + new string('b', 128)];
        foreach (string name in names)
        {
            (string o, List<Recorded> s) = await Upload(Session(session, Synthetic.File("SYNTHETICFILE0001", name.Length > 0 && SharePointIds.IsFileName(name) ? name : Synthetic.FileName, 10, Synthetic.ETag)), name);
            P("GC name " + JsonSerializer.Serialize(name.Length > 30 ? name[..10] + "...len" + name.Length : name) + " IsFileName " + SharePointIds.IsFileName(name) + " => " + o + " | requests " + s.Count + (s.Count > 0 ? " path " + s[0].Address.AbsolutePath.Replace("/v1.0/sites/" + Synthetic.SiteId, "<site>", StringComparison.Ordinal) : string.Empty));
        }
    }

        }
        catch (ArgumentException e)

        }
    }

            .Select(f => f.DeclaringType!.Name + "." + f.Name + (f.IsStatic ? " static" : string.Empty) + (f.IsInitOnly || f.IsLiteral ? " readonly" : " MUTABLE"));
        P("GF fields: " + string.Join(", ", fields));
        using CancellationTokenSource cancelled = new();
        await cancelled.CancelAsync();
        int before = rig.Graph.Seen.Count;
        string outcome;
        try
        {
            await rig.Browser.ListLibrariesAsync(Synthetic.SiteId, rig.Tokens, cancelled.Token);
            outcome = "returned";
        }
        catch (Exception e)
        {
            outcome = "THREW " + Ex(e);
        }

        P("GF pre-cancelled token => " + outcome + " requests " + (rig.Graph.Seen.Count - before));
