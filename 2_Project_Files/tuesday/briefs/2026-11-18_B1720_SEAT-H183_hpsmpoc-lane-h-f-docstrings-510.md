import ast, difflib, sys
FILES = {"scripts/deploy-package.py": ("live_check",), "infra/tests/test_deploy_hpsmpoc208.py": ("plant",)}
def target(node, names):
    if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)) and node.name in names:
        b = node.body[0]
        if isinstance(b, ast.Expr) and isinstance(getattr(b, "value", None), ast.Constant) and isinstance(b.value.value, str):
            return b
    return None
def spans(tree, names):
    return [(b.lineno, b.end_lineno) for b in (target(n, names) for n in ast.walk(tree)) if b is not None]
def blank(tree, names):
    for n in ast.walk(tree):
        b = target(n, names)
        if b is not None:
            b.value.value = ""
    return ast.dump(tree)
bad = 0
for rel, names in FILES.items():
    a = open(sys.argv[1] + "/" + rel, encoding="utf-8").read()
    b = open(sys.argv[2] + "/" + rel, encoding="utf-8").read()
    ta, tb = ast.parse(a), ast.parse(b)
    sa, sb = spans(ta, names), spans(tb, names)
    ops = [o for o in difflib.SequenceMatcher(None, a.splitlines(), b.splitlines(), autojunk=False).get_opcodes() if o[0] != "equal"]
    outside = [o for o in ops if not (any(s[0] <= o[1] + 1 and o[2] <= s[1] and o[2] > o[1] for s in sa)

    bad += len(outside) + (not same) + (len(ops) != 1)
print("DOCDIFF", "OK" if not bad else "FAIL")
