import * as fs from 'node:fs';
import * as path from 'node:path';
import { fileURLToPath } from 'node:url';

import { afterEach, beforeEach, describe, expect, it, vi, type MockInstance } from 'vitest';

import type { GateResult } from '../../../gate/evaluate.ts';
import { printReport } from '../../../gate/report.ts';
import type { K6Summary } from '../../../types/index.ts';

const HERE = path.dirname(fileURLToPath(import.meta.url));

/** The recorded smoke summary - every counter in it is below 1,000. */
const SUMMARY = JSON.parse(
    fs.readFileSync(path.join(HERE, '..', 'fixtures', 'smoke-summary.json'), 'utf8'),
) as K6Summary;

/** A failed rate gate on `http_req_failed`: the only shape that prints the breakdown line. */
const FAILED_RATE_GATE: GateResult = {
    name: 'threshold: http_req_failed rate<0.01',
    passed: false,
    actual: '100.0000%',
    limit: 'rate<0.01',
};

/** Synthetic counts at and above 1,000, so the grouped and ungrouped forms differ. */
const BIG_PASSES = 1234;
const BIG_FAILS = 56789;

let logSpy: MockInstance<typeof console.log>;
let errorSpy: MockInstance<typeof console.error>;

/**
 * Join every argument of every call on a spy into one string.
 *
 * @param {MockInstance<typeof console.log>} spy - A `console.log` / `console.error` spy.
 * @returns {string} The stream's lines, newline-joined.
 */
function joined(spy: MockInstance<typeof console.log>): string {
    return spy.mock.calls.map((call: unknown[]) => call.map(String).join(' ')).join('\n');
}

beforeEach(() => {
    logSpy = vi.spyOn(console, 'log').mockImplementation(() => undefined);
    errorSpy = vi.spyOn(console, 'error').mockImplementation(() => undefined);
});

afterEach(() => {
    vi.restoreAllMocks();
});

describe('KS-1164 follow-up: the breakdown counts print in the locale-grouped form', () => {
    it('RED KS-1164 F-3: a 5xx count of 1,234 / 56,789 prints grouped, not as bare digits', () => {
        const metrics: K6Summary['metrics'] = {
            ...SUMMARY.metrics,
            http_server_error: { passes: BIG_PASSES, fails: BIG_FAILS, value: 0.0213 },
        };

        printReport([FAILED_RATE_GATE], 'smoke', true, metrics);

        const error = joined(errorSpy);
        expect(error).toContain(
            `http_server_error [5xx] passes=${BIG_PASSES.toLocaleString()} fails=${BIG_FAILS.toLocaleString()}`,
        );
        expect(joined(logSpy) + error).not.toContain(`passes=${String(BIG_PASSES)} `);
    });

    it('CONTROL KS-1164 F-3c: the runner locale groups thousands, so the two forms really differ', () => {
        expect(BIG_PASSES.toLocaleString()).not.toBe(String(BIG_PASSES));
        expect(BIG_FAILS.toLocaleString()).not.toBe(String(BIG_FAILS));
    });

    it('CONTROL KS-1164 F-3r: the recorded below-1,000 counts still print exactly as the KS-704 cell reads them', () => {
        printReport([FAILED_RATE_GATE], 'smoke', true, SUMMARY.metrics);

        const server = SUMMARY.metrics['http_server_error'];
        if (!server || !('passes' in server)) {
            throw new Error('fixture lost http_server_error - the control cannot mean anything');
        }
        expect(joined(errorSpy)).toContain(
            `http_server_error [5xx] passes=${String(server.passes)} fails=${String(server.fails)}`,
        );
    });
});
