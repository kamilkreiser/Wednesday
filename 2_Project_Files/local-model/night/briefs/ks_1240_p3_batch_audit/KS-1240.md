## What is wrong (one paragraph)

`proxyRequest` in `routes/batch.ts` (`:132`, the fan-out fetch to originate and transfer) and the audit-log fetch in `routes/audit-export.ts` (`:172`, to the security service) pass no `signal`. An upstream that accepts the connection and never answers holds the caller's request open for as long as the socket lives. Each fetch sits inside a `try` whose `catch` already answers for an unreachable service (`batch.ts`: `{ ok: false, error }` for that item; `audit-export.ts`: the constant 502 BAD_GATEWAY). The change adds, per file, a module constant for the bound and a `signal: AbortSignal.timeout(<constant>)` in the one fetch. When the signal fires the fetch rejects and the existing catch answers as it does for an unreachable service. Nothing else changes for a healthy upstream.

+/** Every `fetch(` call in the source, each as the text from `fetch(` to its balancing `)`. */
+function fetchCalls(src: string): string[] {
+  const calls: string[] = [];
+  const re = /\bfetch\(/g;
+  let match: RegExpExecArray | null;
+  while ((match = re.exec(src)) !== null) {
+    const open = match.index + match[0].length - 1;
+    let depth = 0;
+    let end = -1;
+    for (let i = open; i < src.length; i++) {
+      const ch = src[i];
+      if (ch === '(') depth++;
+      else if (ch === ')') {
+        depth--;
+        if (depth === 0) {
+          end = i;
+          break;
+        }
+      }
+    }
+    expect(end, 'unbalanced fetch call at offset ' + match.index).toBeGreaterThan(-1);
+    calls.push(src.slice(match.index, end + 1));
+  }
+  return calls;
+}
+
+function isBounded(call: string): boolean {
+  return call.includes('AbortSignal.timeout(');
+}
+
+/** The first line of each unbounded call, so a failure names the call. */
+function unbounded(src: string): string[] {
+  return fetchCalls(src)
+    .filter((call) => !isBounded(call))
+    .map((call) => call.split('\n')[0].trim());
+}
+
+function constantMs(src: string, name: string): number {
+  const found = new RegExp('const ' + name + ' = ([0-9_]+);').exec(src);
+  return found ? Number(found[1].replace(/_/g, '')) : Number.NaN;
+}
