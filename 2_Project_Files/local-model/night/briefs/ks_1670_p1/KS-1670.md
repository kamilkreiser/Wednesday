## What is wrong (one paragraph)

The success branch of `GET /api/callback` in `services/m365-integration/src/index.ts` (the `res.redirect(...)` at `:677`, right after the log line `OAuth callback: connection completed`) builds `.../admin/integrations?m365=success&connectionId=${connectionId}&tenant=${encodeURIComponent(connection.tenantName || '')}`. `tenant` is encoded and `connectionId` is not. `connectionId` is `connection.id`, an id the server issued, so this is not exploitable today (gate AM1 N4); it is fragile, because an id format that ever admitted `&`, `#` or `%` would change the shape of the redirect's query. The change wraps the one interpolation: `connectionId=${encodeURIComponent(connectionId)}`. For every id the server issues today (the characters `encodeURIComponent` leaves alone) the redirect is byte-for-byte the same.

+// encodeURIComponent, and that the success redirect still encodes the tenant. C1 to C3 prove the instrument finds the route and the redirects it must not disturb.

+import { readFileSync } from 'node:fs';
+import { join } from 'node:path';

+const SRC = readFileSync(join(__dirname, '..', 'index.ts'), 'utf8');
+const NEEDLE = 'connectionId=${';
+
+/** Every interpolation that follows `connectionId=${` in the source, as the text up to the closing `}` of the template expression. */
+function interpolations(): string[] {
+  const found: string[] = [];
+  let at = SRC.indexOf(NEEDLE);
+  while (at !== -1) {
+    const start = at + NEEDLE.length;
+    let depth = 1;
+    let i = start;
+    while (i < SRC.length && depth > 0) {
+      if (SRC[i] === '{') depth++;
+      else if (SRC[i] === '}') depth--;
+      i++;
+    }
+    found.push(SRC.slice(start, i - 1));
+    at = SRC.indexOf(NEEDLE, i);
+  }
+  return found;
+}
