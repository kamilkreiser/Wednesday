## What is wrong (one paragraph)

Five hand-built `fetch(...)` calls in `routes/proxy.ts` (POST and PATCH `/api/users/admin/...` to auth, GET `/api/signatories` and GET `/api/third-party-verifiers` to originate, POST `/api/mcp/generate-package` to the MCP server) and two in `routes/admin.ts` (POST `/api/users/admin/create` and PATCH `/api/users/admin/:id` to auth) pass no `signal`. An upstream that accepts the connection and never answers holds the caller's request open for as long as the socket lives. Each call sits inside a `try` whose `catch` already answers `502 BAD_GATEWAY`. The change adds, per file, a module constant for the bound and, in each of the seven fetches, `signal: AbortSignal.timeout(<constant>),` as the FIRST property of the init object. When the signal fires the fetch rejects and the existing catch answers 502, exactly as for an unreachable upstream. Nothing else changes for a healthy upstream.

+/** Every `fetch(` call in the source, each as the text from `fetch(` to its balancing `)`. */
+function fetchCalls(src: string): string[] {
+  const calls: string[] = [];
+  const re = /\bfetch\(/g;
+  let match: RegExpExecArray | null;
+  while ((match = re.exec(src)) !== null) {
+    const open = match.index + match[0].length - 1;
+    let depth = 0;
+    let end = -1;
+    for (let i = open; i < src.length; i++) {
+      const ch = src[i];
+      if (ch === '(') depth++;
+      else if (ch === ')') {
+        depth--;
+        if (depth === 0) {
+          end = i;
+          break;
+        }
+      }
+    }
+    expect(end, 'unbalanced fetch call at offset ' + match.index).toBeGreaterThan(-1);
+    calls.push(src.slice(match.index, end + 1));
+  }
+  return calls;
+}
+
+function isBounded(call: string): boolean {
+  return call.includes('AbortSignal.timeout(');
+}
+
+/** The first line of each unbounded call, so a failure names the call. */
+function unbounded(src: string): string[] {
+  return fetchCalls(src)
+    .filter((call) => !isBounded(call))
+    .map((call) => call.split('\n')[0].trim());
+}
+
+function constantMs(src: string, name: string): number {
+  const found = new RegExp('const ' + name + ' = ([0-9_]+);').exec(src);
+  return found ? Number(found[1].replace(/_/g, '')) : Number.NaN;
