const AREA = "firmware-protection";

/** The deck's Firmware Protection statements (content package, partner-readiness set), in the deck's order. */
export function FIRMWARE_ITEMS(): { id: string; text: string }[] {
  const found = locateQuestionSet(READINESS_SET_ID);
  return (found?.set.questions ?? []).filter((q) => q.sectionId === AREA).sort((a, b) => a.order - b.order).map((q) => ({ id: q.id, text: q.text }));
}

function CustomerSwitch({ current }: { current: string }) {
  const router = useRouter();
  const { rows } = useDirectory();
  const [pick, setPick] = useState(current);
  if (rows.length === 0) return null;
  return (
    <form className="flex items-end gap-2" onSubmit={(e) => { e.preventDefault(); if (pick !== current) router.push(`/customers/${encodeURIComponent(pick)}/tasks/${FIRMWARE_TASK.id}`); }}>
      <Select label="Customer" value={pick} onChange={(e) => setPick(e.target.value)}>
        {rows.some((r) => r.routeId === current) ? null : <option value={current}>This customer</option>}
        {rows.map((r) => <option key={r.routeId} value={r.routeId}>{r.name}</option>)}
      </Select>
      <Button type="submit" variant="secondary">Switch customer</Button>
    </form>
  );
}
