#!/bin/bash
# {{LAUNCHER}} — cross-project QA agent, ONE gate (gate46, round 46) over ONE Secuura/Blockchain PR, FROZEN at ONE:
# #1348 KS-1054 (T1, a deploy-path change) ROUND 2 OF 2 under the Tier-1 cap. Author Seat B 45th (wrapped); merger {{MERGE_SEAT}} (the successor).
# EVERY VALUE HERE WAS MEASURED BY THE DRAFTER (pin_gate46.py, a --shared --no-checkout scratch clone fetched from origin) at {{MEASURED_AT}}.
# develop {{DEVELOP}}; END_TREE {{END_TREE}} (#1348 squashed over develop). Shape copied from the gate45 launcher, re-keyed for gate46 (one row).
# exit 2:  QA project missing, or this launcher is not in the gateset dir it was filled for (a MOVED KIT).   exit 3-5: capture / prompt / repo missing.
# exit 6:  the head is not at its branch AND refs/pull/1348/head on origin.   exit 8: thinking directive / capture + commission + Linear read + probe named / an unfilled token / row count.
# exit 10: the GitHub compare develop...head: merge_base / ahead / behind / files != the pins.   exit 17: origin develop != the pinned develop.
# exit 20: capture AND prompt name the head in full.   exit 7: the tier lines + FROZEN at ONE + the round-2-of-2 cap.   exit 31: develop / END_TREE in full.
# exit 32: the prompt and the capture state the ticket.   exit 33: the by-name keyword ladder.   exit 39: the HOLDS.
# exit 26: merge authority + the GO string.   exit 25: MERGE ADDENDUM rules + REPORT-HASH-LAST (nothing into report.md after the mail).   exit 23: verdict subject / sender / report dir.
# exit 21: the LAUNCH path refuses when stdin is not a TTY (`--check` is headless).   exit 16: a launch with a G46_* test override set.
# G46_CUR_DEV / G46_HEAD_1348 / G46_PATHS_1348 / G46_BRIEF / G46_PROMPT: test overrides (--check only).
# Generated by fill_gate46.py at {{FILLED_AT}}. Opus by the configured default: the exec line carries NO --model.
# Usage: {{LAUNCHER}} [--check]
set -u
QA_DIR='/Volumes/DevMASTER/!CODING/Testing Agent MAIN'
GS='{{GS}}'
BRIEF="${G46_BRIEF:-$GS/mail_gate46_ready.md}"
PROMPT_FILE="${G46_PROMPT:-$GS/{{PROMPT}}}"
REPO='/Volumes/DevMASTER/!CODING/Secuura/Blockchain/2_Project_Files'
SECUURA_ENV='/Volumes/DevMASTER/!CODING/Secuura/Blockchain/4_Credentials/.env'
# row: PR|tickets|branch ref|head|files|ahead|merge_base|behind|the PR's own paths (sorted, comma-joined)|tier
PRS=(
{{ROWS}}
)
DEVELOP_SHA='{{DEVELOP}}'
END_TREE='{{END_TREE}}'
[ -d "$QA_DIR" ]      || { echo "QA project missing: $QA_DIR" >&2; exit 2; }
_here="$(dirname "$(/bin/realpath "$0")")"
[ "$_here" = "$GS" ] || { echo "REFUSING: this launcher lives in $_here but was filled for $GS — a MOVED KIT: run repin_and_launch_gate46.sh from its new home (it re-fills there)" >&2; exit 2; }
[ -s "$BRIEF" ]       || { echo "capture missing or empty: $BRIEF" >&2; exit 3; }
[ -s "$PROMPT_FILE" ] || { echo "prompt file missing or empty: $PROMPT_FILE" >&2; exit 4; }
[ -d "$REPO" ]        || { echo "repo under test missing: $REPO" >&2; exit 5; }
[ "${#PRS[@]}" -eq 1 ] || { echo "REFUSING: ${#PRS[@]} rows, this gate is ONE PR (frozen)" >&2; exit 8; }
[ "$(head -n 1 "$PROMPT_FILE")" = "ultrathink" ] || { echo "REFUSING: the prompt does not open with the thinking directive" >&2; exit 8; }
grep -qF "$GS/mail_gate46_ready.md" "$PROMPT_FILE" && grep -qF "$GS/COMMISSION.md" "$PROMPT_FILE" && grep -qF "$GS/linear_gate46.md" "$PROMPT_FILE" && grep -qF "$GS/drafter_return_probe.out" "$PROMPT_FILE" && grep -qF "2026-09-29-batch1348-g45/report.md" "$PROMPT_FILE" || { echo "REFUSING: the prompt does not name the capture, the Linear read, the commission, the drafter probe at $GS and the previous round report" >&2; exit 8; }
[ -s "$GS/linear_gate46.md" ] && grep -qF "b82bebb3" "$GS/linear_gate46.md" || { echo "REFUSING: the Linear read $GS/linear_gate46.md is missing, empty or lacks the raise comment b82bebb3" >&2; exit 8; }
[ -s "$GS/drafter_return_probe.out" ] || { echo "REFUSING: the drafter probe $GS/drafter_return_probe.out is missing" >&2; exit 8; }
grep -qE '\{\{[A-Z0-9_]+\}\}' "$PROMPT_FILE" && { echo "REFUSING: the prompt carries an unfilled double-brace fill token" >&2; exit 8; }
_REFS="refs/heads/develop"
for _row in "${PRS[@]}"; do IFS='|' read -r _n _t _br _rest <<< "$_row"; _REFS="$_REFS $_br refs/pull/$_n/head"; done
_LSALL="$(git -C "$REPO" ls-remote origin $_REFS)"   # ONE read of origin for develop + the branch + pull/head
CUR_DEV="${G46_CUR_DEV:-$(printf '%s\n' "$_LSALL" | awk '$2=="refs/heads/develop"{print $1}')}"
[ "$CUR_DEV" = "$DEVELOP_SHA" ] || { echo "REFUSING: origin develop $CUR_DEV != the pinned develop $DEVELOP_SHA — run repin_and_launch_gate46.sh (it re-pins and re-fills in one action)" >&2; exit 17; }
NOTE="origin develop $CUR_DEV == the pinned develop; END_TREE $END_TREE"
for _row in "${PRS[@]}"; do
  IFS='|' read -r _n _t _br _h _f _a _mb _bh _p _ti <<< "$_row"
  _ov="G46_HEAD_$_n"; [ -n "${!_ov:-}" ] && _h="${!_ov}"
  _ov="G46_PATHS_$_n"; [ -n "${!_ov:-}" ] && _p="${!_ov}"
  # WHOLE-FIELD equality (awk $1==h), never a regex prefix
  if [ "$(printf '%s\n' "$_LSALL" | awk -v h="$_h" -v r="$_br" '$1==h && $2==r' | wc -l | tr -d ' ')" != 1 ] || [ "$(printf '%s\n' "$_LSALL" | awk -v h="$_h" -v r="refs/pull/$_n/head" '$1==h && $2==r' | wc -l | tr -d ' ')" != 1 ]; then
    echo "REFUSING: #$_n — $_h is not at $_br AND refs/pull/$_n/head on origin — the head moved (or the pin is wrong); a verdict is valid ONLY at its head" >&2
    printf '%s\n' "$_LSALL" >&2; exit 6
  fi
  COMPARE="$(set -a; . "$SECUURA_ENV"; set +a   # the token lives only in this subshell, never in the exec'd agent's env
  H="$_h" DEV="$CUR_DEV" python3 - <<'PY'
import json, os, urllib.request
t = os.environ.get("GH_TOKEN", "")
try:
    c = json.load(urllib.request.urlopen(urllib.request.Request("https://api.github.com/repos/Secuura/Distributed_Secuura/compare/%s...%s" % (os.environ["DEV"], os.environ["H"]), headers={"Authorization": "Bearer " + t, "Accept": "application/vnd.github+json"}), timeout=60))
    fs = sorted(x["filename"] for x in (c.get("files") or []))
    print("%s ahead=%d behind=%d files=%d paths=%s" % (c["merge_base_commit"]["sha"], c["ahead_by"], c["behind_by"], len(fs), ",".join(fs)))
except Exception as e: print("ERROR %s" % e)
PY
)"
  [ "$COMPARE" = "${_mb} ahead=${_a} behind=${_bh} files=${_f} paths=${_p}" ] || { echo "REFUSING: #$_n — the compare is not the pinned merge_base / ahead / behind / files:" >&2; echo "  got  $COMPARE" >&2; echo "  want ${_mb} ahead=${_a} behind=${_bh} files=${_f} paths=${_p}" >&2; exit 10; }
  grep -qwF "$_h" "$BRIEF" && grep -qwF "$_h" "$PROMPT_FILE" || { echo "REFUSING: the capture and the prompt do not both name #$_n's head $_h in full" >&2; exit 20; }
  NOTE="$NOTE | #$_n $_ti ${_h:0:12} compare ok (behind $_bh)"
done
PROMPT_JOINED="$(python3 -c 'import re,sys; print(re.sub(r"\n\s*", " ", open(sys.argv[1], encoding="utf-8").read()))' "$PROMPT_FILE")"
has() { printf '%s' "$PROMPT_JOINED" | grep -qF -- "$1"; }
for _kt in 'PR #1348 is KS-1054'; do
  has "$_kt" && grep -qF -- "${_kt#PR }" "$BRIEF" || { echo "REFUSING: the capture and the prompt do not BOTH state '$_kt'" >&2; exit 32; }
done
has '#1348 T1' && has 'The gate is FROZEN at ONE PR' && has 'TIER 1 = FULL WEIGHT' && has 'a DEPLOY-PATH change' && has 'ROUND 2 OF 2 under the Tier-1 cap' && has 'a NO GO on #1348 ships NOTHING of it (no round 3) and goes to Kam as a card' || { echo "REFUSING: the tier lines / FROZEN at ONE / the round-2-of-2 cap" >&2; exit 7; }
has 'MEASURE, not conclude' && has 'RULE WHETHER IT BLOCKS' && has 'NAMES THE TREE' && has 'A check that prints nothing needs a control that prints' || { echo "REFUSING: MEASURE, not conclude / RULE WHETHER IT BLOCKS / NAMES THE TREE / the printing control" >&2; exit 33; }
for _w in {{KEYWORDS}}; do
  has "$_w" || { echo "REFUSING: the prompt does not carry the by-name keyword '$_w'" >&2; exit 33; }
done
has 'NO MERGE, NO PUSH to any PR branch, NO ticket state change' && has 'NO ticket filed' && has 'NEVER WRITE THE SHARED CHECKOUT. Findings only.' && has 'Never `rm`' && has 'No `az` command of any kind' && has 'You NEVER reply to Peter' && has 'Never read or write a real `.env`' && has 'Docker runs are `--rm --network none` with the source mounted READ-ONLY' || { echo "REFUSING: the HOLDS" >&2; exit 39; }
has "WEDNESDAY'S signed GO naming the head" && has '{{GO}}' && has 'The merge seat is {{MERGE_SEAT}}' || { echo "REFUSING: merge authority / the GO string" >&2; exit 26; }
has '## MERGE ADDENDUM' && has 'ONE LINE' && has 'KEY-FREE' && has 'MG-11 each subject <= 92' && has 'the FLEET STOP after these merges' && has 'MG-1 2 over 2 paths' && has 'MODE deploy.sh 100755' && has 'TRUE of its diff' && has 'EVERY SUBJECT YOU PROPOSE MUST BE TRUE OF THE DIFF' && has 'is the LAST thing you write to report.md' && has 'NOTHING to report.md after you send the verdict mail' && has 'sha256 report.md = <hex>' || { echo "REFUSING: the MERGE ADDENDUM rules / REPORT-HASH-LAST" >&2; exit 25; }
grep -qF -- '{{VERDICT_SUBJECT}}' "$PROMPT_FILE" && has 'FROM coagent@agentmail.to' && has 'reports/{{REPORT}}/' || { echo "REFUSING: the verdict subject / sender / report dir" >&2; exit 23; }
grep -qF -- "$DEVELOP_SHA" "$PROMPT_FILE" && grep -qF -- "$END_TREE" "$PROMPT_FILE" || { echo "REFUSING: the launch develop / the END_TREE not in full" >&2; exit 31; }
_ANY_OVR="$(env | grep -c '^G46_')"
if [ "${1:-}" = "--check" ]; then
  echo "all guards pass:"
  echo "  $NOTE" | cut -c1-400
  echo "  capture, Linear read, prompt, QA project and repo present; kit at its filled home; ONE row; thinking directive; capture + Linear read (with b82bebb3) + commission + drafter probe + previous round report named; no unfilled token"
  echo "  1 of 1 head at branch AND pull/head (whole-field); 1 of 1 GitHub compare == the pins; capture and prompt BOTH carry the head in full and the ticket"
  echo "  the prompt carries {{N_KW}} by-name keywords, the tier lines, the holds, the GO string, the addendum + REPORT-HASH-LAST rules, the verdict subject + report dir, develop / END_TREE in full"
  [ "$_ANY_OVR" != 0 ] && echo "  (a G46_* TEST OVERRIDE is set)"
  echo "  a launch (not --check) will refuse unless stdin is a TTY (exit 21)"
  exit 0
fi
[ -t 0 ] || { echo "REFUSING: stdin is not a TTY — run this launcher in a cockpit pane, never inside a Bash tool" >&2; exit 21; }
[ "$_ANY_OVR" = 0 ] || { echo "REFUSING: a launch with test overrides set" >&2; exit 16; }
echo "$NOTE" >&2
cd "$QA_DIR" || { echo "cannot enter $QA_DIR" >&2; exit 16; }
exec claude --dangerously-skip-permissions "$(cat "$PROMPT_FILE")"
