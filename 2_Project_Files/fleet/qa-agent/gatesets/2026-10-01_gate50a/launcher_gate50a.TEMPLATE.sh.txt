#!/bin/bash
# {{LAUNCHER}} — cross-project QA agent, ONE gate (gate50a, round 50a) over ONE Secuura/Blockchain PR, FROZEN at ONE:
# #{{PR}} KS-1378 (T1, ROUND 1): ITEM A, an in-range refresh of 4 package-lock.json files (axios 1.20.0 x4, dompurify 3.4.16 in the root lock), locks only.
# Merger: {{MERGE_SEAT}} (author). EVERY VALUE HERE WAS MEASURED BY THE DRAFTER (pin_gate50a.py, a --shared --no-checkout scratch clone fetched from origin) at {{MEASURED_AT}}.
# develop {{DEVELOP}}; END_TREE {{END_TREE}} (the simulated squash). Shape copied from gate49a's launcher (a NEW COPY, never an edit), one row.
# exit 2:  QA project missing, or this launcher is not in the gateset dir it was filled for (a MOVED KIT).   exit 3-5: capture / prompt / repo missing.
# exit 6:  the head is not at its branch AND refs/pull/<n>/head on origin.   exit 8: thinking directive / the kit files named + present / an unfilled token / row count.
# exit 10: the GitHub compare develop...head: merge_base / ahead / behind / files != the pins.   exit 17: origin develop != the pinned develop.
# exit 20: capture AND prompt name the head in full.   exit 7: the tier line + FROZEN at ONE + the round + the tiering rule.   exit 31: develop / END_TREE in full.
# exit 32: the prompt and the capture state the ticket.   exit 33: the by-name keyword ladder (each keyword AS A TOKEN).   exit 39: the HOLDS + the named exceptions.
# exit 34: the RUNTIME-FIRST rule.   exit 36: the authority line (Wednesday's ANSWER route (a); route (b) REFUSED; its STOP conditions).   exit 26: merge authority + the GO string.
# exit 37: the NEW-TICKET ruling (POST AS-IS / POST AMENDED / DO NOT POST, client-visible).   exit 38: the proportionality line (THE CHANGE IS SMALL, 76%).
# exit 25: MERGE ADDENDUM rules + REPORT-HASH-LAST.   exit 23: verdict subject / sender / report dir.
# exit 21: the LAUNCH path refuses when stdin is not a TTY (`--check` is headless).   exit 16: a launch with a G50A_* test override set.
# G50A_CUR_DEV / G50A_HEAD_<n> / G50A_PATHS_<n> / G50A_BRIEF / G50A_PROMPT: test overrides (--check only).
# Generated by fill_gate50a.py at {{FILLED_AT}}. Opus by the configured default: the exec line carries NO --model.
# Usage: {{LAUNCHER}} [--check]
set -u
QA_DIR='/Volumes/DevMASTER/!CODING/Testing Agent MAIN'
GS='{{GS}}'
BRIEF="${G50A_BRIEF:-$GS/mail_gate50a_ready.md}"
PROMPT_FILE="${G50A_PROMPT:-$GS/{{PROMPT}}}"
REPO='/Volumes/DevMASTER/!CODING/Secuura/Blockchain/2_Project_Files'
SECUURA_ENV='/Volumes/DevMASTER/!CODING/Secuura/Blockchain/4_Credentials/.env'
# row: PR|tickets|branch ref|head|files|ahead|merge_base|behind|the PR's own paths (sorted, comma-joined)|tier
PRS=(
{{ROWS}}
)
DEVELOP_SHA='{{DEVELOP}}'
END_TREE='{{END_TREE}}'
[ -d "$QA_DIR" ]      || { echo "QA project missing: $QA_DIR" >&2; exit 2; }
_here="$(dirname "$(/bin/realpath "$0")")"
[ "$_here" = "$GS" ] || { echo "REFUSING: this launcher lives in $_here but was filled for $GS — a MOVED KIT: run repin_and_launch_gate50a.sh from its new home (it re-fills there)" >&2; exit 2; }
[ -s "$BRIEF" ]       || { echo "capture missing or empty: $BRIEF" >&2; exit 3; }
[ -s "$PROMPT_FILE" ] || { echo "prompt file missing or empty: $PROMPT_FILE" >&2; exit 4; }
[ -d "$REPO" ]        || { echo "repo under test missing: $REPO" >&2; exit 5; }
[ "${#PRS[@]}" -eq 1 ] || { echo "REFUSING: ${#PRS[@]} rows, this gate is ONE PR (frozen)" >&2; exit 8; }
[ "$(head -n 1 "$PROMPT_FILE")" = "ultrathink" ] || { echo "REFUSING: the prompt does not open with the thinking directive" >&2; exit 8; }
for _f in mail_gate50a_ready.md READY_1363_mail.md COMMISSION.md lockdelta_1.out reach_1.out overlaps_1.out gh_read_1.out keyscan_1.out end_tree_crosscheck_1.out gh_body_{{PR}}.md; do
  [ -s "$GS/$_f" ] && grep -qF "$GS/$_f" "$PROMPT_FILE" || { echo "REFUSING: the kit file $GS/$_f is missing, empty, or not named in the prompt" >&2; exit 8; }
done
grep -qF "2026-09-30-batch1356-g49a/report.md" "$PROMPT_FILE" || { echo "REFUSING: the prompt does not name the previous round report (gate49a, the same shape)" >&2; exit 8; }
grep -qE '\{\{[A-Z0-9_]+\}\}' "$PROMPT_FILE" && { echo "REFUSING: the prompt carries an unfilled double-brace fill token" >&2; exit 8; }
grep -qF '<''PR>' "$PROMPT_FILE" && { echo "REFUSING: the prompt carries the unpinned PR placeholder" >&2; exit 8; }   # spelled split so this file itself carries no placeholder
_REFS="refs/heads/develop"
for _row in "${PRS[@]}"; do IFS='|' read -r _n _t _br _rest <<< "$_row"; _REFS="$_REFS $_br refs/pull/$_n/head"; done
_LSALL="$(git -C "$REPO" ls-remote origin $_REFS)"   # ONE read of origin for develop + the branch + pull/head
CUR_DEV="${G50A_CUR_DEV:-$(printf '%s\n' "$_LSALL" | awk '$2=="refs/heads/develop"{print $1}')}"
[ "$CUR_DEV" = "$DEVELOP_SHA" ] || { echo "REFUSING: origin develop $CUR_DEV != the pinned develop $DEVELOP_SHA — run repin_and_launch_gate50a.sh (it re-pins and re-fills in one action)" >&2; exit 17; }
NOTE="origin develop $CUR_DEV == the pinned develop; END_TREE $END_TREE"
for _row in "${PRS[@]}"; do
  IFS='|' read -r _n _t _br _h _f _a _mb _bh _p _ti <<< "$_row"
  _ov="G50A_HEAD_$_n"; [ -n "${!_ov:-}" ] && _h="${!_ov}"
  _ov="G50A_PATHS_$_n"; [ -n "${!_ov:-}" ] && _p="${!_ov}"
  # WHOLE-FIELD equality (awk $1==h), never a regex prefix
  if [ "$(printf '%s\n' "$_LSALL" | awk -v h="$_h" -v r="$_br" '$1==h && $2==r' | wc -l | tr -d ' ')" != 1 ] || [ "$(printf '%s\n' "$_LSALL" | awk -v h="$_h" -v r="refs/pull/$_n/head" '$1==h && $2==r' | wc -l | tr -d ' ')" != 1 ]; then
    echo "REFUSING: #$_n — $_h is not at $_br AND refs/pull/$_n/head on origin — the head moved (or the pin is wrong); a verdict is valid ONLY at its head" >&2
    printf '%s\n' "$_LSALL" >&2; exit 6
  fi
  COMPARE="$(set -a; . "$SECUURA_ENV"; set +a   # the token lives only in this subshell, never in the exec'd agent's env
  H="$_h" DEV="$CUR_DEV" python3 - <<'PY'
import json, os, urllib.request
t = os.environ.get("GH_TOKEN", "")
try:
    c = json.load(urllib.request.urlopen(urllib.request.Request("https://api.github.com/repos/Secuura/Distributed_Secuura/compare/%s...%s" % (os.environ["DEV"], os.environ["H"]), headers={"Authorization": "Bearer " + t, "Accept": "application/vnd.github+json"}), timeout=60))
    fs = sorted(x["filename"] for x in (c.get("files") or []))
    print("%s ahead=%d behind=%d files=%d paths=%s" % (c["merge_base_commit"]["sha"], c["ahead_by"], c["behind_by"], len(fs), ",".join(fs)))
except Exception as e: print("ERROR %s" % e)
PY
)"
  [ "$COMPARE" = "${_mb} ahead=${_a} behind=${_bh} files=${_f} paths=${_p}" ] || { echo "REFUSING: #$_n — the compare is not the pinned merge_base / ahead / behind / files:" >&2; echo "  got  $COMPARE" >&2; echo "  want ${_mb} ahead=${_a} behind=${_bh} files=${_f} paths=${_p}" >&2; exit 10; }
  grep -qwF "$_h" "$BRIEF" && grep -qwF "$_h" "$PROMPT_FILE" || { echo "REFUSING: the capture and the prompt do not both name #$_n's head $_h in full" >&2; exit 20; }
  NOTE="$NOTE | #$_n $_ti ${_h:0:12} compare ok (behind $_bh)"
done
PROMPT_JOINED="$(python3 -c 'import re,sys; print(re.sub(r"\n\s*", " ", open(sys.argv[1], encoding="utf-8").read()))' "$PROMPT_FILE")"
has() { printf '%s' "$PROMPT_JOINED" | grep -qF -- "$1"; }
has 'PR #{{PR}} is KS-1378' && grep -qF -- '#{{PR}} is KS-1378' "$BRIEF" || { echo "REFUSING: the capture and the prompt do not BOTH state 'PR #{{PR}} is KS-1378'" >&2; exit 32; }
has '#{{PR}} T1' && has 'The gate is FROZEN at ONE PR' && has 'TIER 1 = IT UNFREEZES EVERY PUSH, A SECURITY-ADVISORY SURFACE, A REAL SERVICE CONSUMER' && has 'ROUND 1 of its own PR' && has 'a round-1 NO GO goes back to the author for round 2' && has 'ship the closed instances, ticket the residue' || { echo "REFUSING: the tier line / FROZEN at ONE / the round / the tiering rule" >&2; exit 7; }
has 'MEASURE, not conclude' && has 'RULE WHETHER IT BLOCKS' && has 'NAMES THE TREE' && has 'A check that prints nothing needs a control that prints' || { echo "REFUSING: MEASURE, not conclude / RULE WHETHER IT BLOCKS / NAMES THE TREE / the printing control" >&2; exit 33; }
for _w in {{KEYWORDS}}; do   # AS A TOKEN: bounded by a non-[A-Za-z0-9-] character (MODES is not satisfied by MODES-X)
  printf '%s' "$PROMPT_JOINED" | grep -qE -- "(^|[^A-Za-z0-9-])$_w([^A-Za-z0-9-]|\$)" || { echo "REFUSING: the prompt does not carry the by-name keyword '$_w'" >&2; exit 33; }
done
has 'in the SERVED tree of an image you built at the head' && has 'the RUNTIME line FIRST' && has 'A NOT MEASURED runtime reach on a T1 PR is not a GO' || { echo "REFUSING: the RUNTIME-FIRST rule" >&2; exit 34; }
has 'ROUTE (b) IS REFUSED.' && has 'Its STOP conditions' && has 'No manifest change.' && has 'If any advisory survives an in-range refresh, STOP and mail; do not baseline it' || { echo "REFUSING: the authority line (Wednesday's ANSWER route (a))" >&2; exit 36; }
has 'POST AS-IS / POST AMENDED / DO NOT POST' && has 'IT IS CLIENT-VISIBLE: Peter reads the board.' && has 'SENTENCE BY SENTENCE against its sources' && has 'NOTHING IS FILED BY YOU' || { echo "REFUSING: the NEW-TICKET ruling (requirement 9)" >&2; exit 37; }
has 'THE CHANGE IS SMALL' && has 'KEEP THE PASS PROPORTIONATE' && has 'is at 76%' || { echo "REFUSING: the proportionality line" >&2; exit 38; }
has 'NO MERGE, NO PUSH to any PR branch, NO ticket state change' && has 'NO ticket filed' && has 'NEVER WRITE THE SHARED CHECKOUT. Findings only.' && has 'Never `rm`' && has 'No `az` command of any kind' && has 'You NEVER reply to Peter' && has 'Never read or write a real `.env`' && has 'Docker runs are `--rm --network none` with the source mounted READ-ONLY' && has 'NO checklist tick' && has 'no Akto scan against any stack' && has 'No audit-baseline row edited' && has 'THE NAMED EXCEPTIONS to those holds — these and NO others' && has 'NEVER `up`, `down`' && has '"no lock regenerated" means no REAL lock' || { echo "REFUSING: the HOLDS" >&2; exit 39; }
has "WEDNESDAY'S signed GO naming the head" && has '{{GO}}' && has 'The merge seat is {{MERGE_SEAT}}' || { echo "REFUSING: merge authority / the GO string" >&2; exit 26; }
has '## MERGE ADDENDUM' && has 'ONE LINE' && has 'KEY-FREE' && has 'MG-11 subject <= 92' && has 'the FLEET STOP after this merge' && has 'MG-1 4 over 4 paths' && has 'BASELINE audit-baseline.json == develop blob' && has 'FUSE 2026-10-09 rows at END' && has 'RUNTIME <kyc axios' && has 'NEW TICKET <POST AS-IS' && has 'EVERY SUBJECT YOU PROPOSE MUST BE TRUE OF THE DIFF' && has 'is the LAST thing you write to report.md' && has 'NOTHING to report.md after you send the verdict mail' && has 'sha256 report.md = <hex>' || { echo "REFUSING: the MERGE ADDENDUM rules / REPORT-HASH-LAST" >&2; exit 25; }
grep -qF -- '{{VERDICT_SUBJECT}}' "$PROMPT_FILE" && has 'FROM coagent@agentmail.to' && has 'reports/{{REPORT}}/' || { echo "REFUSING: the verdict subject / sender / report dir" >&2; exit 23; }
grep -qF -- "$DEVELOP_SHA" "$PROMPT_FILE" && grep -qF -- "$END_TREE" "$PROMPT_FILE" || { echo "REFUSING: the launch develop / the END_TREE not in full" >&2; exit 31; }
_ANY_OVR="$(env | grep -c '^G50A_')"
if [ "${1:-}" = "--check" ]; then
  echo "all guards pass:"
  echo "  $NOTE" | cut -c1-400
  echo "  capture + the READY copy, lockdelta / reach / overlaps / gh / keyscan / END cross-check reads, the PR body, prompt, QA project and repo present; kit at its filled home; ONE row; thinking directive; every kit file + commission + gate49a report named; no unfilled token, no PR placeholder"
  echo "  1 of 1 head at branch AND pull/head (whole-field); the GitHub compare == the pins; capture and prompt BOTH carry the head in full and the ticket"
  echo "  the prompt carries {{N_KW}} by-name keywords (as tokens), the tier line, the RUNTIME-FIRST rule, the authority line, the NEW-TICKET ruling, the proportionality line, the holds + named exceptions, the GO string, the addendum + REPORT-HASH-LAST rules, the verdict subject + report dir, develop / END_TREE in full"
  [ "$_ANY_OVR" != 0 ] && echo "  (a G50A_* TEST OVERRIDE is set)"
  echo "  a launch (not --check) will refuse unless stdin is a TTY (exit 21)"
  exit 0
fi
[ -t 0 ] || { echo "REFUSING: stdin is not a TTY — run this launcher in a cockpit pane, never inside a Bash tool" >&2; exit 21; }
[ "$_ANY_OVR" = 0 ] || { echo "REFUSING: a launch with test overrides set" >&2; exit 16; }
echo "$NOTE" >&2
cd "$QA_DIR" || { echo "cannot enter $QA_DIR" >&2; exit 16; }
exec claude --dangerously-skip-permissions "$(cat "$PROMPT_FILE")"
