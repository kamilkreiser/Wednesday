/**
 * KS-973 round-5 residue — `provisionActors`' own drift arm quarantines a stale manifest.
 *
 * THE FINDING (KS-973 comment a55ab963, 2026-09-07): round 5's F-1 fix wrapped the quarantine call in
 * `pre-suite.ts`'s catch; the SECOND call — `provision-actors.ts`, the `drifted.length > 0` arm —
 * "was NOT driven by any cell. It cannot break the same contract (it sits inside `runPreSuite`'s own
 * try) but it is untested, and 'covered by an outer try' is a structural argument, not a
 * measurement." `manifestPaths.test.ts` drives `quarantineManifest` directly; nothing drove THIS
 * call site. These cells do, the way the ticket's F1 spec asks: pre-create a manifest, provision
 * with a drifted role, assert the stale copy was moved aside and nothing was published — and the
 * negative control that keeps cell 1 honest.
 *
 * `fetch` is stubbed (the stack is scenery); the manifest module is redirected to a temp directory
 * so the worktree's real `fixtures/generated/` is never written. The quarantine itself is the REAL
 * `quarantineManifest`, handed the temp target — the rename is measured, not mocked.
 *
 * @module tests/unit/fixtures/provisionDriftQuarantine.test
 */

import * as fs from 'node:fs';
import * as path from 'node:path';

import { afterEach, beforeEach, describe, expect, it, vi } from 'vitest';

import { generateActors } from '../../../../fixtures/actors.ts';
import { writeManifest } from '../../../../fixtures/manifest.ts';
import { provisionActors } from '../../../../fixtures/provision-actors.ts';

/** Filled by the mock factory below (it runs before this file's imports). */
const tmp = vi.hoisted(() => ({ dir: '', target: '' }));

vi.mock('../../../../fixtures/manifest.ts', async (importActual) => {
    const actual = await importActual<typeof import('../../../../fixtures/manifest.ts')>();
    const fsMod = await import('node:fs');
    const osMod = await import('node:os');
    const pathMod = await import('node:path');
    const { vi: viMod } = await import('vitest');
    tmp.dir = fsMod.mkdtempSync(pathMod.join(osMod.tmpdir(), 'ks973-drift-'));
    tmp.target = pathMod.join(tmp.dir, 'actors-cell.json');
    return {
        ...actual,
        manifestPath: () => tmp.target,
        // The subject: the real rename, aimed at the temp target instead of fixtures/generated/.
        quarantineManifest: (provisionedAt: string) => actual.quarantineManifest(provisionedAt, tmp.target),
        // Scenery: the rejected sidecar is `manifestPaths.test.ts`'s subject, not this file's — it
        // reports a path and writes nothing, so nothing lands in fixtures/generated/.
        writeRejectedManifest: () => pathMod.join(tmp.dir, 'actors.rejected-cell.json'),
        // A spy, so "nothing published" is a measured call count, not an absence.
        writeManifest: viMod.fn(() => pathMod.join(tmp.dir, 'actors-cell.json')),
    };
});

/** A minimal `Response` for the stub — the two members `postJson` reads. */
function reply(status: number, body: unknown): Response {
    return { status, json: () => Promise.resolve(body) } as unknown as Response;
}

const BASE = 'http://127.0.0.1:1';
const SEED = 's227-drift';
/** The actor whose granted role will DIFFER from the one requested — one drift is enough. */
const DRIFTING_KEY = 'issuer';

/**
 * Stub the whole provisioning conversation: the bootstrap admin logs in, every actor is created by
 * the admin path, and `DRIFTING_KEY` comes back as OWNER instead of its desired role.
 *
 * @returns {void}
 */
function stubProvisioningWithOneDrift(): void {
    const byEmail = new Map(generateActors(SEED).map((spec) => [spec.email, spec]));
    vi.stubGlobal('fetch', (url: string, init?: { body?: string }): Promise<Response> => {
        const pathname = new URL(url).pathname;
        const body = init?.body !== undefined ? (JSON.parse(init.body) as Record<string, unknown>) : {};
        if (pathname === '/api/auth/login') {
            return Promise.resolve(reply(200, { success: true, data: { accessToken: 'admin-token' } }));
        }
        if (pathname === '/api/users/admin/create') {
            const spec = byEmail.get(String(body['email']));
            if (spec === undefined) {
                return Promise.reject(new Error(`stub: unknown actor ${String(body['email'])}`));
            }
            const role = spec.key === DRIFTING_KEY ? 'OWNER' : spec.desiredRole;
            return Promise.resolve(reply(201, { success: true, data: { id: `u-${spec.key}`, role } }));
        }
        return Promise.reject(new Error(`stub has no route for ${pathname}`));
    });
}

/** The pre-existing manifest a previous run left behind — what a consumer would read in preference. */
const STALE_MANIFEST = JSON.stringify({ seed: 'an-earlier-run', provisionedAt: '2026-09-07T07:57:14Z', actors: {} });

beforeEach(() => {
    process.env['SECUURA_STACK_SLOT'] = '1';
    process.env['BOOTSTRAP_ADMIN_PASSWORD'] = 'bootstrap-pw-for-the-cell';
    vi.mocked(writeManifest).mockClear();
    stubProvisioningWithOneDrift();
});

afterEach(() => {
    vi.unstubAllGlobals();
    Reflect.deleteProperty(process.env, 'SECUURA_STACK_SLOT');
    Reflect.deleteProperty(process.env, 'BOOTSTRAP_ADMIN_PASSWORD');
    // The temp directory is this file's own `mkdtemp`; its contents are the cells' own output.
    if (tmp.dir !== '' && fs.existsSync(tmp.dir)) {
        for (const entry of fs.readdirSync(tmp.dir)) {
            fs.rmSync(path.join(tmp.dir, entry), { force: true });
        }
    }
});

/**
 * The superseded copies the quarantine wrote into the temp directory.
 *
 * @returns {string[]} file names starting `actors.superseded-`.
 */
function supersededCopies(): string[] {
    return fs.readdirSync(tmp.dir).filter((name) => name.startsWith('actors.superseded-'));
}

describe('provisionActors — the drift arm moves a stale manifest aside (KS-973 residue, provision-actors.ts)', () => {
    it('quarantines a pre-existing manifest on a drifted role and publishes nothing', async () => {
        fs.writeFileSync(tmp.target, STALE_MANIFEST);

        const result = await provisionActors(['--seed', SEED, '--base-url', BASE]);

        // KS-973 comment bd37f262 (the F1 fix-shape): "Pre-create a manifest, then provision with a
        // drifted role. Assert MANIFEST_PATH no longer resolves to the pre-created content, and that a
        // superseded copy exists." Red-proof: "with only the round-2 fix in place this cell fails,
        // because the stale file is still there and still readable."
        expect(result.drifted).toEqual([DRIFTING_KEY]);
        expect(result.written).toBeNull();
        expect(fs.existsSync(tmp.target)).toBe(false);
        const copies = supersededCopies();
        expect(copies).toHaveLength(1);
        expect(fs.readFileSync(path.join(tmp.dir, copies[0] ?? ''), 'utf8')).toBe(STALE_MANIFEST);
        // "nothing published" as a call count: the writer was never invoked.
        expect(vi.mocked(writeManifest)).not.toHaveBeenCalled();
    });

    it('CONTROL — with no pre-existing manifest, a drifted role quarantines nothing and publishes nothing', async () => {
        expect(fs.existsSync(tmp.target)).toBe(false);

        const result = await provisionActors(['--seed', SEED, '--base-url', BASE]);

        // KS-973 comment bd37f262: "Negative / positive control — provision with NO pre-existing
        // manifest and a drifted role. Assert nothing is quarantined and no manifest is published.
        // Without this, a fix that unconditionally deleted or wrote something would pass cell 1."
        expect(result.drifted).toEqual([DRIFTING_KEY]);
        expect(result.written).toBeNull();
        expect(supersededCopies()).toHaveLength(0);
        expect(fs.existsSync(tmp.target)).toBe(false);
        expect(vi.mocked(writeManifest)).not.toHaveBeenCalled();
    });
});
