#!/bin/bash
# {{LAUNCHER}} — cross-project QA agent, ONE BATCH gate (gate42b, TIER 2, round 1 of 2) over ONE Secuura/Blockchain PR, FROZEN at ONE:
# #1340 KS-530 + KS-729 + KS-528 — the audit-baseline re-date of the four rows Kam named, to 2026-10-09. MUST MERGE before {{FUSE}}.
# EVERY VALUE HERE WAS MEASURED BY THE DRAFTER (pin_gate42b.py, a --shared --no-checkout scratch clone fetched from origin) at {{MEASURED_AT}}.
# develop {{DEVELOP}} == the merge-base; END_TREE {{END_TREE}}. Shape copied from gate42's launcher (one row; no stack; no seat items).
# exit 2:  QA project missing, or this launcher is not in the gateset dir it was filled for (a MOVED KIT).   exit 3-5: capture / prompt / repo missing.
# exit 6:  the head is not at its branch AND refs/pull/1340/head on origin.   exit 8: thinking directive / capture + commission named / an unfilled token.
# exit 10: the GitHub compare develop...head: merge_base / ahead / behind / files != the pins.   exit 17: origin develop != the pinned develop.
# exit 20: capture AND prompt name the head in full.   exit 7: the tier line + FROZEN at ONE.   exit 31: develop / END_TREE in full.
# exit 32: the prompt states the tickets.   exit 33: the by-name keyword ladder.   exit 34: Kam's instruction verbatim + the fuse.
# exit 39: the HOLDS.   exit 26: merge authority + the GO string.   exit 25: MERGE ADDENDUM rules.   exit 23: verdict subject / sender.
# exit 21: the LAUNCH path refuses when stdin is not a TTY (`--check` is headless).   exit 16: a launch with a G42B_* test override set.
# G42B_CUR_DEV / G42B_HEAD / G42B_PATHS / G42B_BRIEF / G42B_PROMPT: test overrides (--check only).
# Generated by fill_gate42b.py at {{FILLED_AT}}. Opus by the configured default: the exec line carries NO --model.
# Usage: {{LAUNCHER}} [--check]
set -u
QA_DIR='/Volumes/DevMASTER/!CODING/Testing Agent MAIN'
GS='{{GS}}'
BRIEF="${G42B_BRIEF:-$GS/mail_gate42b_ready.md}"
PROMPT_FILE="${G42B_PROMPT:-$GS/{{PROMPT}}}"
REPO='/Volumes/DevMASTER/!CODING/Secuura/Blockchain/2_Project_Files'
SECUURA_ENV='/Volumes/DevMASTER/!CODING/Secuura/Blockchain/4_Credentials/.env'
# row: PR|tickets|branch ref|head|files|ahead|merge_base|behind|the PR's own paths|tier
PRS=(
  "1340|KS-530 + KS-729 + KS-528|{{BRANCH}}|{{HEAD}}|1|{{AHEAD}}|{{MERGE_BASE}}|{{BEHIND}}|{{PATH}}|T2"
)
DEVELOP_SHA='{{DEVELOP}}'
END_TREE='{{END_TREE}}'
[ -d "$QA_DIR" ]      || { echo "QA project missing: $QA_DIR" >&2; exit 2; }
_here="$(dirname "$(/bin/realpath "$0")")"
[ "$_here" = "$GS" ] || { echo "REFUSING: this launcher lives in $_here but was filled for $GS — a MOVED KIT: run repin_and_launch_gate42b.sh from its new home (it re-fills there)" >&2; exit 2; }
[ -s "$BRIEF" ]       || { echo "capture missing or empty: $BRIEF" >&2; exit 3; }
[ -s "$PROMPT_FILE" ] || { echo "prompt file missing or empty: $PROMPT_FILE" >&2; exit 4; }
[ -d "$REPO" ]        || { echo "repo under test missing: $REPO" >&2; exit 5; }
[ "${#PRS[@]}" -eq 1 ] || { echo "REFUSING: ${#PRS[@]} rows, this gate is ONE PR (frozen)" >&2; exit 8; }
[ "$(head -n 1 "$PROMPT_FILE")" = "ultrathink" ] || { echo "REFUSING: the prompt does not open with the thinking directive" >&2; exit 8; }
grep -qF "$GS/mail_gate42b_ready.md" "$PROMPT_FILE" && grep -qF "$GS/COMMISSION.md" "$PROMPT_FILE" || { echo "REFUSING: the prompt does not name the capture and the commission at $GS" >&2; exit 8; }
grep -qE '\{\{[A-Z0-9_]+\}\}' "$PROMPT_FILE" && { echo "REFUSING: the prompt carries an unfilled double-brace fill token" >&2; exit 8; }
IFS='|' read -r _n _t _br _h _f _a _mb _bh _p _ti <<< "${PRS[0]}"
[ -n "${G42B_HEAD:-}" ] && _h="$G42B_HEAD"
[ -n "${G42B_PATHS:-}" ] && _p="$G42B_PATHS"
_LSALL="$(git -C "$REPO" ls-remote origin refs/heads/develop "$_br" "refs/pull/$_n/head")"   # ONE read of origin
# WHOLE-FIELD equality (awk $1==h), never a regex prefix
if [ "$(printf '%s\n' "$_LSALL" | awk -v h="$_h" -v r="$_br" '$1==h && $2==r' | wc -l | tr -d ' ')" != 1 ] || [ "$(printf '%s\n' "$_LSALL" | awk -v h="$_h" -v r="refs/pull/$_n/head" '$1==h && $2==r' | wc -l | tr -d ' ')" != 1 ]; then
  echo "REFUSING: #$_n — $_h is not at $_br AND refs/pull/$_n/head on origin — the head moved (or the pin is wrong); a verdict is valid ONLY at its head" >&2
  printf '%s\n' "$_LSALL" >&2; exit 6
fi
CUR_DEV="${G42B_CUR_DEV:-$(printf '%s\n' "$_LSALL" | awk '$2=="refs/heads/develop"{print $1}')}"
[ "$CUR_DEV" = "$DEVELOP_SHA" ] || { echo "REFUSING: origin develop $CUR_DEV != the pinned develop $DEVELOP_SHA — run repin_and_launch_gate42b.sh (it re-pins and re-fills in one action)" >&2; exit 17; }
COMPARE="$(
  set -a; . "$SECUURA_ENV"; set +a
  H="$_h" DEV="$CUR_DEV" python3 - <<'PY'
import json, os, urllib.request
t = os.environ.get("GH_TOKEN", "")
try:
    c = json.load(urllib.request.urlopen(urllib.request.Request("https://api.github.com/repos/Secuura/Distributed_Secuura/compare/%s...%s" % (os.environ["DEV"], os.environ["H"]), headers={"Authorization": "Bearer " + t, "Accept": "application/vnd.github+json"}), timeout=60))
    fs = sorted(x["filename"] for x in (c.get("files") or []))
    print("%s ahead=%d behind=%d files=%d paths=%s" % (c["merge_base_commit"]["sha"], c["ahead_by"], c["behind_by"], len(fs), ",".join(fs)))
except Exception as e: print("ERROR %s" % e)
PY
)"
[ "$COMPARE" = "${_mb} ahead=${_a} behind=${_bh} files=${_f} paths=${_p}" ] || { echo "REFUSING: the compare is not the pinned merge_base / ahead / behind / files:" >&2; echo "  got  $COMPARE" >&2; echo "  want ${_mb} ahead=${_a} behind=${_bh} files=${_f} paths=${_p}" >&2; exit 10; }
grep -qwF "$_h" "$BRIEF" && grep -qwF "$_h" "$PROMPT_FILE" || { echo "REFUSING: the capture and the prompt do not both name #$_n's head $_h in full" >&2; exit 20; }
PROMPT_JOINED="$(python3 -c 'import re,sys; print(re.sub(r"\n\s*", " ", open(sys.argv[1], encoding="utf-8").read()))' "$PROMPT_FILE")"
has() { printf '%s' "$PROMPT_JOINED" | grep -qF -- "$1"; }
has 'PR #1340 is KS-530 + KS-729 + KS-528.' && grep -qF -- '#1340 is KS-530 + KS-729 + KS-528' "$BRIEF" || { echo "REFUSING: the capture and the prompt do not BOTH state the tickets of #1340" >&2; exit 32; }
has '#1340 T2' && has 'The batch is FROZEN at ONE PR' && has 'TIER 2 = THROUGH-CODE ONLY' || { echo "REFUSING: the tier line / FROZEN at ONE" >&2; exit 7; }
has "{{AUTH_TEXT}}" && has '{{AUTH_ID}}' && has 'MUST MERGE BEFORE THEN' && has '{{FUSE}}' || { echo "REFUSING: Kam's instruction verbatim / the Message-ID / the fuse" >&2; exit 34; }
has 'MEASURE, not conclude' && has 'RULE WHETHER IT BLOCKS' || { echo "REFUSING: MEASURE, not conclude / RULE WHETHER IT BLOCKS" >&2; exit 33; }
for _w in {{KEYWORDS}}; do
  has "$_w" || { echo "REFUSING: the prompt does not carry the by-name keyword '$_w'" >&2; exit 33; }
done
has 'NO MERGE, NO PUSH to any PR branch, NO ticket state change' && has 'NO ticket filed' && has 'NEVER WRITE THE SHARED CHECKOUT. Findings only.' && has 'NO baseline edit' || { echo "REFUSING: the HOLDS" >&2; exit 39; }
has "WEDNESDAY'S signed GO naming the head" && has '{{GO}}' && has 'The merge seat is {{MERGE_SEAT}}' || { echo "REFUSING: merge authority / the GO string" >&2; exit 26; }
has '## MERGE ADDENDUM' && has 'ONE LINE' && has 'KEY-FREE' && has 'MG-11 each subject <= 92' && has 'the FLEET STOP after this merge' || { echo "REFUSING: the MERGE ADDENDUM rules" >&2; exit 25; }
grep -qF -- '{{VERDICT_SUBJECT}}' "$PROMPT_FILE" && has 'FROM coagent@agentmail.to' && has 'reports/{{REPORT}}/' || { echo "REFUSING: the verdict subject / sender / report dir" >&2; exit 23; }
grep -qF -- "$DEVELOP_SHA" "$PROMPT_FILE" && grep -qF -- "$END_TREE" "$PROMPT_FILE" || { echo "REFUSING: the launch develop / the END_TREE not in full" >&2; exit 31; }
NOTE="origin develop $CUR_DEV == the pinned develop; #$_n head $_h at branch AND pull/$_n/head; END_TREE $END_TREE"
_ANY_OVR="$(env | grep -c '^G42B_')"
if [ "${1:-}" = "--check" ]; then
  echo "all guards pass:"
  echo "  $NOTE"
  echo "  compare (GitHub API) develop...head: $COMPARE" | cut -c1-240
  echo "  capture, prompt, QA project and repo present; kit at its filled home; ONE row; thinking directive; capture + commission named; no unfilled token"
  echo "  capture and prompt BOTH carry the head in full and the tickets; the prompt carries {{N_KW}} by-name keywords, Kam's instruction verbatim + the Message-ID + the fuse, the holds, the GO string, the addendum rules, the verdict subject + report dir, develop / END_TREE in full"
  [ "$_ANY_OVR" != 0 ] && echo "  (a G42B_* TEST OVERRIDE is set)"
  echo "  a launch (not --check) will refuse unless stdin is a TTY (exit 21)"
  exit 0
fi
[ -t 0 ] || { echo "REFUSING: stdin is not a TTY — run this launcher in a cockpit pane, never inside a Bash tool" >&2; exit 21; }
[ "$_ANY_OVR" = 0 ] || { echo "REFUSING: a launch with test overrides set" >&2; exit 16; }
echo "$NOTE" >&2
cd "$QA_DIR" || { echo "cannot enter $QA_DIR" >&2; exit 16; }
exec claude --dangerously-skip-permissions "$(cat "$PROMPT_FILE")"
