#!/bin/bash
# {{LAUNCHER}} — cross-project QA agent, ONE gate (gate51a, round 51a) over ONE Secuura/Blockchain PR, FROZEN at ONE:
# #{{PR}} KS-1364 (T2, ROUND 1): `required: true` on 11 request bodies in 4 *.openapi.ts registry files, 6 new spec-rendering tests, the generated yaml.
# Merger: {{MERGE_SEAT}} (author). EVERY VALUE HERE WAS MEASURED BY THE DRAFTER (pin_gate51a.py, a --shared --no-checkout scratch clone fetched from origin) at {{MEASURED_AT}}.
# develop {{DEVELOP}}; END_TREE {{END_TREE}} (the simulated squash). Shape copied from gate50b's launcher (a NEW COPY, never an edit), one row.
# exit 2:  QA project missing, or this launcher is not in the gateset dir it was filled for (a MOVED KIT).   exit 3-5: capture / prompt / repo missing.
# exit 6:  the head is not at its branch AND refs/pull/<n>/head on origin.   exit 8: thinking directive / the kit files named + present / an unfilled token / row count.
# exit 10: the GitHub compare develop...head: merge_base / ahead / behind / files != the pins.   exit 17: origin develop != the pinned develop.
# exit 20: capture AND prompt name the head in full.   exit 7: the tier line + FROZEN at ONE + the round + the tiering rule.   exit 31: develop / END_TREE in full.
# exit 32: the prompt and the capture state the ticket.   exit 33: the by-name keyword ladder (each keyword AS A TOKEN) + the filename rule.   exit 39: the HOLDS + the named exceptions.
# exit 34: the NO-RUNTIME rule (no live server, no Schemathesis, no image, no database).   exit 36: the RUNTIME-TRUTH rule (a handler that accepts an absent body is a Major; the control op).
# exit 37: the NOT-THE-GATE'S items (KS-1364 state, mergeable_state).   exit 38: the proportionality line (THE CHANGE IS SMALL, 87%, hard stop 90%).
# exit 26: merge authority + the GO string.   exit 25: MERGE ADDENDUM rules + REPORT-HASH-LAST.   exit 23: verdict subject / sender / report dir.
# exit 21: the LAUNCH path refuses when stdin is not a TTY (`--check` is headless).   exit 16: a launch with a G51A_* test override set.
# G51A_CUR_DEV / G51A_HEAD_<n> / G51A_PATHS_<n> / G51A_BRIEF / G51A_PROMPT: test overrides (--check only).
# Generated by fill_gate51a.py at {{FILLED_AT}}. Opus by the configured default: the exec line carries NO --model.
# Usage: {{LAUNCHER}} [--check]
set -u
QA_DIR='/Volumes/DevMASTER/!CODING/Testing Agent MAIN'
GS='{{GS}}'
BRIEF="${G51A_BRIEF:-$GS/READY_1365_mail.md}"
PROMPT_FILE="${G51A_PROMPT:-$GS/{{PROMPT}}}"
REPO='/Volumes/DevMASTER/!CODING/Secuura/Blockchain/2_Project_Files'
SECUURA_ENV='/Volumes/DevMASTER/!CODING/Secuura/Blockchain/4_Credentials/.env'
# row: PR|tickets|branch ref|head|files|ahead|merge_base|behind|the PR's own paths (sorted, comma-joined)|tier
PRS=(
{{ROWS}}
)
DEVELOP_SHA='{{DEVELOP}}'
END_TREE='{{END_TREE}}'
[ -d "$QA_DIR" ]      || { echo "QA project missing: $QA_DIR" >&2; exit 2; }
_here="$(dirname "$(/bin/realpath "$0")")"
[ "$_here" = "$GS" ] || { echo "REFUSING: this launcher lives in $_here but was filled for $GS — a MOVED KIT: run repin_and_launch_gate51a.sh from its new home (it re-fills there)" >&2; exit 2; }
[ -s "$BRIEF" ]       || { echo "capture missing or empty: $BRIEF" >&2; exit 3; }
[ -s "$PROMPT_FILE" ] || { echo "prompt file missing or empty: $PROMPT_FILE" >&2; exit 4; }
[ -d "$REPO" ]        || { echo "repo under test missing: $REPO" >&2; exit 5; }
[ "${#PRS[@]}" -eq 1 ] || { echo "REFUSING: ${#PRS[@]} rows, this gate is ONE PR (frozen)" >&2; exit 8; }
[ "$(head -n 1 "$PROMPT_FILE")" = "ultrathink" ] || { echo "REFUSING: the prompt does not open with the thinking directive" >&2; exit 8; }
for _f in READY_1365_mail.md READY_1365_CORRECTION_mail.md COMMISSION.md specdiff_1.out handlers_1.out gh_read_1.out keyscan_1.out end_tree_crosscheck_1.out gh_body_{{PR}}.md; do
  [ -s "$GS/$_f" ] && grep -qF "$GS/$_f" "$PROMPT_FILE" || { echo "REFUSING: the kit file $GS/$_f is missing, empty, or not named in the prompt" >&2; exit 8; }
done
grep -qF "2026-10-01-batch1364-g50b/report.md" "$PROMPT_FILE" || { echo "REFUSING: the prompt does not name the previous round report (gate50b)" >&2; exit 8; }
grep -qE '\{\{[A-Z0-9_]+\}\}' "$PROMPT_FILE" && { echo "REFUSING: the prompt carries an unfilled double-brace fill token" >&2; exit 8; }
grep -qF '<''PR>' "$PROMPT_FILE" && { echo "REFUSING: the prompt carries the unpinned PR placeholder" >&2; exit 8; }   # spelled split so this file itself carries no placeholder
_REFS="refs/heads/develop"
for _row in "${PRS[@]}"; do IFS='|' read -r _n _t _br _rest <<< "$_row"; _REFS="$_REFS $_br refs/pull/$_n/head"; done
_LSALL="$(git -C "$REPO" ls-remote origin $_REFS)"   # ONE read of origin for develop + the branch + pull/head
CUR_DEV="${G51A_CUR_DEV:-$(printf '%s\n' "$_LSALL" | awk '$2=="refs/heads/develop"{print $1}')}"
[ "$CUR_DEV" = "$DEVELOP_SHA" ] || { echo "REFUSING: origin develop $CUR_DEV != the pinned develop $DEVELOP_SHA — run repin_and_launch_gate51a.sh (it re-pins and re-fills in one action)" >&2; exit 17; }
NOTE="origin develop $CUR_DEV == the pinned develop; END_TREE $END_TREE"
for _row in "${PRS[@]}"; do
  IFS='|' read -r _n _t _br _h _f _a _mb _bh _p _ti <<< "$_row"
  _ov="G51A_HEAD_$_n"; [ -n "${!_ov:-}" ] && _h="${!_ov}"
  _ov="G51A_PATHS_$_n"; [ -n "${!_ov:-}" ] && _p="${!_ov}"
  # WHOLE-FIELD equality (awk $1==h), never a regex prefix
  if [ "$(printf '%s\n' "$_LSALL" | awk -v h="$_h" -v r="$_br" '$1==h && $2==r' | wc -l | tr -d ' ')" != 1 ] || [ "$(printf '%s\n' "$_LSALL" | awk -v h="$_h" -v r="refs/pull/$_n/head" '$1==h && $2==r' | wc -l | tr -d ' ')" != 1 ]; then
    echo "REFUSING: #$_n — $_h is not at $_br AND refs/pull/$_n/head on origin — the head moved (or the pin is wrong); a verdict is valid ONLY at its head" >&2
    printf '%s\n' "$_LSALL" >&2; exit 6
  fi
  COMPARE="$(set -a; . "$SECUURA_ENV"; set +a   # the token lives only in this subshell, never in the exec'd agent's env
  H="$_h" DEV="$CUR_DEV" python3 - <<'PY'
import json, os, urllib.request
t = os.environ.get("GH_TOKEN", "")
try:
    c = json.load(urllib.request.urlopen(urllib.request.Request("https://api.github.com/repos/Secuura/Distributed_Secuura/compare/%s...%s" % (os.environ["DEV"], os.environ["H"]), headers={"Authorization": "Bearer " + t, "Accept": "application/vnd.github+json"}), timeout=60))
    fs = sorted(x["filename"] for x in (c.get("files") or []))
    print("%s ahead=%d behind=%d files=%d paths=%s" % (c["merge_base_commit"]["sha"], c["ahead_by"], c["behind_by"], len(fs), ",".join(fs)))
except Exception as e: print("ERROR %s" % e)
PY
)"
  [ "$COMPARE" = "${_mb} ahead=${_a} behind=${_bh} files=${_f} paths=${_p}" ] || { echo "REFUSING: #$_n — the compare is not the pinned merge_base / ahead / behind / files:" >&2; echo "  got  $COMPARE" >&2; echo "  want ${_mb} ahead=${_a} behind=${_bh} files=${_f} paths=${_p}" >&2; exit 10; }
  grep -qwF "$_h" "$BRIEF" && grep -qwF "$_h" "$PROMPT_FILE" || { echo "REFUSING: the capture and the prompt do not both name #$_n's head $_h in full" >&2; exit 20; }
  NOTE="$NOTE | #$_n $_ti ${_h:0:12} compare ok (behind $_bh)"
done
PROMPT_JOINED="$(python3 -c 'import re,sys; print(re.sub(r"\n\s*", " ", open(sys.argv[1], encoding="utf-8").read()))' "$PROMPT_FILE")"
has() { printf '%s' "$PROMPT_JOINED" | grep -qF -- "$1"; }
has 'PR #{{PR}} is KS-1364' && grep -qF -- '#{{PR}} (KS-1364)' "$BRIEF" || { echo "REFUSING: the capture and the prompt do not BOTH state the ticket ('PR #{{PR}} is KS-1364' / '#{{PR}} (KS-1364)')" >&2; exit 32; }
has '#{{PR}} T2' && has 'The gate is FROZEN at ONE PR' && has 'TIER 2 = SPEC ANNOTATIONS WITH NO RUNTIME HANDLER CHANGE' && has 'ROUND 1 of its own PR' && has 'a round-1 NO GO goes back to the author for round 2' && has 'ship the closed instances, ticket the residue' || { echo "REFUSING: the tier line / FROZEN at ONE / the round / the tiering rule" >&2; exit 7; }
has 'MEASURE, not conclude' && has 'RULE WHETHER IT BLOCKS' && has 'NAMES THE TREE' && has 'A check that prints nothing needs a control that prints' && has 'ASSERT THE FILENAME TOO' || { echo "REFUSING: MEASURE, not conclude / RULE WHETHER IT BLOCKS / NAMES THE TREE / the printing control / the filename rule" >&2; exit 33; }
for _w in {{KEYWORDS}}; do   # AS A TOKEN: bounded by a non-[A-Za-z0-9-] character (MODES is not satisfied by MODES-X)
  printf '%s' "$PROMPT_JOINED" | grep -qE -- "(^|[^A-Za-z0-9-])$_w([^A-Za-z0-9-]|\$)" || { echo "REFUSING: the prompt does not carry the by-name keyword '$_w'" >&2; exit 33; }
done
has 'THE NO-RUNTIME RULE' && has 'No live server is started, no Schemathesis is run, and no image is built in this gate' && has 'NO DATABASE IS READ IN THIS GATE' && has 'no deploy' || { echo "REFUSING: the NO-RUNTIME rule (no live server / no Schemathesis / no image / no database)" >&2; exit 34; }
has 'A HANDLER THAT ACCEPTS AN ABSENT BODY IS A MAJOR' && has 'the spec would lie' && has 'PATCH /api/platform/tenants/{id}' && has 'must read ACCEPTS' && has 'never from memory' || { echo "REFUSING: the RUNTIME-TRUTH rule (requirement 3 and its control)" >&2; exit 36; }
has "NOT THE GATE'S TO RULE" && has "KS-1364 STATE IS WEDNESDAY'S" && has 'You change no ticket state' || { echo "REFUSING: the NOT-THE-GATE'S items" >&2; exit 37; }
has 'THE CHANGE IS SMALL' && has 'KEEP THE PASS PROPORTIONATE' && has 'is at 87%' && has 'the hard stop is 90%' || { echo "REFUSING: the proportionality line" >&2; exit 38; }
has 'NO MERGE, NO PUSH to any PR branch, NO ticket state change' && has 'NO ticket filed' && has 'NEVER WRITE THE SHARED CHECKOUT. Findings only.' && has 'Never `rm`' && has 'No `az` command of any kind' && has 'You NEVER reply to Peter' && has 'Never read or write a real `.env`' && has 'No SSH to any host, no database read' && has 'NO checklist tick' && has 'no Akto scan against any stack' && has 'Reverts and plants live only in a SCRATCH worktree you created' && has 'THE NAMED EXCEPTIONS to those holds — these and NO others' && has 'No image is built.' || { echo "REFUSING: the HOLDS" >&2; exit 39; }
has "WEDNESDAY'S signed GO naming the head" && has '{{GO}}' && has 'The merge seat is {{MERGE_SEAT}}' || { echo "REFUSING: merge authority / the GO string" >&2; exit 26; }
has '## MERGE ADDENDUM' && has 'ONE LINE' && has 'KEY-FREE' && has 'MG-11 subject <= 92' && has 'the FLEET STOP after this merge' && has 'MG-1 11 over 11 paths' && has 'REQUIRED 11 bodies' && has 'HANDLERS 11 of 11 reject an absent body' && has 'SUITES 4 services 0 new reds' && has 'EVERY SUBJECT YOU PROPOSE MUST BE TRUE OF THE DIFF' && has 'is the LAST thing you write to report.md' && has 'NOTHING to report.md after you send the verdict mail' && has 'sha256 report.md = <hex>' || { echo "REFUSING: the MERGE ADDENDUM rules / REPORT-HASH-LAST" >&2; exit 25; }
grep -qF -- '{{VERDICT_SUBJECT}}' "$PROMPT_FILE" && has 'FROM coagent@agentmail.to' && has 'reports/{{REPORT}}/' || { echo "REFUSING: the verdict subject / sender / report dir" >&2; exit 23; }
grep -qF -- "$DEVELOP_SHA" "$PROMPT_FILE" && grep -qF -- "$END_TREE" "$PROMPT_FILE" || { echo "REFUSING: the launch develop / the END_TREE not in full" >&2; exit 31; }
_ANY_OVR="$(env | grep -c '^G51A_')"
if [ "${1:-}" = "--check" ]; then
  echo "all guards pass:"
  echo "  $NOTE" | cut -c1-400
  echo "  capture (READY_1365 + its CORRECTION), specdiff / handlers / gh / keyscan / END cross-check reads, the PR body, prompt, QA project and repo present; kit at its filled home; ONE row; thinking directive; every kit file + commission + gate50b report named; no unfilled token, no PR placeholder"
  echo "  1 of 1 head at branch AND pull/head (whole-field); the GitHub compare == the pins; capture and prompt BOTH carry the head in full and the ticket"
  echo "  the prompt carries {{N_KW}} by-name keywords (as tokens), the tier line, the NO-RUNTIME rule, the RUNTIME-TRUTH rule, the NOT-THE-GATE'S items, the proportionality line, the holds + named exceptions, the GO string, the addendum + REPORT-HASH-LAST rules, the verdict subject + report dir, develop / END_TREE in full"
  [ "$_ANY_OVR" != 0 ] && echo "  (a G51A_* TEST OVERRIDE is set)"
  echo "  a launch (not --check) will refuse unless stdin is a TTY (exit 21)"
  exit 0
fi
[ -t 0 ] || { echo "REFUSING: stdin is not a TTY — run this launcher in a cockpit pane, never inside a Bash tool" >&2; exit 21; }
[ "$_ANY_OVR" = 0 ] || { echo "REFUSING: a launch with test overrides set" >&2; exit 16; }
echo "$NOTE" >&2
cd "$QA_DIR" || { echo "cannot enter $QA_DIR" >&2; exit 16; }
exec claude --dangerously-skip-permissions "$(cat "$PROMPT_FILE")"
