# G52_CUR_DEV / G52_HEAD_<n> / G52_PATHS_<n> / G52_BRIEF / G52_PROMPT: test overrides (--check only).
# Generated by fill_gate52.py at {{FILLED_AT}}. Opus by the configured default: the exec line carries NO --model.

  COMPARE="$(set -a; . "$SECUURA_ENV"; set +a   # the token lives only in this subshell, never in the exec'd agent's env
