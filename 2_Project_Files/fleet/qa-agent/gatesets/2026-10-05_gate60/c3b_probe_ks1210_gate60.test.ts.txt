async function call(who: Who, method: string, url: string, body?: unknown): Promise<{ status: number; json: any; text: string }> {
  const server = app.listen(0, '127.0.0.1');
  try {
    await new Promise<void>((r) => server.once('listening', () => r()));
    const port = (server.address() as { port: number }).port;
    const r = await fetch(`http://127.0.0.1:${port}${url}`, {
      method, headers: { 'Content-Type': 'application/json', Authorization: 'Bearer g60-' + who },
      body: body === undefined ? undefined : JSON.stringify(body),
    });
    const text = await r.text(); let json: any = null;
    try { json = JSON.parse(text); } catch { json = null; }
    return { status: r.status, json, text };
  } finally { server.close(); }
}
