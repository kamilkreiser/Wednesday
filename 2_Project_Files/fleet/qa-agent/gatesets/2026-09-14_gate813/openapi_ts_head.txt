// KS-367: charset-safety constraint for id-style path parameters. Every id the
// platform mints (doc-<ts>-<8hex>, ce_<uuid>, plain UUIDs, webhook ids) fits
// this pattern; what it EXCLUDES is NUL/control bytes and other unsafe
// characters, which gateways (local nginx, ACA Envoy) reject with their own
// undocumented 400 before the request reaches a service. Constraining the
// spec makes such values schema-invalid input rather than a documented-
// response violation. Charset safety only — NOT an entity id format.
// Exported for runtime enforcement (KS-466 §8: routes/documents.ts rejects
// charset-illegal path ids before any DB query — the published grammar and
// the runtime guard must stay one definition).
export const SAFE_ID_PATTERN = /^[A-Za-z0-9][A-Za-z0-9._-]{0,127}$/;

// Reusable `{id}` path-params objects carrying the safe-charset constraint.
//
// KS-665 item 2: these were ONE shared `idPathParams` used by 18 routes across
// three different resource types. That is fine for the constraint (the grammar
// is the same) but it makes a published example impossible: a single `example:`
// here would document a document id on `/api/webhooks/{id}`. That is the same
// defect class as the 16 `address` fields in §3 and the `userAgent` in §4 — a
// legitimate fixture value landing on a field it does not describe — so the
// helper is split by resource and each carries its own example.
//
// The examples below are FORMAT-TRUE AND ACCOUNT-FALSE by design: FX names no
// live row (see the fixture-set header). A path-param example documents the
// SHAPE a caller must send, not an id they can fetch.
const idPathParamsBase = (description: string) =>
  z.string().regex(SAFE_ID_PATTERN).openapi({ description });

const documentIdPathParams = z.object({
  id: idPathParamsBase(
    'Document id (1-128 chars: alphanumeric plus `.`, `_`, `-`; must start alphanumeric)',

});

//
// These two are the only /api/* operations whose request body is binary: the

// KS-695 ask 1 — connector-driven erasure addressed by external_ref.
// security is declared explicitly on both (KS-424 guard). The caller presents
// `x-api-key: sk_…`; the gateway exchanges it for a connector JWT and forwards
// a Bearer, so bearerAuth is the accurate declaration at this hop.
const ConnectorErasureResponseSchema = z.object({
  success: z.boolean().openapi({ example: true }),
  data: z.object({
    // Both values are TERMINAL — S stops retrying on either. `denied` was added
    // for QA F-1: `data_subject_requests.status` admits 'denied' (schema CHECK,
    // `docker/init/01-schema.sql:490`), and treating it as non-terminal caused a
    // human-refused erasure to be re-driven and executed. It is reported, not
    // raised as an error, because a 5xx would put S back into unbounded retry.
    status: z.enum(['completed', 'denied', 'unresolvable']).openapi({
      example: 'completed',
      description:
        'Terminal outcome. `completed` — K holds nothing further erasable for this subject. '
        + '`denied` — an erasure request for this subject was REFUSED by a human (e.g. an Art. 17(3) '
        + 'exemption); the decision stands and the caller must stop retrying. '
        + '`unresolvable` — K could not identify the subject SAFELY and erased NOTHING; see `reason`. '
        + 'All three are terminal: none should be re-driven.',
    }),
    alreadyErased: z.boolean().openapi({
      example: false,
      description: 'Always false when `status` is `denied` or `unresolvable` — nothing was erased.',
    }),
    // QA Blocker 2. A CONTRACT CHANGE for legacy keys: a connector key minted
    // WITHOUT an organizationId previously received `completed` here, because
    // an unresolvable subject fell through to the branch that blanks provenance
    // and terminates. It now receives `unresolvable` + `org_less_key` and K
    // erases nothing on that path. The old answer was the more dangerous one —
    // it reported an Art. 17 obligation discharged that had not been — but the
    // change is visible to S and is called out for that reason.
    reason: z.enum([
      'org_less_key',
      'subject_outside_connector_organisation',
      'ambiguous_subject',
    ]).optional().openapi({
      description:
        'Present only when `status` is `unresolvable`. `org_less_key` — the calling key carries no '
        + 'Organisation, so K cannot attribute the subject to it (re-mint the key with an '
        + 'organizationId). `subject_outside_connector_organisation` — the matched user belongs to no '
        + 'Organisation, so membership of the caller\'s cannot be shown. `ambiguous_subject` — the ref '
        + 'matched MORE THAN ONE user; K refuses to guess which one to erase irreversibly.',
      example: 'org_less_key',
    }),
  }),
}).openapi('ConnectorErasureResponse');

const ConnectorErasureRequestSchema = z.object({
  externalRef: z.string().trim().min(1).max(128).openapi({
    description: "Platform S's Person.PersonGuid — the subject key K receives as onBehalfOf.externalRef. "
      + 'Limit matches the stored column (VARCHAR(128)) and the write-side validator (QA/Peter F4): '
      + 'a longer or untrimmed ref matches no row and must not be reported as a discharged erasure.',
    example: '7f3c1a90-2b44-4d31-9e5a-0c8f2d61b7aa',
  }),
}).openapi('ConnectorErasureRequest');
