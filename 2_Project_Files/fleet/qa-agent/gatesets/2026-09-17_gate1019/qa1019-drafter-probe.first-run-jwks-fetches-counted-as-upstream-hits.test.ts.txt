function listen(server: http.Server): Promise<string> {
  return new Promise((resolve) => server.listen(0, '127.0.0.1', () => resolve(`http://127.0.0.1:${(server.address() as any).port}`)));
}
async function closeServer(server: http.Server | undefined) {
  if (!server) return;
  (server as any).closeAllConnections?.();
  await new Promise<void>((r) => server.close(() => r()));
}
function token(o: { scopes?: string[]; email?: boolean; userId?: string; role?: string; authMethod?: string }): string {
  const p: any = { userId: o.userId || randomUUID(), role: o.role || 'connector', authMethod: o.authMethod || 'api_key', scopes: o.scopes || [],
    tenantId: 'a0000000-0000-4000-8000-000000000001', organizationId: 'd0000000-0000-4000-8000-00000000118d', verificationLevel: 'BASIC' };
  if (o.email !== false) p.email = 'qa-drafter@secuura.invalid';
  return 'Bearer ' + jwt.sign(p, process.env.__TEST_JWT_PRIVATE_PEM as string, { algorithm: 'RS256', expiresIn: '10m' });
}

function send(base: string, method: string, path: string, auth: string | null): Promise<any> {
  return new Promise((resolve) => {
    const t = new URL(base); const before = seen.length; const capBefore = captured.length;
    const body = method === 'GET' ? undefined : '{"externalRef":"qa-drafter-ref"}';
    const headers: Record<string, string> = {};
    if (auth) headers.authorization = auth;
    if (body !== undefined) { headers['content-type'] = 'application/json'; headers['content-length'] = String(Buffer.byteLength(body)); }
    const req = http.request({ hostname: t.hostname, port: t.port, path, method, agent: false, headers }, (res) => {
      const c: Buffer[] = [];
      res.on('data', (d) => c.push(d));
      res.on('end', () => {
        const txt = Buffer.concat(c).toString(); let code: string | null = null; let msg: string | null = null;
        try { const j = JSON.parse(txt); code = j?.error?.code ?? null; msg = (j?.error?.message ?? null) && String(j.error.message).slice(0, 120); } catch { /* non-JSON */ }
        setTimeout(() => resolve({ status: res.statusCode, code, msg, hits: seen.slice(before), location: res.headers.location ?? null,
          rlLimit: res.headers['x-ratelimit-limit'] ?? null, rlRemaining: res.headers['x-ratelimit-remaining'] ?? null,
          graceLines: captured.slice(capBefore).filter((l) => l.includes('grace is ON')).length,
          headerErr: captured.slice(capBefore).filter((l) => /Invalid value .*x-user-email/i.test(l)).length }), 60);
      });
    });
    req.on('error', (e) => resolve({ status: -1, code: 'CLIENT_ERROR', msg: String(e).slice(0, 120), hits: seen.slice(before) }));
    if (body !== undefined) req.end(body); else req.end();
  });
}

  (process.stderr as any).write = (ch: any, ...a: any[]) => { captured.push(String(ch)); return w2(ch, ...a); };
  for (const k of ['warn', 'error', 'log', 'info'] as const) { const o = (console as any)[k].bind(console); (console as any)[k] = (...a: any[]) => { captured.push(a.map((x) => (typeof x === 'string' ? x : JSON.stringify(x))).join(' ')); o(...a); }; }

async function runMatrix(stage: string, url: string, prefix: string, rows: any[]) {
  for (const s of SHAPES) {
    const path = (s.abs ? s.abs : '') + prefix + s.sub;
    for (const t of s.toks) {
      const r = await send(url, s.method, path, TOK[t]());
      rows.push({ id: s.id, cls: s.cls, method: s.method, target: path, tok: t, ...r });
    }
  }
  // Tightening A census: every originate gdprRouter route, params filled, admitted scope token and no-scope token
  for (const [m, p] of OUT.originateRoutes as Array<[string, string]>) {
    const filled = p.replace(/:[A-Za-z]+/g, 'u-1');
    for (const t of D) {
      const r = await send(url, m, prefix + filled, TOK[t]());
      rows.push({ id: 'O ' + m + ' ' + p, cls: p.startsWith('/erasures') ? 'originate-door-route' : 'originate-route', method: m, target: prefix + filled, tok: t, ...r });
    }
  }
}
