# QAB1213_CUR_DEV / QAB1213_HEAD_1217 / QAB1213_BRIEF / QAB1213_PROMPT: test overrides (--check only); a launch with any set refuses (exit 16).
# Generated by fill_gate21T1.py at {{GENERATED_AT}}. Opus by the configured default: the exec line carries NO --model.

  "1214|KS-528|refs/heads/feature/ks-528-frontends-react-router-v6-v7-migration-3-moderate-client-r21-dompatch-1|{{HEAD_1214}}|5"
  "1216|KS-975|refs/heads/feature/ks-975-explicitscope-null-is-malformed-l2-scopenull-1|{{HEAD_1216}}|2"
  "1217|KS-976|refs/heads/feature/ks-976-reset-400-names-the-failing-field-l2-msg400-1|{{HEAD_1217}}|3"
)
BASE_SHA='{{BASE}}'

    printf '%s\n' "$_lsr" >&2
    exit 6
  fi
  HEADS_NOTE="$HEADS_NOTE #$_n@${_h:0:9}"

CUR_DEV="${QAB1213_CUR_DEV:-$(git -C "$REPO" ls-remote origin refs/heads/develop | cut -f1)}"

  PRS_FLAT="${PRS[*]}" CUR_DEV="$CUR_DEV" python3 - <<'PY'

    try:
        c = get(api + os.environ["CUR_DEV"] + "..." + h); print("%s %s ahead=%d behind=%d files=%d" % (n, c["merge_base_commit"]["sha"], c["ahead_by"], c["behind_by"], len(c.get("files") or [])))
    except Exception as e: print("%s ERROR %s" % (n, e))
PY

_bad=0
for _pr in "${PRS[@]}"; do
  IFS='|' read -r _n _t _br _h _f <<< "$_pr"

has 'NO MERGE, NO PUSH to any PR branch, NO ticket state change' && has 'NO deploy of any kind. Findings only.' || { echo "REFUSING: the HOLDS" >&2; exit 39; }
has "WEDNESDAY'S signed GO naming each head" && has '`GO: merge #1213, #1214, #1216, #1217 batch`' || { echo "REFUSING: merge authority / the GO string" >&2; exit 26; }
