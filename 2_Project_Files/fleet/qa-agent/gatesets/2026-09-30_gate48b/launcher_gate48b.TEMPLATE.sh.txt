#!/bin/bash
# {{LAUNCHER}} — cross-project QA agent, ONE gate (gate48b, round 48b) over TWO Secuura/Blockchain PRs, FROZEN at TWO, merged IN ORDER:
# #1354 KS-470 (T2, ROUND 2): the GHSA-r53p row made PERMANENT on Kam's card ruling (c) + GRANDFATHERED_NO_EXPIRY +1 + js-yaml 5.4.2; THEN
# #1355 KS-1378 (T1, ROUND 1): the undici 7.30.0 override in both manifests + both locks regenerated + the same js-yaml bytes (a NO-OP after #1354).
# Merger: {{MERGE_SEAT}}. EVERY VALUE HERE WAS MEASURED BY THE DRAFTER (pin_gate48b.py, a --shared --no-checkout scratch clone fetched from origin) at {{MEASURED_AT}}.
# develop {{DEVELOP}}; END_TREE {{END_TREE}} (the #1354 -> #1355 squash chain; the reverse order gives the same tree). Shape copied from gate48a's launcher, two rows.
# exit 2:  QA project missing, or this launcher is not in the gateset dir it was filled for (a MOVED KIT).   exit 3-5: capture / prompt / repo missing.
# exit 6:  a head is not at its branch AND refs/pull/<n>/head on origin.   exit 8: thinking directive / the kit files named + present / an unfilled token / row count.
# exit 35: a pinned head is SUPERSEDED (kit.json superseded_heads: #1354's gate48a head 4370be410bbf — Kam's ruling needs its NEW head).
# exit 10: the GitHub compare develop...head: merge_base / ahead / behind / files != the pins.   exit 17: origin develop != the pinned develop.
# exit 20: capture AND prompt name each head in full.   exit 7: the tier lines + FROZEN at TWO + the rounds + the tiering rule.   exit 31: develop / END_TREE in full.
# exit 32: the prompt and the capture state both tickets.   exit 33: the by-name keyword ladder (each keyword AS A TOKEN).   exit 39: the HOLDS + the named exceptions.
# exit 34: the RUNTIME-FIRST rule.   exit 36: Kam's ruling as the authority for #1354.   exit 26: merge authority + the GO string.
# exit 25: MERGE ADDENDUM rules + REPORT-HASH-LAST.   exit 23: verdict subject / sender / report dir.
# exit 21: the LAUNCH path refuses when stdin is not a TTY (`--check` is headless).   exit 16: a launch with a G48B_* test override set.
# G48B_CUR_DEV / G48B_HEAD_<n> / G48B_PATHS_<n> / G48B_BRIEF / G48B_PROMPT: test overrides (--check only).
# Generated by fill_gate48b.py at {{FILLED_AT}}. Opus by the configured default: the exec line carries NO --model.
# Usage: {{LAUNCHER}} [--check]
set -u
QA_DIR='/Volumes/DevMASTER/!CODING/Testing Agent MAIN'
GS='{{GS}}'
BRIEF="${G48B_BRIEF:-$GS/mail_gate48b_ready.md}"
PROMPT_FILE="${G48B_PROMPT:-$GS/{{PROMPT}}}"
REPO='/Volumes/DevMASTER/!CODING/Secuura/Blockchain/2_Project_Files'
SECUURA_ENV='/Volumes/DevMASTER/!CODING/Secuura/Blockchain/4_Credentials/.env'
# row: PR|tickets|branch ref|head|files|ahead|merge_base|behind|the PR's own paths (sorted, comma-joined)|tier
PRS=(
{{ROWS}}
)
SUPERSEDED='4370be410bbf37b839b1030f3e28f95d11c454fe'
DEVELOP_SHA='{{DEVELOP}}'
END_TREE='{{END_TREE}}'
[ -d "$QA_DIR" ]      || { echo "QA project missing: $QA_DIR" >&2; exit 2; }
_here="$(dirname "$(/bin/realpath "$0")")"
[ "$_here" = "$GS" ] || { echo "REFUSING: this launcher lives in $_here but was filled for $GS — a MOVED KIT: run repin_and_launch_gate48b.sh from its new home (it re-fills there)" >&2; exit 2; }
[ -s "$BRIEF" ]       || { echo "capture missing or empty: $BRIEF" >&2; exit 3; }
[ -s "$PROMPT_FILE" ] || { echo "prompt file missing or empty: $PROMPT_FILE" >&2; exit 4; }
[ -d "$REPO" ]        || { echo "repo under test missing: $REPO" >&2; exit 5; }
[ "${#PRS[@]}" -eq 2 ] || { echo "REFUSING: ${#PRS[@]} rows, this gate is TWO PRs (frozen)" >&2; exit 8; }
[ "$(head -n 1 "$PROMPT_FILE")" = "ultrathink" ] || { echo "REFUSING: the prompt does not open with the thinking directive" >&2; exit 8; }
for _f in mail_gate48b_ready.md COMMISSION.md row_reason_gate48b.md lockdelta_1.out baseline_1.out lockcensus_1.out lockcensus_2_develop.out overlaps_1.out image_read_1.out gh_read_1.out gh_body_1354.md gh_body_1355.md; do
  [ -s "$GS/$_f" ] && grep -qF "$GS/$_f" "$PROMPT_FILE" || { echo "REFUSING: the kit file $GS/$_f is missing, empty, or not named in the prompt" >&2; exit 8; }
done
grep -qF "2026-09-30-batch1354-g48a/report.md" "$PROMPT_FILE" || { echo "REFUSING: the prompt does not name the previous round report (gate48a)" >&2; exit 8; }
grep -qF 'GHSA-r53p-7pc4-xj5r' "$GS/row_reason_gate48b.md" && grep -qE '^1[0-9]?\. ' "$GS/row_reason_gate48b.md" || { echo "REFUSING: the row reason file lacks the row id or its numbered sentences" >&2; exit 8; }
grep -qE '\{\{[A-Z0-9_]+\}\}' "$PROMPT_FILE" && { echo "REFUSING: the prompt carries an unfilled double-brace fill token" >&2; exit 8; }
for _row in "${PRS[@]}"; do IFS='|' read -r _n _t _br _h _rest <<< "$_row"; _ov="G48B_HEAD_$_n"; [ -n "${!_ov:-}" ] && _h="${!_ov}"
  for _s in $SUPERSEDED; do [ "$_h" = "$_s" ] && { echo "REFUSING: #$_n is pinned at the SUPERSEDED head $_s (gate48a's) — Kam's ruling (c) needs the NEW head: re-pin when it lands" >&2; exit 35; }; done
done
_REFS="refs/heads/develop"
for _row in "${PRS[@]}"; do IFS='|' read -r _n _t _br _rest <<< "$_row"; _REFS="$_REFS $_br refs/pull/$_n/head"; done
_LSALL="$(git -C "$REPO" ls-remote origin $_REFS)"   # ONE read of origin for develop + each branch + each pull/head
CUR_DEV="${G48B_CUR_DEV:-$(printf '%s\n' "$_LSALL" | awk '$2=="refs/heads/develop"{print $1}')}"
[ "$CUR_DEV" = "$DEVELOP_SHA" ] || { echo "REFUSING: origin develop $CUR_DEV != the pinned develop $DEVELOP_SHA — run repin_and_launch_gate48b.sh (it re-pins and re-fills in one action)" >&2; exit 17; }
NOTE="origin develop $CUR_DEV == the pinned develop; END_TREE $END_TREE"
for _row in "${PRS[@]}"; do
  IFS='|' read -r _n _t _br _h _f _a _mb _bh _p _ti <<< "$_row"
  _ov="G48B_HEAD_$_n"; [ -n "${!_ov:-}" ] && _h="${!_ov}"
  _ov="G48B_PATHS_$_n"; [ -n "${!_ov:-}" ] && _p="${!_ov}"
  # WHOLE-FIELD equality (awk $1==h), never a regex prefix
  if [ "$(printf '%s\n' "$_LSALL" | awk -v h="$_h" -v r="$_br" '$1==h && $2==r' | wc -l | tr -d ' ')" != 1 ] || [ "$(printf '%s\n' "$_LSALL" | awk -v h="$_h" -v r="refs/pull/$_n/head" '$1==h && $2==r' | wc -l | tr -d ' ')" != 1 ]; then
    echo "REFUSING: #$_n — $_h is not at $_br AND refs/pull/$_n/head on origin — the head moved (or the pin is wrong); a verdict is valid ONLY at its head" >&2
    printf '%s\n' "$_LSALL" >&2; exit 6
  fi
  COMPARE="$(set -a; . "$SECUURA_ENV"; set +a   # the token lives only in this subshell, never in the exec'd agent's env
  H="$_h" DEV="$CUR_DEV" python3 - <<'PY'
import json, os, urllib.request
t = os.environ.get("GH_TOKEN", "")
try:
    c = json.load(urllib.request.urlopen(urllib.request.Request("https://api.github.com/repos/Secuura/Distributed_Secuura/compare/%s...%s" % (os.environ["DEV"], os.environ["H"]), headers={"Authorization": "Bearer " + t, "Accept": "application/vnd.github+json"}), timeout=60))
    fs = sorted(x["filename"] for x in (c.get("files") or []))
    print("%s ahead=%d behind=%d files=%d paths=%s" % (c["merge_base_commit"]["sha"], c["ahead_by"], c["behind_by"], len(fs), ",".join(fs)))
except Exception as e: print("ERROR %s" % e)
PY
)"
  [ "$COMPARE" = "${_mb} ahead=${_a} behind=${_bh} files=${_f} paths=${_p}" ] || { echo "REFUSING: #$_n — the compare is not the pinned merge_base / ahead / behind / files:" >&2; echo "  got  $COMPARE" >&2; echo "  want ${_mb} ahead=${_a} behind=${_bh} files=${_f} paths=${_p}" >&2; exit 10; }
  grep -qwF "$_h" "$BRIEF" && grep -qwF "$_h" "$PROMPT_FILE" || { echo "REFUSING: the capture and the prompt do not both name #$_n's head $_h in full" >&2; exit 20; }
  NOTE="$NOTE | #$_n $_ti ${_h:0:12} compare ok (behind $_bh)"
done
PROMPT_JOINED="$(python3 -c 'import re,sys; print(re.sub(r"\n\s*", " ", open(sys.argv[1], encoding="utf-8").read()))' "$PROMPT_FILE")"
has() { printf '%s' "$PROMPT_JOINED" | grep -qF -- "$1"; }
has 'PR #1354 is KS-470' && has 'PR #1355 is KS-1378' && grep -qF -- '#1354 is KS-470' "$BRIEF" && grep -qF -- '#1355 is KS-1378' "$BRIEF" || { echo "REFUSING: the capture and the prompt do not BOTH state 'PR #1354 is KS-470' and 'PR #1355 is KS-1378'" >&2; exit 32; }
has '#1354 T2' && has '#1355 T1' && has 'The gate is FROZEN at TWO PRs' && has 'TIER 1 = a DEPENDENCY CHANGE IN A SHIPPED IMAGE' && has 'ROUND 1 of its own PR' && has 'a round-1 NO GO goes back to the author for round 2' && has 'ship the closed instances, ticket the residue' || { echo "REFUSING: the tier lines / FROZEN at TWO / the rounds / the tiering rule" >&2; exit 7; }
has 'MEASURE, not conclude' && has 'RULE WHETHER IT BLOCKS' && has 'NAMES THE TREE' && has 'A check that prints nothing needs a control that prints' || { echo "REFUSING: MEASURE, not conclude / RULE WHETHER IT BLOCKS / NAMES THE TREE / the printing control" >&2; exit 33; }
for _w in {{KEYWORDS}}; do   # AS A TOKEN: bounded by a non-[A-Za-z0-9-] character (MODES is not satisfied by MODES-X)
  printf '%s' "$PROMPT_JOINED" | grep -qE -- "(^|[^A-Za-z0-9-])$_w([^A-Za-z0-9-]|\$)" || { echo "REFUSING: the prompt does not carry the by-name keyword '$_w'" >&2; exit 33; }
done
has 'If @connectrpc/connect-node code is reachable in a SHIPPED artefact' && has 'the RUNTIME line FIRST' && has 'A NOT MEASURED runtime question on a T1 PR is not a GO' || { echo "REFUSING: the RUNTIME-FIRST rule" >&2; exit 34; }
has "KAM'S RULING — not the advisory-baseline grant" && has 'Accept permanently, like the twelve siblings' && has 'the exception question is closed by his word' || { echo "REFUSING: Kam's ruling as the authority for #1354" >&2; exit 36; }
has 'NO MERGE, NO PUSH to any PR branch, NO ticket state change' && has 'NO ticket filed' && has 'NEVER WRITE THE SHARED CHECKOUT. Findings only.' && has 'Never `rm`' && has 'No `az` command of any kind' && has 'You NEVER reply to Peter' && has 'Never read or write a real `.env`' && has 'Docker runs are `--rm --network none` with the source mounted READ-ONLY' && has 'NO checklist tick' && has 'no Akto scan against any stack' && has 'No audit-baseline row edited' && has 'THE NAMED EXCEPTIONS to those holds — these and NO others' && has 'NEVER `up`, `down`' && has '"no lock regenerated" means no REAL lock' || { echo "REFUSING: the HOLDS" >&2; exit 39; }
has "WEDNESDAY'S signed GO naming the heads" && has '{{GO}}' && has 'The merge seat is {{MERGE_SEAT}}' || { echo "REFUSING: merge authority / the GO string" >&2; exit 26; }
has '## MERGE ADDENDUM' && has 'ONE LINE' && has 'KEY-FREE' && has 'MG-11 subject <= 92' && has 'the FLEET STOP after this merge' && has 'MG-1 3 over 3 then 5 over 5 paths' && has 'BASELINE audit-baseline.json == #1354 head blob' && has 'FUSE 2026-10-09 rows at END' && has 'RUNTIME <connect-node not shipped' && has 'EVERY SUBJECT YOU PROPOSE MUST BE TRUE OF THE DIFF' && has 'is the LAST thing you write to report.md' && has 'NOTHING to report.md after you send the verdict mail' && has 'sha256 report.md = <hex>' || { echo "REFUSING: the MERGE ADDENDUM rules / REPORT-HASH-LAST" >&2; exit 25; }
grep -qF -- '{{VERDICT_SUBJECT}}' "$PROMPT_FILE" && has 'FROM coagent@agentmail.to' && has 'reports/{{REPORT}}/' || { echo "REFUSING: the verdict subject / sender / report dir" >&2; exit 23; }
grep -qF -- "$DEVELOP_SHA" "$PROMPT_FILE" && grep -qF -- "$END_TREE" "$PROMPT_FILE" || { echo "REFUSING: the launch develop / the END_TREE not in full" >&2; exit 31; }
_ANY_OVR="$(env | grep -c '^G48B_')"
if [ "${1:-}" = "--check" ]; then
  echo "all guards pass:"
  echo "  $NOTE" | cut -c1-400
  echo "  capture, row reason, lockdelta / baseline / lockcensus (+ develop control) / overlaps / image / gh reads, both PR bodies, prompt, QA project and repo present; kit at its filled home; TWO rows; no SUPERSEDED head; thinking directive; every kit file + commission + gate48a report named; no unfilled token"
  echo "  2 of 2 heads at branch AND pull/head (whole-field); 2 of 2 GitHub compares == the pins; capture and prompt BOTH carry each head in full and both tickets"
  echo "  the prompt carries {{N_KW}} by-name keywords (as tokens), the tier lines, the RUNTIME-FIRST rule, Kam's ruling as #1354's authority, the holds + named exceptions, the GO string, the addendum + REPORT-HASH-LAST rules, the verdict subject + report dir, develop / END_TREE in full"
  [ "$_ANY_OVR" != 0 ] && echo "  (a G48B_* TEST OVERRIDE is set)"
  echo "  a launch (not --check) will refuse unless stdin is a TTY (exit 21)"
  exit 0
fi
[ -t 0 ] || { echo "REFUSING: stdin is not a TTY — run this launcher in a cockpit pane, never inside a Bash tool" >&2; exit 21; }
[ "$_ANY_OVR" = 0 ] || { echo "REFUSING: a launch with test overrides set" >&2; exit 16; }
echo "$NOTE" >&2
cd "$QA_DIR" || { echo "cannot enter $QA_DIR" >&2; exit 16; }
exec claude --dangerously-skip-permissions "$(cat "$PROMPT_FILE")"
