/**
 * gateD2 C3 REQUEST-BYTES DIFFERENTIAL PROBE — written by the gate kit. Runs at BASE (buildTimestampRequest = the node-forge encoder) and at
 * HEAD (the local der.ts writer) over the SAME 24 inputs, and writes `{case: hex}` to $GD2_REQ_OUT. c3_reqbytes_gateD2.py compares the two
 * files byte for byte. Cell 16 pins 3 vectors captured by the builder; this probe re-derives forge's output itself, including the quirks
 * Wednesday ruled must be kept (AT MOST one leading 0x00 stripped from the nonce; NO sign prefix on a high-bit nonce).
 * Copied into a SCRATCH worktree only as services/timestamping/src/__tests__/zz-gD2-probe-reqbytes.test.ts, run alone, QUARANTINED.
 */
import { describe, it, expect } from 'vitest';
import * as fs from 'fs';
import { buildTimestampRequest } from '../tsa/rfc3161-client';

const NONCES = ['0011223344556677', '0000112233445566', '00007f0011223344', '0080112233445566', 'ff00112233445566', '8000000000000000',
  '7fffffffffffffff', '0000000000000001', '00', '01', 'a1b2c3d4e5f60718', '00ff'];
const ALGS: Array<['SHA-256' | 'SHA-384' | 'SHA-512', number]> = [['SHA-256', 32], ['SHA-384', 48], ['SHA-512', 64]];

describe('gateD2 request-bytes differential probe', () => {
  it('R1 builds every case and writes them to GD2_REQ_OUT', () => {
    const out: Record<string, string> = {};
    for (const [alg, n] of ALGS) for (const nonce of NONCES) for (const certReq of [true, false]) for (const pol of [undefined, '1.3.6.1.4.1.99999.1.1']) {
      if (alg !== 'SHA-256' && (pol || !certReq)) continue;
      const k = `${alg}|${nonce}|${certReq}|${pol ?? '-'}`;
      out[k] = buildTimestampRequest({ hash: 'ab'.repeat(n), hashAlgorithm: alg, nonce, certReq, ...(pol ? { policyOid: pol } : {}) }).toString('hex');
    }
    const f = process.env.GD2_REQ_OUT; expect(f, 'GD2_REQ_OUT must be set').toBeTruthy();
    fs.writeFileSync(f as string, JSON.stringify(out, null, 1));
    console.log(`GD2 REQ wrote ${Object.keys(out).length} cases to ${f}`);
    expect(Object.keys(out).length).toBeGreaterThan(20);
  });
});
