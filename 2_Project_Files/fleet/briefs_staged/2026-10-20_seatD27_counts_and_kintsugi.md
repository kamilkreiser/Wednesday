```
BEGIN READ ONLY;
SELECT set_config('app.tenant_scope_bypass', 'platform_admin', true);
WITH cand AS (
  SELECT d.external_id, d.metadata->>'versionedById' AS minter,
         p.external_id AS parent_ext, p.tenant_id AS parent_tenant
    FROM documents d
    LEFT JOIN documents p ON p.id = d.parent_document_id
   WHERE d.metadata ? 'versionAction'
     AND d.external_id IS NOT NULL
     AND NOT EXISTS (SELECT 1 FROM action_provenance x
                      WHERE x.document_id = d.external_id
                        AND x.action IN ('create', 'create:connector'))
), pairs AS (
  SELECT parent_tenant, parent_ext, minter, COUNT(*) AS children
    FROM cand
   WHERE minter LIKE 'connector:%' AND parent_ext IS NOT NULL
   GROUP BY 1, 2, 3
)
SELECT pr.parent_tenant, pr.parent_ext, pr.minter, pr.children,
       (SELECT COUNT(*) FROM action_provenance ap
         WHERE ap.document_id = pr.parent_ext AND ap.tenant_id = pr.parent_tenant
           AND ap.action LIKE 'version:%' AND ap.connector_id = pr.minter) AS rows_naming_minter,
       (SELECT COUNT(*) FROM cand) AS control_all_candidates,
       (SELECT COUNT(*) FROM documents) AS control_documents_total
  FROM pairs pr
 ORDER BY (pr.children) DESC, 1, 2, 3;
COMMIT;
```
