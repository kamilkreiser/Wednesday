def wait():
    for _ in range(120):
        if os.getloadavg()[0]<7.5: return
        time.sleep(5)
    raise SystemExit("load stayed high")
def vt(tag,*files):
    wait()
    env=dict(os.environ,TZ="UTC"); env.pop("DATABASE_URL",None)
    p=f"{O}/{tag}.json"
    if os.path.exists(p): os.rename(p,p+f".{int(time.time())}")
    r=subprocess.run(["nice","-n","5",DEV+"/node_modules/.bin/vitest","run","--root",SVC,"--reporter=json","--outputFile="+p,*files],env=env,capture_output=True,text=True,timeout=280)
    open(f"{O}/{tag}.log","w").write(r.stdout+r.stderr)
    return json.load(open(p)),r
def cells(j): return [(a["fullName"],a["status"]) for f in j["testResults"] for a in f["assertionResults"]]
def sha(p): return hashlib.sha256(open(p,"rb").read()).hexdigest()
FILES=[IDX,ACC,MINT]; S={p:sha(p) for p in FILES}
def tamper(tag,path,old,new,must_red,must_green=()):
    b=open(path,"rb").read(); s=b.decode()
    assert s.count(old)>=1, f"ANCHOR MISSING {tag}"
    n=s.replace(old,new,1) if s.count(old)==1 else s.replace(old,new)
    assert n!=s
    open(path,"wb").write(n.encode())
    assert new=="" or new in open(path).read(), "tamper did not land"
    try: j,_=vt(tag,DOOR,UNIT)
    finally: open(path,"wb").write(b)
    assert sha(path)==S[path]
    c=cells(j); red=[x for x,st in c if st=="failed"]
    miss=[m for m in must_red if not any(m in x for x in red)]
    wrong=[m for m in must_green if any(m in x for x in red)]
    print(f"{tag}: {len(red)} of {len(c)} red | required red missing {miss} | required green but red {wrong} | loaded {len(c)>0}")
    return red
