#!/bin/zsh
set -u
G=/private/tmp/claude-501/-Volumes-DevMASTER-WEDNESDAY/f2077cbf-bb12-49ee-8d59-f5f537b335e0/scratchpad/g1527
D=$G/head/Blockchain/Dev
PORT=55424; PGBIN=/opt/homebrew/opt/postgresql@15/bin
export PATH=$PGBIN:$PATH PGHOST=127.0.0.1 PGPORT=$PORT PGUSER=secuura POSTGRES_USER=secuura
export APP_DB_PASSWORD=$(cat $G/app_pw)
for f in $D/docker/init/*; do
  b=$(basename $f)
  case $b in
    00-*) continue;;
    *.sql) psql -d secuura -v ON_ERROR_STOP=1 -q -f $f > $G/init_$b.log 2>&1; echo "init $b rc=$?";;
    *.sh) bash $f > $G/init_$b.log 2>&1; echo "init $b rc=$?";;
  esac
done
DATABASE_URL=postgresql://secuura@127.0.0.1:$PORT/secuura sh $D/scripts/run-migrations.sh > $G/migrations.log 2>&1; echo "run-migrations rc=$?"
tail -4 $G/migrations.log
psql -d secuura -At -c "SELECT relname, relrowsecurity, relforcerowsecurity FROM pg_class WHERE relname='svc_api_keys'"
psql -d secuura -At -c "SELECT rolname, rolsuper, rolbypassrls FROM pg_roles WHERE rolname='secuura_app'"
psql -d secuura -At -c "SHOW default_transaction_isolation"
print "export TEST_DATABASE_URL=postgresql://secuura@127.0.0.1:$PORT/secuura\nexport TEST_APP_DATABASE_URL=postgresql://secuura_app:$APP_DB_PASSWORD@127.0.0.1:$PORT/secuura" > $G/pg.env
echo SETUP DONE
