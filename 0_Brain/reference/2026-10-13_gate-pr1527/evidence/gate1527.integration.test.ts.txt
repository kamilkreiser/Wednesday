const { privateKey, publicKey } = crypto.generateKeyPairSync('rsa', { modulusLength: 2048 });
const PRIVATE_PEM = privateKey.export({ type: 'pkcs8', format: 'pem' }).toString();
const PUBLIC_PEM = publicKey.export({ type: 'spki', format: 'pem' }).toString();
const b64url = (b: Buffer): string => b.toString('base64').replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
function token(): string {
  const header = b64url(Buffer.from(JSON.stringify({ alg: 'RS256', typ: 'JWT' })));
  const now = Math.floor(Date.now() / 1000);
  const payload = b64url(Buffer.from(JSON.stringify({ sub: 'gate1527', iat: now, exp: now + 600, role: 'super_admin' })));
  const sig = b64url(crypto.sign('RSA-SHA256', Buffer.from(`${header}.${payload}`), PRIVATE_PEM));
  return `${header}.${payload}.${sig}`;
}

const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));

type Minted = { status: number; id: string | null; key: string | null; priorKeysRevoked: unknown; ms: number };
async function mint(tenant: string, connectorId: string, rotate: boolean, name: string): Promise<Minted> {
  const t0 = Date.now();
  const res = await fetch(`${base}/api/keys`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token()}` },
    body: JSON.stringify({ name, scopes: ['documents:read'], tenantId: tenant, organizationId: ORG, connectorId, rotate }),
  });
  const text = await res.text();
  let d: any = {};
  try { d = JSON.parse(text).data ?? {}; } catch { /* */ }
  return { status: res.status, id: d.id ?? null, key: d.key ?? null, priorKeysRevoked: d.priorKeysRevoked, ms: Date.now() - t0 };
}
async function validate(key: string): Promise<{ status: number; valid: unknown; reason: unknown; ms: number }> {
  const t0 = Date.now();
  const res = await fetch(`${base}/api/keys/validate`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ key }) });
  const j: any = await res.json().catch(() => ({}));
  return { status: res.status, valid: j?.data?.valid, reason: j?.data?.reason ?? j?.error?.message, ms: Date.now() - t0 };
}
type Row = { id: string; is_active: boolean; expires_at: Date | null; name: string };
async function rowsOf(tenant: string, c: string): Promise<Row[]> {
  return (await superPool.query<Row>('SELECT id, is_active, expires_at, name FROM svc_api_keys WHERE connector_id=$1 AND tenant_id=$2::uuid', [c, tenant])).rows;
}
const act = (rows: Row[]) => rows.filter((r) => r.is_active).map((r) => r.id).sort();
const memActive = (tenant: string, c: string) => [...mem.values()].filter((k) => k.connectorId === c && k.tenantId === tenant && k.isActive).map((k) => k.id).sort();
/** Every DB row's is_active must equal memory's; returns the disagreements. */
function memDisagree(rows: Row[]): string[] {
  const bad: string[] = [];
  for (const r of rows) {
    const m = mem.get(r.id);
    if (!m) { bad.push(`${r.id}: in DB, absent from memory`); continue; }
    if (m.isActive !== r.is_active) bad.push(`${r.id}: db=${r.is_active} mem=${m.isActive}`);
  }
  return bad;
}
const conn = (cell: string) => `${PREFIX}${cell}:${crypto.randomUUID()}`;
async function seeded(tenant: string, c: string): Promise<Minted> {
  const s = await mint(tenant, c, false, 'seed');
  expect(s.status, 'arrange: seed 201').toBe(201);
  return s;
}
function settledFlag<T>(p: Promise<T>) { const f = { done: false }; p.then(() => { f.done = true; }, () => { f.done = true; }); return f; }

  it('A1 mixed rotate/non-rotate N=2..50: FAIL if any non-201, rotate-key actives != 1, actives == 0, or memory != DB', async () => {
    const out: any[] = []; const bad: string[] = [];
    for (const N of [2, 3, 5, 10, 20, 35, 50]) {
      for (let t = 0; t < 3; t += 1) {
        const c = conn(`a1-${N}-${t}`);
        await seeded(T1, c);
        const kinds = Array.from({ length: N }, (_, i) => (i % 3 !== 1)); // >=1 rotate, >=1 plain when N>=2
        const res = await Promise.all(kinds.map((rot, i) => mint(T1, c, rot, `a1 ${rot ? 'rot' : 'plain'} ${i}`)));
        const rows = await rowsOf(T1, c);
        const rotIds = new Set(res.filter((_, i) => kinds[i]).map((r) => r.id));
        const activeRot = rows.filter((r) => r.is_active && rotIds.has(r.id)).length;
        const a = act(rows);
        const dis = memDisagree(rows);
        const non201 = res.filter((r) => r.status !== 201).length;
        const nullRevoked = res.filter((r, i) => kinds[i] && typeof r.priorKeysRevoked !== 'number').length;
        out.push({ N, t, rows: rows.length, active: a.length, activeRot, non201, nullRevoked, memDisagree: dis.length });
        if (non201 || activeRot !== 1 || a.length < 1 || dis.length || rows.length !== N + 1 || nullRevoked) bad.push(JSON.stringify(out[out.length - 1]) + ' ' + dis.join(';'));
        if (JSON.stringify(memActive(T1, c)) !== JSON.stringify(a)) bad.push(`N=${N} t=${t} memActive != dbActive`);
      }
    }
    results.A1 = out;
    expect(bad).toEqual([]);
  }, 120_000);

  it('A2 same connector id, two tenants: FAIL if T2 rotate blocks on T1 lock, T1 rotate does NOT block (control), or a tenant ends != 1 active', async () => {
    const c = conn('a2');
    const s1 = await seeded(T1, c); const s2 = await seeded(T2, c);
    const holder: PoolClient = await superPool.connect();
    await holder.query('BEGIN');
    await holder.query('SELECT pg_advisory_xact_lock(hashtext($1), hashtext($2))', [T1, c]);
    const p1 = mint(T1, c, true, 'a2 t1 rot'); const f1 = settledFlag(p1);
    const t0 = Date.now();
    const r2 = await Promise.race([mint(T2, c, true, 'a2 t2 rot'), sleep(3000).then(() => null)]);
    const t2ms = Date.now() - t0;
    await sleep(Math.max(0, 1500 - t2ms));
    const t1BlockedWhileHeld = !f1.done;
    const t1rowsWhileHeld = (await rowsOf(T1, c)).length;
    await holder.query('COMMIT'); holder.release();
    const r1 = await p1;
    const a1 = act(await rowsOf(T1, c)); const a2 = act(await rowsOf(T2, c));
    // concurrent double-submits in both tenants on the same id, 10 trials
    const bad: string[] = [];
    for (let t = 0; t < 10; t += 1) {
      const cc = conn(`a2x-${t}`);
      await seeded(T1, cc); await seeded(T2, cc);
      const rs = await Promise.all([mint(T1, cc, true, 'x1a'), mint(T2, cc, true, 'x2a'), mint(T1, cc, true, 'x1b'), mint(T2, cc, true, 'x2b')]);
      const x1 = act(await rowsOf(T1, cc)); const x2 = act(await rowsOf(T2, cc));
      if (rs.some((r) => r.status !== 201) || x1.length !== 1 || x2.length !== 1 || !rs.map((r) => r.id).includes(x1[0]) || !rs.map((r) => r.id).includes(x2[0])) bad.push(`trial ${t}: t1=${x1.length} t2=${x2.length}`);
      const dis = memDisagree([...(await rowsOf(T1, cc)), ...(await rowsOf(T2, cc))]); if (dis.length) bad.push(`trial ${t} mem: ${dis.join(';')}`);
    }
    results.A2 = { t2Status: r2 && (r2 as Minted).status, t2ms, t1BlockedWhileHeld, t1rowsWhileHeld, r1: r1.status, a1, a2, s1: s1.id, s2: s2.id, r1id: r1.id, r2id: r2 && (r2 as Minted).id, bad };
    expect(r2 && (r2 as Minted).status, 'T2 rotate completes while T1 lock held').toBe(201);
    expect(t1BlockedWhileHeld, 'CONTROL: T1 rotate blocked by externally held (hashtext(T1),hashtext(c))').toBe(true);
    expect(t1rowsWhileHeld, 'CONTROL: T1 insert had not landed while lock held').toBe(1);
    expect(a1).toEqual([r1.id]); expect(a2).toEqual([(r2 as Minted).id]);

  it('A11 pool pressure: 50 rotates on one connector + 20 validates of another key: FAIL if any validate is not 200/valid or takes >5s', async () => {
