38 // =============================================================================
39 // KS-522 — honest labelling of simulated anchors
40 // =============================================================================
41 // A `tx_sim_…`/`mock_tx_`/`mock_batch_tx_`/`tx_…` transaction hash corresponds to NO Cardano
42 // transaction (SIMULATE_ANCHORING or the legacy dev mock wrote it). Simulation
43 // is legitimate for local/demo use — presenting its output as on-chain proof
44 // is not, because the verify response is the artefact a relying party acts on.
45 // Every verify response passes its blockchain blob through this presenter:
46 // a simulated anchor keeps document-level verification (content integrity +
47 // certification status are real) but makes NO on-chain claim — `txHash` is
48 // nulled (the placeholder moves to `simulatedTxRef`), `anchored` flips false,
49 // and a machine-readable `simulated: true` marker is set (also surfaced
50 // top-level by the response builders).
51 const FABRICATED_TXHASH_RE = /^(tx_sim_|mock_tx_|mock_batch_tx_|tx_)/;

53 export function presentBlockchainHonestly(blob: unknown): unknown {
54   if (!blob || typeof blob !== 'object') return blob;
55   const b = blob as Record<string, unknown>;
56   const tx = typeof b.txHash === 'string' ? b.txHash : undefined;
57   // KS-1628: any TRUTHY `simulated` counts ('true', 1), not only `=== true` — the rule the retired
58   // gateway handler applied (`Boolean(persistedSimulated)`, KS-1069). Absent or false still does not.
59   const simulated = Boolean(b.simulated) || (tx !== undefined && FABRICATED_TXHASH_RE.test(tx));
60   if (!simulated) return blob;
61   return {
62     ...b,
63     simulated: true,
64     anchored: false,
65     txHash: null,
66     ...(tx ? { simulatedTxRef: tx } : {}),
67     // A scan URL for a non-existent tx is itself a false claim.
68     cardanoScanUrl: undefined,
69   };
70 }

73 // KS-563 — certified is not anchored
74 // =============================================================================
75 // `checks.isCertified` used to be true for `certified | anchored | signed`, so a
76 // document that was merely originated and auto-anchored (every Platform S
77 // upload) reported certification nobody performed — a live false "Certified by
78 // issuer" in front of end users. Certification is now the narrow claim it
79 // names: an issuer certified this document. Anchoring gets its own honest
80 // flag, and the certification act — who/when/which certification — is
81 // surfaced as an object that is ABSENT when no certification happened.
82 //
83 // `verified` / `integrityVerified` deliberately keep the OLD broad set
84 // (VERIFIABLE_STATUSES): their meaning is "in the registry, hash intact, not
85 // revoked", which is unchanged by this fix. Narrowing them would flip every
86 // upload-only document to `verified: false` and break consumers for a defect
87 // that lives in the certification flag alone.
88 const CERTIFIED_STATUS = 'certified';
89 const VERIFIABLE_STATUSES = new Set([CERTIFIED_STATUS, 'anchored', 'signed']);

101 /**
102  * KS-1628: a positive integer NUMBER, or a positive-integer digit STRING. The retired gateway handler
103  * took numbers only (KS-1069 item 1); Wednesday ruled (b) on 2026-10-19 to accept the digit string too,
104  * because it is a real stored shape - originate's heal path writes anchoring's raw pg BIGINT (KS-1129),

110   return typeof value === 'string' && DIGIT_HEIGHT_RE.test(value);

113 /**
114  * True only for a real, non-simulated on-chain anchor. KS-522 rules apply: a placeholder
115  * (`tx_sim_`/`mock_tx_`/`mock_batch_tx_`/`tx_…`, KS-1628) is not an anchor; never sets this.
116  * Takes the blob ALREADY passed through presentBlockchainHonestly (which flips
117  * `anchored` false and nulls `txHash` on simulated rows).
118  */
119 export function isAnchoredHonestly(honestChain: unknown, status?: unknown): boolean {
120   if (honestChain && typeof honestChain === 'object') {
121     // A chain blob is the authoritative answer — do NOT fall back to the
122     // document status when one exists. `documents.status` flips to 'anchored'
123     // when the anchor is SUBMITTED, so a still-pending anchor (status
124     // 'pending', txHash null) would otherwise report isAnchored: true — the
125     // same unearned claim this ticket exists to remove, one field over.
126     const c = honestChain as Record<string, unknown>;
127     // KS-1628: truthy, not `=== true` — see presentBlockchainHonestly.
128     if (c.simulated) return false;
129     // KS-1628: a failed anchor is not an anchor, whatever else the blob says. The chain-first blobs
130     // carry `anchored: true` and no status, so they are unaffected; a stored blob that pairs
131     // `anchored: true` with a failed status was on-chain before this line.
132     if (FAILED_ANCHOR_STATUSES.has(String(c.status))) return false;
133     // The stored blob is `{status, txHash, network, anchorId, anchoredAt,
134     // blockHeight}` — anchoring's vocabulary is pending | submitted |
135     // confirmed | failed (services/anchoring/src/index.ts) and there is NO
136     // `anchored` boolean on a real row. The chain-first branches DO build a
137     // blob with `anchored: true`, so accept either shape. Only 'confirmed'
138     // counts: 'submitted' means the tx is in flight and may still fail — the
139     // KS-535 class — so it has not earned the claim yet.
140     const settled = c.anchored === true || c.status === 'confirmed';
141     // KS-1628: a real hash is 64 hex characters (REAL_TXHASH_RE), not merely a non-empty string, and a
142     // height that IS reported must be a real one (isAnchorHeight). A height that is not reported at all
143     // (undefined, or the null anchoring's toBlockNumber returns when it has none) keeps today's answer:
144     // the chain-first branches build their blob from that value, and ks563 pins that shape as on-chain.
145     const heightOk = c.blockHeight === undefined || c.blockHeight === null || isAnchorHeight(c.blockHeight);
146     return settled && typeof c.txHash === 'string' && REAL_TXHASH_RE.test(c.txHash) && heightOk;
147   }
148   // No chain blob at all — the document's own status is all we have.
149   return status === 'anchored';
150 }

154  * Built from `certification_metadata`, which `POST /api/certifications/issue`
155  * already writes (issuerName / certificationId / certificationType /

160   if (!isCertifiedStatus(doc?.status)) return undefined;

173     certifierOrganizationId: certMeta.issuerId ?? null,
174   };

211 // =============================================================================
212 // KS-1561 — a certification is presented only for the holder of the ORIGINAL registration
213 // =============================================================================
214 // Before KS-1561 a 'certified' row won selection whoever held it, and siblingCertification lent any certified sibling's
215 // certification to the selected row. So organisation A could register organisation B's bytes AFTER B, certify its own
216 // copy, and make verify(H) answer "certified by Org A" (Seat J 5th, 2026-10-12, real routes, real Postgres).
217 // Kam ruled option a (card secuura-ks1561-duplicate-hash-certifier-1102, 2026-11-02): "Certify only for the org holding
218 // the ORIGINAL registration". Wednesday's design ruling (2026-11-04, D1-D4) defines it:
219 //   D1 the ORIGINAL rows are those tied at the earliest created_at. created_at is server-set on every write path
220 //      (documentRepo saveDocument NOW(); certificationRepo new Date()), so a caller cannot choose it, and a later
221 //      registrant cannot overtake an original whose anchor failed (earliest ANCHORED was ruled out for that reason);
222 //   D2 a row may testify to a certification only when it IS an original row, or carries the SAME non-null
223 //      issuer_organization_id or the SAME non-null owner_user_id as an original row; a KS-1557-trusted lineage row
224 //      also qualifies when its parent is an original row. Unknown identity is not a match (fail closed): a portal
225 //      registration carries no organisation (issuer_organization_id is written only on a bound organizationUuid
226 //      claim, KS-597), so a re-registration by a different user of the same portal organisation no longer shows as
227 //      certified. Kam's card names that cost: "Some legitimate re-registrations stop showing as certified".
228 /** KS-1561: a comparable identity, or null for an absent / blank value (null never matches anything). */
229 const identityOf = (value: unknown): string | null => {
230   if (value === null || value === undefined) return null;
231   const s = String(value).trim().toLowerCase();
232   return s.length > 0 ? s : null;
233 };

235 export interface CertificationEligibility {
236   /** KS-1561 D2: may this row's OWN 'certified' status be presented for the set's bytes? */
237   certifiedRow: (row: any) => boolean;
238   /** KS-1561 D2: may the KS-1557-trusted evidence this (derived) row carries be presented for the set's bytes? */
239   evidenceRow: (row: any) => boolean;
240 }
241 
242 /** KS-1561: the eligibility rule over ONE same-hash registration set (see the block comment above). */
243 export function certificationEligibility(rows: any[]): CertificationEligibility {
244   const list = rows || [];
245   const earliest = list.reduce((min: number, r: any) => Math.min(min, rowCreatedMs(r)), Number.MAX_SAFE_INTEGER);
246   // D1: every row tied at the earliest instant is an original; rows with no readable created_at sort last
247   // (rowCreatedMs gives MAX_SAFE_INTEGER), so they are originals only when no row carries a readable time.
248   const originals = list.filter((r: any) => rowCreatedMs(r) === earliest);
249   const originalRows = new Set<unknown>(originals);
250   const collect = (pick: (r: any) => unknown): Set<string> =>
251     new Set(originals.map((r: any) => identityOf(pick(r))).filter((v): v is string => v !== null));
252   const originalIds = collect((r) => r?.id);
253   const originalOrgs = collect((r) => r?.issuer_organization_id);
254   const originalOwners = collect((r) => r?.owner_user_id);
255   const certifiedRow = (row: any): boolean => {
256     if (originalRows.has(row)) return true;
257     const org = identityOf(row?.issuer_organization_id);
258     if (org !== null && originalOrgs.has(org)) return true;
259     const owner = identityOf(row?.owner_user_id);
260     return owner !== null && originalOwners.has(owner);
261   };
262   const evidenceRow = (row: any): boolean => {
263     if (certifiedRow(row)) return true;
264     const parent = identityOf(row?.parent_document_id);
265     return parent !== null && originalIds.has(parent);
266   };
267   return { certifiedRow, evidenceRow };
268 }

270 /**
271  * Pick the row that best evidences the platform's claims for these bytes:
272  *   1. a `certified` row (the legacy certify path writes one) — KS-1561: only one the ORIGINAL holder may present;
273  *   2. else a row with a REAL on-chain tx (KS-522, KS-1628: no FABRICATED_TX_PREFIX
274  *      placeholder counts) — 'confirmed' chain state first, then the ORIGINAL registration
275  *      (created_at ASC), since derived signed_document rows are later copies;
276  *   3. else the oldest row — deterministic, where the legacy NULLS LAST order
277  *      was arbitrary.
278  * Input rows keep the query's certified_at DESC pre-sort, so within rule 1 the
279  * most recent ELIGIBLE certification still wins.
280  * KS-1561: a certified row the original's holder may NOT present is left out of every rule, not merely out of rule 1:
281  * its stored status is 'certified', so selecting it under rule 2 or 3 would still present its certification
282  * (buildCertification and isCertifiedStatus both read the status). An original row is always a candidate, so the
283  * candidate set is never empty. Before KS-1561 every row was a candidate and rule 1 took any certified row.
284  */
285 function selectVerifiableRow(rows: any[], judged?: CertificationEligibility): any | undefined {
286   if (!rows || rows.length === 0) return undefined;
287   if (rows.length === 1) return rows[0];
288   const eligibility = judged ?? certificationEligibility(rows); // KS-1561; KS-1986: v1's verify passes its cross-tenant judgment
289   const candidates = rows.filter((r) => !isCertifiedStatus(r?.status) || eligibility.certifiedRow(r)); // KS-1561
290   const certified = candidates.filter((r) => isCertifiedStatus(r?.status));
291   if (certified.length > 0) return certified[0];
292   const anchored = candidates.filter(rowHasRealTx).sort((a, b) => { // KS-1561: was rows.filter
293     const aConfirmed = chainBlobOfRow(a)?.status === 'confirmed' ? 0 : 1;
294     const bConfirmed = chainBlobOfRow(b)?.status === 'confirmed' ? 0 : 1;
295     if (aConfirmed !== bConfirmed) return aConfirmed - bConfirmed;
296     return rowCreatedMs(a) - rowCreatedMs(b);
297   });
298   if (anchored.length > 0) return anchored[0];
299   return [...candidates].sort((a, b) => rowCreatedMs(a) - rowCreatedMs(b))[0]; // KS-1561: was [...rows]
300 }

302 /**
303  * KS-1557 (read side). Which derived 'signed' rows may testify to a certification act?
304  *
305  * The lineage certify path (`POST /api/certifications/issue` with parentDocumentId) records the act on a DERIVED row (status 'signed',
306  * `metadata.certificationId`) and leaves the parent untouched. The derived row's `metadata` is filled from the CALLER'S `data` (the
307  * caller's keys are spread AFTER the server's), so everything in it - certificationId, certifiedAt, certificationType, issuerId - is
308  * caller-chosen. Reading the evidence from there let a caller choose what verify presented, and let a row whose hash is not its
309  * parent's count. The rule now: a derived row counts only when ALL of these hold, and the presented id, type and time come from the
310  * real certification row, never from the derived row's metadata:
311  *   1. status 'signed', and `metadata.certificationId` is a UUID;
312  *   2. its parent link is set, and the parent's hash equals its own (the parent is read from the same-hash set, else one lookup);
313  *   3. its external id carries the first 8 hex characters of that certification id (the route generates doc-<ms>-<certId8>), so a row
314  *      cannot cite someone else's genuine certification;
315  *   4. that certification id RESOLVES, in the row's own tenant, to a real certification row (a `documents` row written by
316  *      saveCertification: certification_metadata carries certificationType and holderId);
317  *   5. the derived row's owner is the certification's issuer (owner_user_id equals issuer_user_id; both NULL for a connector principal).
318  * `certifiedBy` stays the caller-asserted display string it always was (ruled); `certifierOrganizationId` is null here because the
319  * certification row stores the issuer's USER id, not an organisation id (the organisation-id fix is the write side).
320  */
321 export const MAX_EVIDENCE_LOOKUPS = 8;
322 const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
323 const DERIVED_EXTERNAL_ID_RE = /^doc-\d+-([0-9a-f]{8})$/i;
324 const normaliseRowHash = (h: unknown): string => String(h ?? '').replace(/^sha256:/i, '').toLowerCase();

329 /** The certification id a derived row CLAIMS, when it is shaped like a certify-lineage child. No I/O. */
330 function claimedCertificationId(row: any): string | null {
331   if (row?.status !== 'signed' || !row?.parent_document_id) return null;
332   const certId = (row?.metadata as Record<string, unknown> | undefined)?.certificationId;
333   if (typeof certId !== 'string' || !UUID_RE.test(certId)) return null;
334   const m = DERIVED_EXTERNAL_ID_RE.exec(String(row?.external_id ?? ''));
335   if (!m || m[1].toLowerCase() !== certId.slice(0, 8).toLowerCase()) return null;
336   return certId.toLowerCase();
337 }

352   const parent = String(row.parent_document_id);

373 /**
374  * Resolve, for every row of the set that claims to be a certify-lineage child, whether it is a GENUINE act, and the evidence to
375  * present. Fail closed: any miss, mismatch or error leaves the row out. At most MAX_EVIDENCE_LOOKUPS claimed rows are examined.
376  * The result is keyed by the row OBJECT, so callers pass it to siblingCertification / ownRowCertification with the same rows.
377  */
378 export async function resolveTrustedCertificationEvidence(rows: any[], ctx: CertificationEvidenceContext): Promise<TrustedEvidence> {
379   const trusted = new Map<unknown, Record<string, unknown>>();
380   let examined = 0;
381   for (const r of rows || []) {
382     const claimed = claimedCertificationId(r);
383     if (!claimed) continue;
384     if (examined >= MAX_EVIDENCE_LOOKUPS) break;
385     examined += 1;
386     try {
387       const tenant = String(r?._tenantId ?? r?.tenant_id ?? '');
388       if (!UUID_RE.test(tenant)) continue;
389       const own = normaliseRowHash(r.content_hash);
390       const inSet = rows.find((p: any) => p !== r && String(p?.id) === String(r.parent_document_id));
391       const parentHash = inSet ? normaliseRowHash(inSet.content_hash) : await fetchParentHash(r, tenant, ctx);
392       if (parentHash === null || parentHash !== own) continue;
393       const cert = await fetchCertificationRow(r, claimed, tenant, ctx);
394       if (!isCertificationRow(cert, claimed, r)) continue;
395       // The certify route saves the derived row owned by the ISSUER and the certification row with the same issuer, so the two agree (both a
396       // UUID for a person, both NULL for a connector principal). A sign-cert / sign-wallet row is owned by the SOURCE's owner, so it cannot cite
397       // an actor's certification unless that actor owns the source.
398       if (String(r.owner_user_id ?? '') !== String(cert.issuer_user_id ?? '')) continue;
399       const meta = (r?.metadata || {}) as Record<string, any>;
400       const cm = (cert.certification_metadata || {}) as Record<string, any>;
401       const at = cert.certified_at ?? cert.created_at ?? null;
402       trusted.set(r, {
403         certificationId: String(cert.id),
404         certificationType: cm.certificationType,
405         certifiedAt: at ? new Date(at).toISOString() : null,
406         certifiedBy: meta.issuerName ?? meta.issuedBy ?? r?.organization_name ?? null,
407         certifierOrganizationId: null,
408       });
409     } catch (err: any) {
410       logger.warn('KS-1557: certification evidence lookup failed - the row is not counted', { rowId: r?.id, error: err?.message });
411     }
412   }
413   return trusted;
414 }

418  * derived 'signed' row that resolveTrustedCertificationEvidence accepted. `trusted` is REQUIRED (no default): a call site that forgot to
419  * resolve would otherwise fail open.
420  */
421 function siblingCertification(rows: any[], selected: any, trusted: TrustedEvidence, judged?: CertificationEligibility): Record<string, unknown> | undefined {
422   if (!rows || rows.length < 2) return undefined;
423   // KS-1561: a sibling lends its certification only when the ORIGINAL registration's holder may present it. Before
424   // KS-1561 the first certified sibling, or the first trusted evidence, was returned whoever held it.
425   const eligibility = judged ?? certificationEligibility(rows); // KS-1986: v1's verify passes its cross-tenant judgment
426   for (const r of rows) {
427     if (r === selected) continue;
428     if (isCertifiedStatus(r?.status)) {
429       if (eligibility.certifiedRow(r)) return buildCertification(r);
430       continue; // KS-1561: an ineligible certified sibling is passed over (it used to return here unconditionally)
431     }
432     const evidence = trusted.get(r);
433     if (evidence && eligibility.evidenceRow(r)) return evidence; // KS-1561: was `if (evidence)`
434   }
435   return undefined;
436 }

476 // KS-584 (interim): bounded rowset for the same-hash registration scan. Far
477 // above the real duplicate counts (max observed: 5 rows per hash) while
478 // keeping a pathological hash from dragging the whole set through JS.
479 export const VERIFY_ROWSET_LIMIT = 24;

481 // =============================================================================
482 // KS-1561 round 2 (gate 188, G188-F1) — the original comes from the FULL same-hash set
483 // =============================================================================
484 // Every same-hash row-set read is `ORDER BY certified_at DESC NULLS LAST LIMIT VERIFY_ROWSET_LIMIT`. An uncertified
485 // original sorts LAST, and a certified one with an older certification sorts below newer ones, so once
486 // VERIFY_ROWSET_LIMIT later rows are certified the original is not fetched. certificationEligibility then took
487 // MIN(created_at) over the later rows only, and a latecomer's certification was presented again (gate 188 round 1: 24
488 // later certified copies, measured on PostgreSQL 15). Wednesday's ruling on G188-F1: "the original must be computed over
489 // the FULL same-hash set, never over the rows a query happened to fetch". So when a read comes back FULL, the earliest
490 // registration(s) are read by their own bounded query and merged in by id. A read that came back short already holds the
491 // whole set (its LIMIT was not reached), so the common case costs no extra query.
492 /**
493  * KS-1561 G188-F1: the most rows tied at the earliest registration that a verify reads (one more is asked for, to see a
494  * cut). Past it the rest are not counted as originals. Eligibility only grows with the originals set, so a cut tie group
495  * presents fewer certifications, never more; the MIN instant itself stays exact (Wednesday's ANSWER, 2026-11-04).
496  */
497 export const ORIGINAL_ROWS_LIMIT = VERIFY_ROWSET_LIMIT;
498 
499 /** KS-1561 G188-F1: a row-set read that returned VERIFY_ROWSET_LIMIT rows may have been cut by its LIMIT. */
500 export function rowSetMayBeCut(rows: any[] | null | undefined): boolean {
501   return Array.isArray(rows) && rows.length >= VERIFY_ROWSET_LIMIT;
502 }
503 
504 /**
505  * KS-1561 G188-F1: the fetched rows plus each original the fetch did not return, compared by row id. The fetched object is
506  * kept when the same id comes back: certificationEligibility identifies original rows by object, so one object per id
507  * keeps its MIN over the merged set equal to the full-set MIN. Added rows go last, after the certified_at DESC NULLS LAST
508  * order the read returned.
509  */
510 export function withOriginalRows(fetched: any[], originals: any[]): any[] {
511   const list = fetched || [];
512   let tied = originals || [];
513   if (tied.length > ORIGINAL_ROWS_LIMIT) {
514     logger.warn('KS-1561: more registrations tie at the earliest instant than verify reads; the rest are not counted as originals', {
515       read: ORIGINAL_ROWS_LIMIT,
516     });
517     tied = tied.slice(0, ORIGINAL_ROWS_LIMIT);
518   }
519   const have = new Set(list.map((r: any) => identityOf(r?.id)).filter((v): v is string => v !== null));
520   const missing = tied.filter((r: any) => {
521     const id = identityOf(r?.id);
522     return id !== null && !have.has(id);
523   });
524   return missing.length > 0 ? [...list, ...missing] : list;
525 }

569 // =============================================================================
570 // KS-1986 — v1 judges "the original registration" across tenants, as v2 does
571 // =============================================================================
572 // After KS-1561, v1 (/verify, verify-file) judged the original over the caller's own tenant whenever it had rows there,
573 // and v2 over every tenant. With the original in another tenant and a later certified copy in the caller's, v1 presented
574 // the latecomer's certification and v2 the original (gate 188, G188-F2, cell N-SPLIT). Kam's card
575 // secuura-ks1561-duplicate-hash-certifier-1102 = a: "Stops the later registrant outranking the original".
576 // Design (Wednesday's ANSWER 2026-11-05 19:32:34Z, D-B1 to D-B4):
577 //  D-B1 a local row presents a certification, or lends sibling evidence, only when BOTH today's rule over the local rows
578 //       AND v2's rule over the local rows plus every other tenant's originals allow it. Head is never more permissive than
579 //       base. The one difference from v2: a local row whose org or owner id equals a FOREIGN original's (one UUID in two
580 //       tenant databases) is accepted by v2 and refused here.
581 //  D-B2 local rows are presented first. When none is left, the foreign original is presented, decorated as
582 //       crossTenantLookup decorates its answer so the cross-tenant scrub applies. crossTenantLookup is NOT the fallback:
583 //       it searches the registry by document_id first, which would land a latecomer's own id on its own tenant.
584 //  D-B3 on that new path a certification carries no certifierOrganizationId (Kam's KS-590 card = a: no org UUID).
585 //  D-B4 if any cross-tenant read fails, no local row presents a certification; if every local row is certified, the
586 //       route answers as for "no rows" without crossTenantLookup and WITHOUT the chain: not found (round 2, gate 192
587 //       G192-F2: the chain presented its own certification, issuer org UUID and issuer name there). With no local row
588 //       the chain still answers, as before.
589 // Single-tenant deployments (no tenant manager: demo, local) read nothing more and present exactly as before.
590 
591 /** KS-1986: tenants consulted per hash for their originals: the same cap and the same order as v2's V2_CROSS_TENANT_LIMIT. */
592 export const V1_CROSS_TENANT_ORIGINALS_LIMIT = 8;
593 
594 /** KS-1986: every other tenant's original rows for these bytes, or ok:false when any read failed (D-B4). */
595 export type CrossTenantOriginals = { ok: true; rows: any[] } | { ok: false };
596 
597 /**
598  * KS-1986: the earliest registration(s) of these bytes in each tenant the platform registry lists (ranked as v2 ranks them),
599  * each read by ORIGINAL_ROWS_POOL_SQL over that tenant's FULL same-hash set, never a LIMIT-ed row-set fetch (G188-F1).
600  * Rows whose id is already among the local rows are dropped (the caller's own tenant is in the registry).
601  */
602 export async function readCrossTenantOriginals(hash: string, hashNorm: string, localRows: any[]): Promise<CrossTenantOriginals> {
603   const mgr = getTenantManager();
604   if (!mgr) return { ok: true, rows: [] };
605   const params = [hash, hashNorm, `sha256:${hashNorm}`];
606   try {
607     const reg = await mgr.getPlatformPool().query(
608       `SELECT tenant_id, tenant_slug FROM platform_document_registry
609        WHERE content_hash = $1 OR content_hash = $2 OR content_hash = $3
610        GROUP BY tenant_id, tenant_slug
611        ORDER BY MIN(created_at) ASC, tenant_id
612        LIMIT ${V1_CROSS_TENANT_ORIGINALS_LIMIT}`,
613       params,
614     );
615     const local = new Set((localRows || []).map((r: any) => identityOf(r?.id)).filter((v): v is string => v !== null));
616     const out: any[] = [];
617     for (const t of reg.rows) {
618       const originals = await runWithTenantId(t.tenant_id, () => queryWithTenantGuc(mgr.getPool(t.tenant_id), ORIGINAL_ROWS_POOL_SQL, params));
619       let tied: any[] = originals.rows || [];
620       if (tied.length > ORIGINAL_ROWS_LIMIT) {
621         logger.warn('KS-1986: more registrations tie at a tenant\'s earliest instant than verify reads; the rest are not counted as originals', {
622           read: ORIGINAL_ROWS_LIMIT,
623         });
624         tied = tied.slice(0, ORIGINAL_ROWS_LIMIT);
625       }
626       for (const row of tied) {
627         const id = identityOf(row?.id);
628         if (id !== null && local.has(id)) continue;
629         out.push({ ...row, _tenantId: t.tenant_id, _tenantSlug: t.tenant_slug });
630       }
631     }
632     return { ok: true, rows: out };
633   } catch (err) {
634     let reason: string;
635     try {
636       reason = String(err instanceof Error ? err.message : err);
637     } catch {
638       reason = 'unreadable thrown value';
639     }
640     logger.warn('KS-1986: cross-tenant originals read failed; no local row presents a certification', { error: reason });
641     return { ok: false };
642   }
643 }

645 /** KS-1986 D-B1: eligible only where BOTH judgments allow it. */
646 export function bothEligible(a: CertificationEligibility, b: CertificationEligibility): CertificationEligibility {
647   return {
648     certifiedRow: (row: any) => a.certifiedRow(row) && b.certifiedRow(row),
649     evidenceRow: (row: any) => a.evidenceRow(row) && b.evidenceRow(row),
650   };
651 }

653 /** KS-1986 D-B4: nothing may present a certification. */
654 const NO_ROW_ELIGIBLE: CertificationEligibility = { certifiedRow: () => false, evidenceRow: () => false };

656 /** KS-1986 D-B3: a certification presented on the new foreign-original path carries no certifier org UUID. */
657 function withoutCertifierOrganizationId(cert: Record<string, unknown> | undefined): Record<string, unknown> | undefined {
658   return cert ? { ...cert, certifierOrganizationId: null } : cert;

661 export interface V1CrossTenantJudgment {
662   /** the rows v1 presents from */
663   rows: any[];
664   /** the eligibility to select and lend sibling evidence with; undefined for a single decorated foreign original */
665   eligibility?: CertificationEligibility;
666   /** true: answer as for "no rows" WITHOUT crossTenantLookup (D-B2, D-B4) */
667   skipCrossTenantLookup: boolean;
668   /** KS-1986 G192-F2: true only when a cross-tenant read failed (D-B4): the route answers not found and does not ask the chain */
669   crossTenantReadFailed: boolean;
670 }

672 /** KS-1986: v1's same-hash local rows judged against every tenant's original (D-B1 to D-B4). Called only with local rows. */
673 export async function judgeAcrossTenants(localRows: any[], hash: string, hashNorm: string): Promise<V1CrossTenantJudgment> {
674   const foreign = await readCrossTenantOriginals(hash, hashNorm, localRows);
675   const eligibility = foreign.ok
676     ? bothEligible(certificationEligibility(localRows), certificationEligibility([...localRows, ...foreign.rows]))
677     : NO_ROW_ELIGIBLE;
678   if (localRows.some((r: any) => !isCertifiedStatus(r?.status) || eligibility.certifiedRow(r))) {
679     return { rows: localRows, eligibility, skipCrossTenantLookup: false, crossTenantReadFailed: false };
680   }
681   if (!foreign.ok) return { rows: [], skipCrossTenantLookup: true, crossTenantReadFailed: true };
682   // Every local row is a certified copy the original's holder may not present: the original is in another tenant.
683   const union = [...localRows, ...foreign.rows];
684   const earliest = union.reduce((min: number, r: any) => Math.min(min, rowCreatedMs(r)), Number.MAX_SAFE_INTEGER);
685   const originals = foreign.rows.filter((r: any) => rowCreatedMs(r) === earliest);
686   if (originals.length === 0) return { rows: [], skipCrossTenantLookup: true, crossTenantReadFailed: false };
687   const crossRule = certificationEligibility(union);
688   const selected = selectVerifiableRow(originals, crossRule);
689   const trusted = await resolveTrustedCertificationEvidence(originals, {});
690   const certMeta = (selected?.certification_metadata || {}) as Record<string, unknown>;
691   return {
692     rows: [{
693       ...selected,
694       certification_metadata: { ...certMeta, issuerId: null }, // D-B3: buildCertification reads issuerId as certifierOrganizationId
695       _siblingCertification: withoutCertifierOrganizationId(siblingCertification(originals, selected, trusted, crossRule)),
696     }],
697     skipCrossTenantLookup: false,
698     crossTenantReadFailed: false,
699   };
700 }

764 /**
765  * Look up a document across all tenants when it can't be found in the current DB.
766  * Queries the platform_document_registry, then the specific tenant DB.
767  */
768 async function crossTenantLookup(documentId: string, hashToVerify?: string): Promise<any | null> {
769   const mgr = getTenantManager();
770   if (!mgr) return null;
771 
772   try {
773     const platformPool = mgr.getPlatformPool();
774 
775     // Search registry by documentId or hash
776     let registryRows;
777     if (documentId) {
778       registryRows = await platformPool.query(
779         'SELECT * FROM platform_document_registry WHERE document_id = $1 LIMIT 1',
780         [documentId],
781       );
782     }
783     if ((!registryRows || registryRows.rows.length === 0) && hashToVerify) {
784       // KS-1561 (Wednesday's D4, 2026-11-04): the tenant that answers is the one holding the EARLIEST registration of these
785       // bytes (the registry's created_at is the first registration in that tenant; the upsert never rewrites it), consistent
786       // with D1. Before KS-1561 this was LIMIT 1 with no ORDER BY, so with two tenants holding the hash the pick was arbitrary.
787       registryRows = await platformPool.query(
788         'SELECT * FROM platform_document_registry WHERE content_hash = $1 ORDER BY created_at ASC, id LIMIT 1',
789         [hashToVerify],
790       );
791     }
792 
793     if (!registryRows || registryRows.rows.length === 0) return null;
794 
795     const reg = registryRows.rows[0];
796     const tenantPool = mgr.getPool(reg.tenant_id);
797 
798     // Query the tenant's DB for full document details — search by hash first, then ID.
799     // KS-458: documents/organizations are fail-closed flip tables; the registry
800     // row's tenant must WIN over the request context, so scope explicitly to
801     // reg.tenant_id and bundle the GUC in the same transaction as each query.
802     const searchHash = hashToVerify ? hashToVerify.replace(/^sha256:/, '') : null;
803     let docRows;
804     if (searchHash) {
805       // KS-584 (interim): fetch the same-hash SET and select the row that
806       // evidences its claims, instead of an arbitrary LIMIT 1.
807       docRows = await runWithTenantId(reg.tenant_id, () => queryWithTenantGuc(
808         tenantPool,
809         `SELECT d.id, d.external_id, d.title, d.document_type, d.content_hash, d.status,
810                 d.certification_metadata, d.metadata, d.created_at, d.certified_at,
811                 d.tenant_id, d.parent_document_id, d.owner_user_id, d.issuer_organization_id, o.name AS organization_name -- KS-1561: issuer_organization_id is read by the original-holder rule (it was
812          FROM documents d
813          LEFT JOIN organizations o ON d.issuer_organization_id = o.id
814          WHERE d.content_hash = $1 OR d.content_hash = $2 OR d.content_hash = $3
815          ORDER BY d.certified_at DESC NULLS LAST LIMIT ${VERIFY_ROWSET_LIMIT}`,
816         [hashToVerify, searchHash, `sha256:${searchHash}`],
817       ));
818       // KS-1561 G188-F1: a full read may have cut the original out; read the earliest registration(s), merge them by id.
819       if (rowSetMayBeCut(docRows.rows)) {
820         const originals = await runWithTenantId(reg.tenant_id, () => queryWithTenantGuc(
821           tenantPool, ORIGINAL_ROWS_POOL_SQL, [hashToVerify, searchHash, `sha256:${searchHash}`],
822         ));
823         docRows = { ...docRows, rows: withOriginalRows(docRows.rows, originals.rows) };
824       }
825     }
826     // KS-1119: the id lookup says which document an id names; it says nothing about a hash. It runs only when NO hash
827     // was supplied (the documentId-only strategy, which never claims integrity). When a hash was supplied and the hash
828     // query above found nothing, the answer is "not found": falling back to the id made strategy 1 present an
829     // uncompared row as a hash match, and made verify-file (documentId '') match any row with an empty external_id.
830     if ((!docRows || docRows.rows.length === 0) && !hashToVerify) {
831       docRows = await runWithTenantId(reg.tenant_id, () => queryWithTenantGuc(
832         tenantPool,
833         `SELECT d.id, d.external_id, d.title, d.document_type, d.content_hash, d.status,
834                 d.certification_metadata, d.metadata, d.created_at, d.certified_at,
835                 d.tenant_id, o.name AS organization_name
836          FROM documents d
837          LEFT JOIN organizations o ON d.issuer_organization_id = o.id
838          WHERE d.external_id = $1 LIMIT 1`,
839         [documentId],
840       ));
841     }
842 
843     if (!docRows || docRows.rows.length === 0) return null;
844 
845     const selected = selectVerifiableRow(docRows.rows);
846     // KS-1557: the rows were read from THIS tenant's pool, so the evidence lookups run there too (readForRow keys on _tenantId).
847     for (const r of docRows.rows) r._tenantId = reg.tenant_id;
848     const trusted = await resolveTrustedCertificationEvidence(docRows.rows, {});
849     return {
850       ...selected,
851       _tenantId: reg.tenant_id,
852       _tenantSlug: reg.tenant_slug,
853       // KS-584 (interim): certification evidence can live on a same-hash
854       // sibling (the certify lineage path); carry it so response builders

858   } catch (err: any) {

860     return null;
861   }

868 export function generateHash(data: Record<string, unknown> | string): string {
869   const content = typeof data === 'string' ? data : JSON.stringify(data, Object.keys(data).sort());
