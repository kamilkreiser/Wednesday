// KS-1868: `dbGetParticipant` and `dbGetParticipantByAddress` (reached through `getParticipant`) must still
// log their ERROR lines and resolve undefined when the thrown value itself cannot be read (a revoked Proxy
// throws on `instanceof`; an Error whose `message` getter throws, throws on `.message`).
// The db stand-in is a plain function, not a jest.fn(): a spy inspects what it rejects with and
// could itself fail on a revoked Proxy before the code under test saw it.
const mockHolder: { toThrow: unknown; error: jest.Mock } = { toThrow: undefined, error: jest.fn() };

jest.mock('../db', () => ({
  isDbAvailable: () => true,
  query: () => Promise.reject(mockHolder.toThrow),
}));
jest.mock('../utils/logger', () => ({
  logger: { info: jest.fn(), warn: jest.fn(), debug: jest.fn(), error: (...a: unknown[]) => mockHolder.error(...a) },
}));

const { getParticipant } = require('../services/governanceService');

const BY_ID = () => mockHolder.error.mock.calls.filter((c) => String(c[0]) === 'DB get participant failed');
const BY_ADDRESS = () => mockHolder.error.mock.calls.filter((c) => String(c[0]) === 'DB get participant by address failed');

describe('KS-1868 `getParticipant` reads the thrown value safely', () => {
  beforeEach(() => {
    mockHolder.toThrow = undefined;
    mockHolder.error = jest.fn();
  });

  it('control ks1868 C1 an ordinary Error still logs its message on both lookups and resolves undefined', async () => {
    mockHolder.toThrow = new Error('db down');
    await expect(getParticipant('p-1')).resolves.toBeUndefined();
    expect(BY_ID()).toHaveLength(1);
    expect(BY_ID()[0][1]).toEqual({ id: 'p-1', error: 'db down' });
    expect(BY_ADDRESS()).toHaveLength(1);
    expect(BY_ADDRESS()[0][1]).toEqual({ walletAddress: 'p-1', error: 'db down' });
  });

  it('RED ks1868 R1 a revoked Proxy thrown value logs a constant reason on both lookups and resolves undefined', async () => {
    const { proxy, revoke } = Proxy.revocable({}, {});
    revoke();
    mockHolder.toThrow = proxy;
    await expect(getParticipant('p-1')).resolves.toBeUndefined();
    expect(BY_ID()).toHaveLength(1);
    expect(BY_ID()[0][1]).toEqual({ id: 'p-1', error: 'unreadable thrown value' });
    expect(BY_ADDRESS()).toHaveLength(1);
    expect(BY_ADDRESS()[0][1]).toEqual({ walletAddress: 'p-1', error: 'unreadable thrown value' });
  });

  it('RED ks1868 R2 an Error whose message getter throws logs a constant reason on both lookups and resolves undefined', async () => {
    mockHolder.toThrow = Object.create(Error.prototype, {
      message: {
        get() {
          throw new Error('getter boom');
        },
      },
    });
    await expect(getParticipant('p-1')).resolves.toBeUndefined();
    expect(BY_ID()).toHaveLength(1);
    expect(BY_ID()[0][1]).toEqual({ id: 'p-1', error: 'unreadable thrown value' });
    expect(BY_ADDRESS()).toHaveLength(1);
    expect(BY_ADDRESS()[0][1]).toEqual({ walletAddress: 'p-1', error: 'unreadable thrown value' });
  });
});
