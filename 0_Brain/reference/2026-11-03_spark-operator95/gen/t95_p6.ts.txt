import { describe, it, expect, vi, afterEach } from 'vitest';

// KS-1868: IdentusClient.resolveDID (src/prism/identus-client.ts) is a resolver: a failed lookup is a result with
// `didDocument: null` and `didResolutionMetadata.error: 'notFound'`, never a throw. Its catch reads the failure into `errorMessage`
// as `err instanceof Error ? err.message : String(err)`, unguarded. When the value cannot be read (a revoked Proxy throws on
// `instanceof`; an Error whose `message` getter throws, throws on `.message`) the catch throws, so resolveDID REJECTS where an
// ordinary Error gives the notFound result. The fetch stand-in is a plain function rejecting with the cell's value (stubbed with
// `vi.stubGlobal`, removed after each cell); the logger is a plain no-op object; no network.
vi.mock('../logger', () => ({
  logger: {
    debug: () => undefined,
    info: () => undefined,
    warn: () => undefined,
    error: () => undefined,
  },
}));

import { IdentusClient } from '../prism/identus-client';

/** The resolution result, or the string 'rejected' when resolveDID rejects (an assertion can then fail on it). */
async function resolve(value: unknown): Promise<unknown> {
  vi.stubGlobal('fetch', () => Promise.reject(value));
  const client = new IdentusClient({ cloudAgentUrl: 'http://agent.invalid' });
  try {
    return await client.resolveDID('did:prism:ks1868');
  } catch {
    return 'rejected';
  }
}

const notFound = (errorMessage: string): unknown => ({
  didDocument: null,
  didResolutionMetadata: { error: 'notFound', errorMessage },
  didDocumentMetadata: {},
});

const revokedProxy = (): unknown => {
  const { proxy, revoke } = Proxy.revocable({}, {});
  revoke();
  return proxy;
};

const getterThrows = (): unknown =>
  Object.create(Error.prototype, {
    message: {
      get() {
        throw new Error('getter boom');
      },
    },
  });

describe('KS-1868 IdentusClient.resolveDID reads the thrown value safely', () => {
  afterEach(() => {
    vi.unstubAllGlobals();
  });

  it('control ks1868 C1 an ordinary Error gives the notFound result with its message', async () => {
    expect(await resolve(new Error('connection refused'))).toEqual(notFound('connection refused'));
  });

  it('control ks1868 C2 a thrown plain string gives the notFound result with that text', async () => {
    expect(await resolve('plain text failure')).toEqual(notFound('plain text failure'));
  });

  it('RED ks1868 R1 a revoked Proxy gives the notFound result with a constant reason', async () => {
    expect(await resolve(revokedProxy())).toEqual(notFound('unreadable thrown value'));
  });

  it('RED ks1868 R2 an Error whose message getter throws gives the notFound result with a constant reason', async () => {
    expect(await resolve(getterThrows())).toEqual(notFound('unreadable thrown value'));
  });
});
