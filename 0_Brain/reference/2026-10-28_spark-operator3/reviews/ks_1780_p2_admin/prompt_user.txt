92: /**
93:  * KS-730: the only place in this router that turns a caught error into a 500.
94:  *
95:  * WHAT WAS WRONG. Forty-six inline handlers returned `err.message` verbatim unless
96:  * `NODE_ENV === 'production'`, so `development`, `demo`, `test` and an UNSET NODE_ENV all answered an
97:  * admin-configuration route with raw internal text. This is the largest of the three files KS-730
98:  * enumerates, and the reason the ticket asked for a helper rather than 67 edited ternaries.
99:  *
100:  * WHY A HELPER AND NOT A DELETION. KS-727's fix was SUBTRACTIVE — its shared handlers already logged
101:  * at entry, so removing the ternary lost nothing. Measured across the three files this ticket
102:  * enumerates: **0 of 65 sites log the error before returning it.** Deleting the ternary alone would
103:  * destroy the diagnostic at all 46 sites here. The log is added in the same change.
104:  *
105:  * Each call site passes its own context naming the route, and all 46 are DISTINCT — a mislabelled
106:  * context is worse than none, because it sends a reader to the wrong handler. The contexts were
107:  * GENERATED from each site's enclosing route registration rather than hand-written, since 46 typed
108:  * strings is 46 chances to mislabel one, and a cell asserts the count and the distinctness.
109:  */
110: function fail500(res: Response, context: string, err: unknown): void {
111:   // KS-1772: building this log line, and writing it, can themselves throw (a message getter, a Proxy trap, a revoked Proxy, a logger that fails). That throw left this helper from inside
112:   // the handler's own catch, so the handler's promise rejected and the request got no response at all. Describe the value in its own try and log in another, so the constant 500 below is
113:   // always sent. The line is unchanged for every value that could be described before; one that cannot be described logs the fallback in its place. Same shape as api-gateway's KS-1524 (a).
114:   let described: string;
115:   try {
116:     described = err instanceof Error ? err.message : typeof err === 'string' ? err : typeof err === 'object' && err !== null ? 'thrown ' + (Object.getPrototypeOf(err)?.constructor?.name ?? 'object') + ' with fields [' + Object.keys(err).join(', ') + ']' : 'thrown ' + typeof err;
117:   } catch {
118:     described = 'thrown value that could not be described';
119:   }
120:   try {
121:     logger.error(context, { error: described });
122:   } catch {
123:     // Nothing more can be said if the logger itself fails; the caller still gets the 500 below.
124:   }
125:   res.status(500).json({ success: false, error: { code: 'INTERNAL_ERROR', message: 'Internal server error' } });
126: }
127: 
128: /**
129:  * KS-1780: a catch that tests `err?.message?.includes(...)` or `err?.code` BEFORE it reaches fail500 reads the thrown value itself. A revoked Proxy, or an object whose `message` getter throws,
130:  * makes THAT read throw from inside the catch, so the handler's promise rejects and the request gets no response at all (the KS-1772 shape, one step earlier). These two readers answer false
131:  * when the value cannot be inspected: such a value is not a "table missing" or "unique constraint" error, so it falls through to fail500 and gets the constant 500.
132:  */
133: function errorMessageIncludes(err: any, needle: string): boolean {
134:   try {
135:     return err?.message?.includes(needle) === true;
136:   } catch {
137:     return false;
138:   }
139: }

965: adminConfigRouter.post('/api-keys', async (req: Request, res: Response) => {
966:   try {
967:     const d = req.body;
968:     // KS-1206: the security service bounds this field to an integer from 1 to 10000; refuse anything else before the INSERT.
969:     if (d.rateLimit !== undefined && !(Number.isInteger(d.rateLimit) && d.rateLimit >= 1 && d.rateLimit <= 10000)) {
970:       return res.status(400).json({ success: false, error: { code: 'BAD_REQUEST', message: 'rateLimit must be an integer from 1 to 10000' } });
971:     }
972:     if (!d.name) {
973:       return res.status(400).json({ success: false, error: { code: 'BAD_REQUEST', message: 'name is required' } });
974:     }
975:     // KS-1594: this handler INSERTs into svc_api_keys with a raw $queryRaw below, so NEITHER
976:     // `decideMint` NOR `decideTenantAccess` runs on it. That made it a SECOND hop 1: an ORG_ADMIN
977:     // could mint a provisioning-capable key here carrying no, a whitespace, or a FOREIGN
978:     // organisation, and hop 2 then minted into that organisation with its connector (pull request
979:     // 1541 Opus gate, finding M1, measured 201 three of three). KS-1584 shut the security route's
980:     // hop 1; it does not reach this one.
981:     //
982:     // All three decisions below go through the SAME shared predicates the security route uses, so
983:     // each rule has one definition rather than a second copy that can drift apart from it.
984:     // (ks1594CommentOnlyMarker: named here in a comment ONLY, so the KS-1594 cell's
985:     // comment-stripper has a control — a predicate named in a comment must not satisfy its
986:     // source assertion.)
987:     const requestedScopes = Array.isArray(d.permissions) && d.permissions.length > 0
988:       ? d.permissions as string[]
989:       : ['read'];
990:     const callerIsPlatform = isPlatformRole((req as any).user?.role as string | undefined);
991:     // (m2, pull request 1547 gate) A non-string `organizationId` must be refused BEFORE
992:     // normalisation. `normaliseOrgId` returns null for ANY non-string, so an array, object, number
993:     // or boolean was read as "absent" — and for a platform caller that silently turned a malformed
994:     // request into an ORG-LESS, and therefore UNBOUNDED, provisioning key. Refuse the shape.
995:     if (d.organizationId !== undefined && d.organizationId !== null && typeof d.organizationId !== 'string') {
996:       return res.status(400).json({
997:         success: false,
998:         error: { code: 'BAD_REQUEST', message: 'organizationId must be a string' },
999:       });
1000:     }
1001:     // (b) Normalise BEFORE comparing or storing, so '' and '   ' are absent rather than a distinct
1002:     // organisation, and the comparison in (c) folds case and surrounding whitespace.
1003:     const requestedOrg = normaliseOrgId(d.organizationId as string | null | undefined);
1004:     const callerOrg = normaliseOrgId((req as any).user?.organizationId as string | null | undefined);
1005:     // (a) A non-platform caller may NOT mint an UNQUALIFIED provisioning-capable key — one whose
1006:     // body names no organisation — because an org-less provisioning key is the unbounded,
1007:     // platform-minted shape (`keyRevokePolicy.ts` treats a null organisation as unbounded), and
1008:     // there would be nothing to bound it to.
1009:     //
1010:     // It MAY mint one that names an organisation, which (c) then bounds to its own. That is Kam's
1011:     // ruling on card `secuura-provisioning-scope-escapes-the-org-bound`, 2026-10-13 12:55:49 AEDT,
1012:     // option b: "Allow it, but bound the provisioning key to the minting admin's own scope". The
1013:     // rejected option a was "only platform roles may mint a key with the provisioning scope", and
1014:     // the first version of this guard implemented that rejected option (pull request 1547 gate, m1).
1015:     //
1016:     // The test is on `requestedOrg` — the BODY value — with NO fall-back to the caller's claim, so
1017:     // this door matches the sibling door exactly: `services/security/src/index.ts` refuses when
1018:     // `normaliseOrgId(data.organizationId) === null`, and its comment states the same default-deny
1019:     // reasoning. Testing the post-fall-back value instead would make THIS door more permissive than
1020:     // that one, on the single key class whose unboundedness is the whole concern.
1021:     if (isProvisioningCapable(requestedScopes) && !callerIsPlatform && requestedOrg === null) {
1022:       return res.status(403).json({
1023:         success: false,
1024:         error: {
1025:           code: 'FORBIDDEN',
1026:           message: 'A provisioning-capable key must name an organisation in your scope; only a platform role may mint one without an organisation',
1027:         },
1028:       });
1029:     }
1030:     // (c) A non-platform caller is bound to its own organisation — the rule the security route
1031:     // already applies to an ORG_ADMIN by token claim (the KS-1533 arm). A blank or whitespace
1032:     // value is not a foreign organisation: it normalises to absent and falls through to the
1033:     // caller's own claim below, which leaves the key BOUNDED rather than unbounded.
1034:     if (!callerIsPlatform && requestedOrg !== null && requestedOrg !== callerOrg) {
1035:       return res.status(403).json({
1036:         success: false,
1037:         error: { code: 'FORBIDDEN', message: 'organizationId must be your own organisation' },
1038:       });
