===== TICKET =====
KS-1240 — T-2: the connector-token / key-validate fetch has no timeout — a slow upstream hangs the request [Backlog]

**The sharpest of the four candidates the gate surfaced, and the only one that is request-reachable today.**

The api-gateway's connector-token exchange / key-validate `fetch` carries **no timeout**. A slow or non-answering upstream turns into a gateway request with no bound on how long it hangs, while the process carries on serving everything else.

## Measured on BOTH trees (head and develop) — so it is pre-existing

* an exchange that **never answers** → the request hangs;
* an exchange whose body arrives **2.5 s late** → the request hangs for that long.

Graded **Major**. Request-reachable, unlike T-1.

## What to do

Give the fetch an explicit timeout (AbortSignal / `AbortController`) and decide the refusal shape — a 503 on timeout is consistent with how the connector branch already answers a refused exchange.

+/** Every `fetch(` call in the source, each as the text from `fetch(` to its balancing `)`. */
+function fetchCalls(src: string): string[] {
+  const calls: string[] = [];
+  const re = /\bfetch\(/g;
+  let match: RegExpExecArray | null;
+  while ((match = re.exec(src)) !== null) {
+    const open = match.index + match[0].length - 1;
+    let depth = 0;
+    let end = -1;
+    for (let i = open; i < src.length; i++) {
+      const ch = src[i];
+      if (ch === '(') depth++;
+      else if (ch === ')') {
+        depth--;
+        if (depth === 0) {
+          end = i;
+          break;
+        }
+      }
+    }
+    expect(end, 'unbalanced fetch call at offset ' + match.index).toBeGreaterThan(-1);
+    calls.push(src.slice(match.index, end + 1));
+  }
+  return calls;
+}
+
+function isBounded(call: string): boolean {
+  return call.includes('AbortSignal.timeout(');
+}
+
+/** The first line of each unbounded call, so a failure names the call. */
+function unbounded(src: string): string[] {
+  return fetchCalls(src)
+    .filter((call) => !isBounded(call))
+    .map((call) => call.split('\n')[0].trim());
+}

+function constantMs(src: string, name: string): number {
+  const found = new RegExp('const ' + name + ' = ([0-9_]+);').exec(src);
+  return found ? Number(found[1].replace(/_/g, '')) : Number.NaN;
+}

311: /**
312:  * KS-1187: which gdpr sub-path the erasure door must judge, decided on the
313:  * CANONICAL path rather than the raw spelling. `subUrl` is `req.url` inside the
314:  * `/api/gdpr` mount, where express keeps an absolute-form target's
315:  * `scheme://authority` in front of the path. Canonical means: the path express
316:  * routes on (no query, fragment or authority), each segment's `;params`
317:  * dropped, percent-decoded, `.`/`..` resolved and empty segments removed. The
318:  * first segment is compared the way the door's router compares (case-folded
319:  * unless it is case sensitive).
320:  *
321:  * - `door`: the canonical path is `/erasures` or below it. The door judges it.
322:  * - `not-door`: the canonical path plainly is not the door. Forwarded as before,
323:  *   even when a later segment is malformed.
324:  * - `undetermined`: the first segment cannot be decoded, or `..` climbs out of the
325:  *   mount, so nothing can say whether it names the door. Refused fail-closed.
326:  *   Also any `.` or `..` segment once the path so far names the door
327:  *   (`/erasures/..`, `/erasures/%2e%2e`, `/erasures/..;x`, `/erasures/%2e%2e%2fabc`):
328:  *   resolving it would judge a path that express does not route. The upstream
329:  *   routes the RAW path, where `/erasures/..` is `/erasures/:externalRef` with the
330:  *   reference `..`, so popping the door segment turned an erasure read into
331:  *   "not the door" (KS-1187 round 2, F-1019-1).
332:  *
333:  * The door and its tests both use this function; there is no second copy.
334:  */
335: export function erasureDoorVerdict(
336:   subUrl: string,
337:   caseSensitive: boolean,
338: ): { verdict: 'door' | 'not-door' | 'undetermined'; canonicalPath: string | null } {
339:   let path = subUrl;
340:   const tailStart = path.search(/[?#]/);
341:   if (tailStart !== -1) path = path.slice(0, tailStart);
342:   const scheme = /^[A-Za-z][A-Za-z0-9+\-.]*:\/\//.exec(path);
343:   if (scheme) {
344:     const pathStart = path.indexOf('/', scheme[0].length);
345:     path = pathStart === -1 ? '/' : path.slice(pathStart);
346:   }
347:   const namesDoor = (segment: string | null | undefined): boolean =>
348:     typeof segment === 'string' && (caseSensitive ? segment : segment.toLowerCase()) === 'erasures';
349:   const segments: Array<string | null> = [];
350:   for (const raw of path.split('/')) {
351:     let decoded: string | null;
352:     try {
353:       decoded = decodeURIComponent(raw.split(';')[0]);
354:     } catch {
355:       decoded = null;
356:     }
357:     for (const segment of decoded === null ? [null] : decoded.split('/')) {
358:       if (segment === '') continue;
359:       if (segment === '.' || segment === '..') {
360:         if (namesDoor(segments[0])) return { verdict: 'undetermined', canonicalPath: null };
361:         if (segment === '.') continue;
362:         if (segments.length === 0) return { verdict: 'undetermined', canonicalPath: null };
363:         segments.pop();
364:         continue;
365:       }
366:       segments.push(segment);
367:     }
368:   }
369:   const first = segments[0];
370:   if (first === undefined) return { verdict: 'not-door', canonicalPath: '/' };
371:   if (first === null) return { verdict: 'undetermined', canonicalPath: null };
372:   if (!namesDoor(first)) {
373:     return { verdict: 'not-door', canonicalPath: null };
374:   }
375:   return { verdict: 'door', canonicalPath: '/' + segments.map((x) => x ?? '').join('/') };
376: }

790:    * silently forwards a normalised copy of what it was sent hides what it

12: /** Every `fetch(` call in the source, each as the text from `fetch(` to its balancing `)`. */
13: function fetchCalls(src: string): string[] {
14:   const calls: string[] = [];
15:   const re = /\bfetch\(/g;
16:   let match: RegExpExecArray | null;
17:   while ((match = re.exec(src)) !== null) {
18:     const open = match.index + match[0].length - 1;
19:     let depth = 0;
20:     let end = -1;
21:     for (let i = open; i < src.length; i++) {
22:       const ch = src[i];
23:       if (ch === '(') depth++;
24:       else if (ch === ')') {
25:         depth--;
26:         if (depth === 0) {
27:           end = i;
28:           break;
29:         }
30:       }
31:     }
32:     expect(end, 'unbalanced fetch call at offset ' + match.index).toBeGreaterThan(-1);
33:     calls.push(src.slice(match.index, end + 1));

35:   return calls;
36: }
