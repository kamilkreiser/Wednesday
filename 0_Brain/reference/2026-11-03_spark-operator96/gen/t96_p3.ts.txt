import { describe, it, expect, vi, beforeEach } from 'vitest';

// KS-1868: RedisCSRFStore.generate and RedisCSRFStore.validate (src/security/redis-security-store.ts) catch a
// Redis failure and log it as `{ error: String(error) }`; generate then rethrows the original value, validate answers false. The `String(error)` read is unguarded. When the value cannot be read (a revoked
// Proxy throws on String(); an Error whose `message` getter throws, throws in Error.prototype.toString) the catch itself
// throws, so the call rejects with a TypeError where an ordinary Error gives the original value (generate) or false (validate). The Redis client is a plain object of plain
// functions rejecting with the cell's value (not vi.fn(): a spy inspects what it rejects with); the logger is a plain
// recorder; no network.
const h = vi.hoisted(() => ({
  toThrow: undefined as unknown,
  errors: [] as unknown[][],
}));

vi.mock('../utils/logger', () => ({
  logger: {
    debug: () => undefined,
    info: () => undefined,
    warn: () => undefined,
    error: (...args: unknown[]) => {
      h.errors.push(args);
    },
  },
}));

vi.mock('../cache/redis-client', () => ({
  CACHE_TTL: { SESSION: 300 },
  redisClient: {
    get: () => Promise.reject(h.toThrow),
    set: () => Promise.reject(h.toThrow),
    delete: () => Promise.resolve(true),
    exists: () => Promise.resolve(false),
  },
}));

import { RedisCSRFStore } from '../security/redis-security-store';

type Settled = { threw: boolean; value: unknown };

/** The call's result, or threw:true when it rejects (an assertion can then fail on it). */
async function settle(call: () => Promise<unknown>): Promise<Settled> {
  try {
    return { threw: false, value: await call() };
  } catch {
    return { threw: true, value: undefined };
  }
}

const logged = (needle: string): unknown[][] => h.errors.filter((a) => String(a[0]).includes(needle));

const revokedProxy = (): unknown => {
  const { proxy, revoke } = Proxy.revocable({}, {});
  revoke();
  return proxy;
};

const getterThrows = (): unknown =>
  Object.create(Error.prototype, {
    message: {
      get() {
        throw new Error('getter boom');
      },
    },
  });

// Boxed on purpose: a promise that resolves with a revoked Proxy throws when it reads `.then` on it.
const genAnswer = async (): Promise<{ caught: unknown }> => {
  try {
    await new RedisCSRFStore().generate('s');
    return { caught: 'resolved' };
  } catch (e) {
    return { caught: e };
  }
};
const valAnswer = async (): Promise<Settled> => settle(() => new RedisCSRFStore().validate('s', 't'));

describe('KS-1868 RedisCSRFStore reads a Redis failure safely', () => {
  beforeEach(() => {
    h.toThrow = undefined;
    h.errors = [];
  });

  it('control ks1868 C1 generate: an ordinary Error is rethrown as is and its text logged', async () => {
    const err = new Error('redis down');
    h.toThrow = err;
    expect((await genAnswer()).caught === err).toBe(true);
    expect(logged('error generating token')[0][1]).toEqual({ error: 'Error: redis down' });
  });

  it('control ks1868 C2 validate: a thrown plain string gives false and logs that text', async () => {
    h.toThrow = 'plain text failure';
    expect(await valAnswer()).toEqual({ threw: false, value: false });
    expect(logged('error validating token')[0][1]).toEqual({ error: 'plain text failure' });
  });

  it('RED ks1868 R1 generate: a revoked Proxy is rethrown as is and a constant reason logged', async () => {
    const v = revokedProxy();
    h.toThrow = v;
    expect((await genAnswer()).caught === v).toBe(true);
    expect(logged('error generating token')[0][1]).toEqual({ error: 'unreadable thrown value' });
  });

  it('RED ks1868 R2 generate: an Error whose message getter throws is rethrown as is and a constant reason logged', async () => {
    const v = getterThrows();
    h.toThrow = v;
    expect((await genAnswer()).caught === v).toBe(true);
    expect(logged('error generating token')[0][1]).toEqual({ error: 'unreadable thrown value' });
  });

  it('RED ks1868 R3 validate: a revoked Proxy gives false and logs a constant reason', async () => {
    h.toThrow = revokedProxy();
    expect(await valAnswer()).toEqual({ threw: false, value: false });
    expect(logged('error validating token')[0][1]).toEqual({ error: 'unreadable thrown value' });
  });

  it('RED ks1868 R4 validate: an Error whose message getter throws gives false and logs a constant reason', async () => {
    h.toThrow = getterThrows();
    expect(await valAnswer()).toEqual({ threw: false, value: false });
    expect(logged('error validating token')[0][1]).toEqual({ error: 'unreadable thrown value' });
  });
});
