+    /// Email addresses (the owner's ruling: email addresses are blanked in previews). The whole
+    /// address (local part, `@` and domain) is one span, so it becomes one [`super::MASK`].
+    ///
+    /// **What it matches**, at each `@` of the text (the text [`super::PatternRedactor`] matches
+    /// on, after [`super::strip_format_chars`]):
+    /// - the local part: the longest run of ASCII `[A-Za-z0-9._%+-]` ending just before the `@`
+    ///   (plus-tags and dotted local parts included), leading `.` trimmed, at least 1 byte. The run
+    ///   stops at any other byte: `@`, `:`, `/`, `<`, `"`, `'`, whitespace, and every non-ASCII byte;
+    /// - the domain: the longest run of ASCII `[A-Za-z0-9.-]` starting just after the `@`, trailing
+    ///   `.` and `-` trimmed, holding at least two labels separated by `.`, no empty label, and a last
+    ///   label of at least 2 ASCII letters only, or a last label starting `xn--` (punycode; compared
+    ///   case-insensitively).
+    ///
+    /// So an address is found inside angle brackets (`<a@b.example>`), quotes and JSON strings,
+    /// after `mailto:` (the `:` is not a local-part byte: `mailto:` is kept), in a `Signed-off-by:`
+    /// trailer, and before `.`, `,`, `)` or `: `.
+    ///
+    /// **Not an address** (not masked by this rule; other rules may still mask part of it):
+    /// - an `@` inside a `scheme://` token, from the `://` to the end of the token as
+    ///   [`url_userinfo`] delimits it (whitespace, `"`, `'`, `<`, `>`, `` ` ``, a non-ASCII byte):
+    ///   that token is the URL rule's, so `https://user@host.example/` (no password) is masked by
+    ///   neither rule;
+    /// - an scp or ssh target: a domain followed immediately by `:` and then a byte that is not
+    ///   whitespace (`user@host.example.com:/path`, `git@host.example:org/repo`); such a target keeps
+    ///   its user name. A `:` followed by whitespace or the end of the text does not stop the match;
+    /// - `@mentions` and decorators (`@name`, `@app.route`): no local part;
+    /// - version pins (`pkg@1.2.3`): the last label is not letters;
+    /// - `a@b`, `user@localhost`: no dot in the domain.
+    ///
+    /// **Not caught:** quoted local parts (`"john doe"@example.com`); any address with a non-ASCII
+    /// byte in it (the domain run stops at the first one, so `alice@bücher.example` keeps
+    /// `alice@b`); a percent-encoded `@` (`%40`), HTML entities (`&#64;`), spelled-out forms
+    /// (`alice at example dot com`), an address split across lines or by spaces; a single-label or
+    /// numeric domain (`a@localhost`, `a@[192.0.2.1]`); an address cut by the end of
+    /// [`super::REDACT_WINDOW_BYTES`]; an address directly followed by more domain bytes that make
+    /// the last label something other than letters (`alice@example.com9f86…`,
+    /// `alice@example.com--flag`: the run is not a domain); addresses inside a `scheme://` token
+    /// or an scp target (above).

+    /// `name@file.rs` or a Python matrix product with an attribute `x@self.weight`.

+    /// domain byte of the `@` before it and once as a local byte of the `@` after it; the domain
+    /// check reads the domain run once; the `scheme://` tokens are found in one pass before the scan
+    /// and walked with one forward index.
+    pub fn emails(b: &[u8], r: &mut Vec<(usize, usize)>) {
+        let local =
+            |c: u8| c.is_ascii_alphanumeric() || matches!(c, b'.' | b'_' | b'%' | b'+' | b'-');
+        let domain = |c: u8| c.is_ascii_alphanumeric() || matches!(c, b'.' | b'-');

+            while let Some(p) = find_from(b, b"://", i) {
+                let start = p + 3;
+                let mut end = start;
+                while end < b.len()

+                    && !matches!(b[end], b'"' | b'\'' | b'<' | b'>' | b'`')
+                    && b[end] < 0x80
+                {
+                    end += 1;
+                }
+                urls.push((start, end));
+                i = end.max(start);
+            }
+        }
+        let mut u = 0;
+        for at in 0..b.len() {

+                continue;
+            }

+                u += 1;
+            }

+            }

+            }

+                r.push((s, at));
+            } else {

+        }

+/// What of `addr` survives in `p`: `Some("address")` when a piece holding its `@` with a byte on
+/// each side does (or the whole address, if it is shorter than 8 bytes); `Some("domain")` when its
+/// `@` and domain do; `Some("part of the address")` when any other 8-byte window does; else `None`.
+fn survives(p: &str, addr: &str) -> Option<&'static str> {
+    let (pb, ab) = (p.as_bytes(), addr.as_bytes());
+    let at = ab
+        .iter()
+        .position(|&c| c == b'@')
+        .expect("an address has an @");
+    for s in 0..at {
+        for e in at + 2..=ab.len() {
+            if contains(pb, &ab[s..e]) {
+                return Some("address");
+            }
+        }
+    }
+    if ab.len() < 8 && contains(pb, ab) {
+        return Some("address");
+    }
+    if contains(pb, &ab[at..]) {
+        return Some("domain");
+    }
+    if windows8(ab).iter().any(|w| contains(pb, w)) {
+        return Some("part of the address");
+    }
+    None
+}

+        "[A-Za-z0-9._%+-]{1,300}",
+        prop::collection::vec("[A-Za-z0-9-]{1,20}", 1..=5),
+        prop_oneof!["[A-Za-z]{2,10}", "xn--[a-z0-9]{1,10}"],
+        0..POSITIONS.len(),
+        200usize..=260,
+        0..=NEIGHBOURS.len(),
+        0..JOINS.len(),
+        any::<bool>(),
+        0..DECOR.len(),
+        any::<usize>(),
+    )
+        .prop_map(
+            |(local, labels, tld, position, filler, neighbour, join, after, decor, cut)| Case {
+                // a local part of dots only is not an address (leading dots are trimmed)
+                local: if local.bytes().all(|c| c == b'.') {
+                    format!("a{local}")
+                } else {
+                    local
+                },
+                labels,
+                tld,
+                position,
+                filler,
+                neighbour,
+                join,
+                after,
+                decor,
+                cut,
+            },
+        )

+/// The join actually used: the generator does not produce the shapes the rule does not catch by
+/// design (item 1's "not an address" and "not caught" lists), and falls back to a space for them:
+/// - before the address, a `scheme://` token joined by nothing, `:` or `=` (the address would be
+///   inside the URL token);
+/// - after the address, `:` (an scp target);
+/// - after the address, nothing, when the neighbour starts with domain bytes that are not letters
+///   only, or with letters then `:` and a non-space (the run is not a domain, or an scp target).
+fn effective_join(n: &str, join: usize, after: bool) -> usize {
+    let j = JOINS[join];
+    let ok = if !after {
+        !(n.contains("://") && j != " ")
+    } else if j == ":" {
+        false
+    } else if j.is_empty() {
+        let b = n.as_bytes();
+        let run = b
+            .iter()
+            .take_while(|c| c.is_ascii_alphanumeric() || **c == b'.' || **c == b'-')
+            .count();
+        let letters = b[..run].iter().all(u8::is_ascii_alphabetic);
+        let scp = run + 1 < b.len() && b[run] == b':' && !b[run + 1].is_ascii_whitespace();
+        letters && !scp
+    } else {
+        true
+    };
+    if ok {
+        join
+    } else {
+        0
+    }
+}

+/// (raw text, the clean address, reach keys)
+fn build(c: &Case) -> (String, String, Vec<String>) {
+    let mut reach = Vec::new();
+    let domain = format!("{}.{}", c.labels.join("."), c.tld);
+    let addr = format!("{}@{}", c.local, domain);
+    reach.push(format!("labels {}", c.labels.len() + 1));
+    reach.push(format!(
+        "tld {}",
+        if c.tld.starts_with("xn--") {
+            "xn--"
+        } else {
+            "letters"
+        }
+    ));
+    reach.push(format!(
+        "local {}",
+        match c.local.len() {
+            1 => "1",
+            2..=8 => "2-8",
+            9..=64 => "9-64",
+            _ => "65-300",
+        }
+    ));
+    // decoration
+    let fmt = |ch: char| {
+        let k = c.cut % (addr.len() + 1);
+        format!("{ch}{}{ch}{}{ch}", &addr[..k], &addr[k..])
+    };
+    let decorated = match c.decor {
+        1 => format!("é{addr}é"),
+        2 => format!("€{addr}€"),
+        3 => format!("😀{addr}😀"),
+        4 => fmt('\u{200B}'),
+        5 => fmt('\u{00AD}'),
+        6 => fmt('\u{202E}'),
+        7 => format!("\"{addr}\""),
+        8 => format!("'{addr}'"),
+        9 => format!("<{addr}>"),
+        10 => format!("mailto:{addr}"),
+        _ => addr.clone(),
+    };
+    reach.push(format!("decor {}", DECOR[c.decor]));
+    // neighbour
+    let line = if c.neighbour < NEIGHBOURS.len() {
+        let (nname, n) = NEIGHBOURS[c.neighbour];
+        let j = effective_join(n, c.join, c.after);
+        let side = if c.after { "after" } else { "before" };
+        reach.push(format!("neighbour {nname} {side} join {:?}", JOINS[j]));
+        if c.after {
+            format!("{decorated}{}{n}", JOINS[j])
+        } else {
+            format!("{n}{}{decorated}", JOINS[j])
+        }
+    } else {
+        reach.push("neighbour none".into());
+        decorated
+    };
+    // position
+    let raw = match c.position {
+        0 => format!("{line} tail words"),
+        1 => format!("head words {line}"),
+        2 => format!("some words {line} more words"),
+        3..=6 => {
+            let before = "a line of text\n".repeat(c.position - 3);
+            format!("{before}{line}\nthe next line")
+        }
+        7 => {
+            let filler: String = "word ".repeat(60).chars().take(c.filler).collect();
+            format!("{filler}{line}")
+        }
+        _ => format!("{} {line}", "a1".repeat(c.filler)),
+    };
+    reach.push(format!("position {}", POSITIONS[c.position]));
+    (raw, addr, reach)
+}
+
+/// The reach keys every run must hit at least once.
+fn expected_reach() -> Vec<String> {
+    let mut v: Vec<String> = Vec::new();
+    v.extend((2..=6).map(|n| format!("labels {n}")));
+    v.extend(["tld xn--", "tld letters"].map(String::from));
+    v.extend(["1", "2-8", "9-64", "65-300"].map(|s| format!("local {s}")));
+    v.extend(DECOR.iter().map(|d| format!("decor {d}")));
+    v.extend(POSITIONS.iter().map(|p| format!("position {p}")));
+    v.push("neighbour none".into());
+    for (nname, n) in NEIGHBOURS {
+        for after in [false, true] {
+            for (j, join) in JOINS.iter().enumerate() {
+                if effective_join(n, j, after) == j {
+                    let side = if after { "after" } else { "before" };
+                    v.push(format!("neighbour {nname} {side} join {join:?}"));
+                }
+            }
+        }
+    }
+    v
+}
