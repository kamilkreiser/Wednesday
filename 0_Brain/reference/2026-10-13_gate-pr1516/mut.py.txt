def run(cmd, env=None, timeout=600):
    p = subprocess.run(cmd, cwd=O, env=env, capture_output=True, text=True, timeout=timeout)
    return p.returncode, p.stdout + p.stderr

def jest_summary(out):
    out = re.sub(r"\x1b\[[0-9;]*m", "", out)
    t = [l.strip() for l in out.splitlines() if l.startswith("Tests:")]
    fails = [re.sub(r" \(\d+ ms\)", "", l.strip()) for l in out.splitlines() if l.strip().startswith("✕")]
    return (t[0] if t else "NO Tests: line"), fails
