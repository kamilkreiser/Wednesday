function token(c: Record<string, unknown>) {
  const h = b64(Buffer.from(JSON.stringify({ alg: 'RS256', typ: 'JWT' }))); const n = Math.floor(Date.now() / 1000);
  const p = b64(Buffer.from(JSON.stringify({ sub: 'gm', iat: n, exp: n + 600, ...c })));
  return `${h}.${p}.${b64(crypto.sign('RSA-SHA256', Buffer.from(`${h}.${p}`), privateKey.export({ type: 'pkcs8', format: 'pem' }).toString()))}`;
}
