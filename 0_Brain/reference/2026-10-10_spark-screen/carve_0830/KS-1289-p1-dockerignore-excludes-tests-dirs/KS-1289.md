# =============================================================================
# TESTS for Blockchain/Dev/.dockerignore - KS-1289: it must exclude __tests__

set -uo pipefail

IGNORE="${DOCKERIGNORE_FILE:-$HERE/../../.dockerignore}"

PASS=0

# Line number of the first `**/__tests__` pattern in file $1 (empty when there is none).
pattern_line() { grep -nE '^\*\*/__tests__/?$' "$1" | head -n1 | cut -d: -f1; }
# Line number of the last `!` negation in file $1 (empty when there is none).
last_negation_line() { grep -nE '^!' "$1" | tail -n1 | cut -d: -f1; }
# 0 when the pattern exists and sits after every negation.
excludes_tests_dirs() {
  local p n
  p="$(pattern_line "$1")"; n="$(last_negation_line "$1")"
  [[ -n "$p" ]] || return 1
  [[ -z "$n" || "$p" -gt "$n" ]]
}

# CELL 1 (RED at the tip) - a line excludes every __tests__ directory at any depth.

fi

# CELL 2 (RED at the tip) - that line comes after the last negation, because the last match wins.
if excludes_tests_dirs "$IGNORE"; then

printf '!connectors/x\n**/__tests__\n' > "$WORK/pattern_after_negation.ignore"
if ! excludes_tests_dirs "$WORK/only_tests.ignore" \
   && ! excludes_tests_dirs "$WORK/pattern_before_negation.ignore" \
