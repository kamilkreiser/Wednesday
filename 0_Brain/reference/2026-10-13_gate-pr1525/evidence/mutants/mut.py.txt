def suite(name, cmd, tag):
    log=f'{L}/{tag}_{name}.log'
    p=subprocess.run(cmd, cwd=O, env=env, stdout=open(log,'w'), stderr=subprocess.STDOUT)
    txt=open(log).read()
    m=re.search(r'Tests:\s+(.*)', txt)
    compile_err = ('error TS' in txt) or ('Test suite failed to run' in txt)
    return {'rc':p.returncode, 'tests': m.group(1) if m else None, 'compile_or_load_error': compile_err, 'log': log}
