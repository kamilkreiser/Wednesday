  const issue = async (principal: object, body: Record<string, unknown>) => {
    mockSaveCertification.mockClear();

        newerVersions: (j?.newerVersions ?? j?.lineage?.newerVersions ?? j?.document?.newerVersions)?.length, keys: j ? Object.keys(j).slice(0, 40) : null, raw: JSON.stringify(j).slice(0, 1500) };

    const vBefore = await verify(P.userB, { contentHash: hashB });
    const vBeforeId = await verify(P.userB, { documentId: bExt });
    // A attaches a derived document under B's document, carrying B's content hash and a chosen issuer name
    const r = await issue(P.userA, { parentDocumentId: bExt, data: { title: 'A derived ' + RUN, contentHash: hashB, issuedBy: 'Org B Official Registrar' } });
    const derived = (await sql.query(`SELECT external_id, owner_user_id::text AS owner, status, content_hash, parent_document_id::text AS parent,
       metadata->>'certificationId' AS cert_id FROM documents WHERE parent_document_id = (SELECT id FROM documents WHERE external_id=$1)`, [bExt])).rows;
    const after = strip(await row(bExt));
    const vAfter = await verify(P.userB, { contentHash: hashB });
    const vAfterId = await verify(P.userB, { documentId: bExt });
    const vAnon = await verify({ tenantId: T }, { contentHash: hashB });
    rec({ cell: 'ITEM5 lineage under B by userA', status: r.status, saved: r.saved, body: r.text.slice(0, 300), bRowBefore: before, bRowAfter: after,
      derived, bOwner: before?.owner, vBefore, vBeforeId, vAfter, vAfterId, vAnon });
    // same attack with documentId = B's doc AND parentDocumentId = A's own doc (the legacy branch is skipped)
    const a = await create(P.userA);
    const d = await create(P.userB);
    const b4 = strip(await row(d.id));
    const r2 = await issue(P.userA, { documentId: d.id, parentDocumentId: a.id });
    rec({ cell: 'ITEM5b documentId=B doc + parentDocumentId=A own', status: r2.status, saved: r2.saved, bRowBefore: b4, bRowAfter: strip(await row(d.id)) });
    // class check: the verify claim does not need B's document as the parent - A's OWN document as parent, B's bytes' hash in data
    const hashC = newHash();
    const resC = await fetch(base + '/api/documents', { method: 'POST', headers: { 'content-type': 'application/json', 'x-probe-user': JSON.stringify(P.userB) },
      body: JSON.stringify({ type: 'DOCUMENT', contentHash: hashC, data: { title: 'B second ' + RUN } }) });
    await resC.text(); await new Promise((r) => setTimeout(r, 300));
    const vC0 = await verify({ tenantId: T }, { contentHash: hashC });
    const a2 = await create(P.userA);
    const r3 = await issue(P.userA, { parentDocumentId: a2.id, data: { title: 'A derived 2 ' + RUN, contentHash: hashC, issuedBy: 'Org B Official Registrar' } });
    const vC1 = await verify({ tenantId: T }, { contentHash: hashC });
    rec({ cell: 'ITEM5c parent = A OWN doc, data.contentHash = B bytes', status: r3.status, vBeforeCert: vC0.certification ?? null, vAfterCert: vC1.certification ?? null,
      vBeforeVerified: vC0.verified, vAfterVerified: vC1.verified });
  }, 90000);

  it('ITEM3 timing: unknown vs not-yours (recording)', async () => {
    if (!TIMING_N) { rec({ cell: 'ITEM3', skipped: 'PROBE_TIMING_N=0' }); return; }
    const targets = { notYours: (await create(P.userB)).id as string };
    const series: Record<string, number[]> = { userA_unknownExt: [], userA_notYours: [], userA_unknownExt_ctl: [], connA_unknownExt: [], connA_notYours: [], userA_unknownUuid: [] };
    const time = async (who: object, documentId: string) => {
      const t0 = process.hrtime.bigint();
      const res = await fetch(base + '/api/certifications/issue', { method: 'POST', headers: { 'content-type': 'application/json', 'x-probe-user': JSON.stringify(who) },
        body: JSON.stringify({ type: 'verification_certificate', data: { title: 't' }, documentId }) });
      await res.text();
      if (res.status !== 403) throw new Error('timing cell expected a refusal, got ' + res.status);
      return Number(process.hrtime.bigint() - t0) / 1e6;
    };
    for (let i = 0; i < 20; i++) { await time(P.userA, 'doc-warm'); }
    for (let i = 0; i < TIMING_N; i++) { // interleaved so drift hits every series alike
      series.userA_unknownExt.push(await time(P.userA, 'doc-' + randomUUID()));
      series.userA_notYours.push(await time(P.userA, targets.notYours));
      series.userA_unknownExt_ctl.push(await time(P.userA, 'doc-' + randomUUID()));
      series.connA_unknownExt.push(await time(P.connA, 'doc-' + randomUUID()));
      series.connA_notYours.push(await time(P.connA, targets.notYours));
      series.userA_unknownUuid.push(await time(P.userA, randomUUID()));
    }
    const med = (a: number[]) => { const s = [...a].sort((x, y) => x - y); return s[Math.floor(s.length / 2)]; };
    const p = (a: number[], q: number) => { const s = [...a].sort((x, y) => x - y); return s[Math.floor(s.length * q)]; };
    rec({ cell: 'ITEM3', n: TIMING_N, summary: Object.fromEntries(Object.entries(series).map(([k, v]) => [k, { median: +med(v).toFixed(3), p10: +p(v, 0.1).toFixed(3), p90: +p(v, 0.9).toFixed(3) }])), series });
  }, 600000);
