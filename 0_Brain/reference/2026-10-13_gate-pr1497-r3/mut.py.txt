def sha(p): return hashlib.sha256(open(p, "rb").read()).hexdigest()
def nth(s, old, new, n):
    i = -1
    for _ in range(n + 1):
        i = s.find(old, i + 1)
        assert i >= 0, ("anchor", old, n)
    return s[:i] + new + s[i + len(old):]
def rep(s, old, new, count):
    assert s.count(old) == count, ("count", old, s.count(old), count); return s.replace(old, new)

def reds(t):
    out = []
    for l in t.splitlines():
        m = re.match(r"\s+✕\s+(.*?)(?:\s+\(\d+ ms\))?$", l)
        if m: out.append(re.sub(r"^🔴\s*", "", m.group(1))[:60])
    return out

def layer(name, tag):

        r = subprocess.run(["npx", "jest", "ks695-ask3-erase-connector-erasure"], capture_output=True, text=True, cwd=O); t = r.stdout + r.stderr
    else:
        env = dict(os.environ)

        elif name in ("aF", "aT"): pat, extra = "gate1497r3-attack", []
        elif name in ("sF", "sT"): pat, extra = "gate1497r3-attack", ["B2"]; env["GATE_SLOW"] = "1"; env["GATE_FAST"] = "0"
        mt = "true" if name.endswith("T") else "false"
        r = subprocess.run(["zsh", G + "/tools/integ.sh", "mut_" + tag + "_" + name, "app", mt, pat] + extra, capture_output=True, text=True, env=env)
        t = open(G + "/logs/integ_mut_" + tag + "_" + name + ".log", encoding="utf-8", errors="replace").read()
    has = bool(re.search(r"^Tests:", t, re.M))
    tot = (re.search(r"^Tests:.*$", t, re.M) or [""])[0] if has else "NO Tests: LINE"
    return has, reds(t), tot
