/**
 * =============================================================================
 * KS-1240 (part 2, auth social providers) — every provider fetch is bounded, and a provider that does not answer in
 * time is CLASSIFIED as unavailable
 * =============================================================================
 * The six social provider adapters (`providers/social/*.ts`) made 12 fetches with no timeout. A provider that never
 * answered, or answered the headers and stalled the body, held the sign-in request open with no bound: on the login
 * callback (`routes/auth.ts`, the OAuth state already consumed) and on `POST /social/link` alike.
 *
 * Now each call is bounded at 10 s, and a call that does not finish in time surfaces from `exchangeCode` as the
 * codebase's existing `ServiceUnavailableError` (503 SERVICE_UNAVAILABLE, KS-253), thrown from the provider file, so
 * `routes/auth.ts` is unchanged on both paths:
 *  - `POST /social/link` has no local try, so the error reaches the error handler and answers 503;
 *  - the login callback wraps every exchange failure in `BadRequestError('OAuth authentication failed: ' + message)`,
 *    so it still answers 400, now with the classified text instead of the raw abort message. The state was consumed
 *    before the exchange, as before: a timed-out sign-in needs a fresh one (pinned below).
 * A refused or unreachable provider connection is classified the same way, with its own text (gate 5, round 1). A
 * provider that answers with an error status is not: its Error now carries only the status, and the provider's own
 * error body is logged server-side, because the login callback echoed it to the client (gate 5 finding 1788-1). A
 * profile call answered 401 still rejects with its adapter's own profile error, never a profile and never 503.
 *
 * The cells assert the status, code and message, never `instanceof ServiceUnavailableError`: at this base
 * `AppError`'s constructor resets every subclass instance's prototype to `AppError.prototype` (KS 1467), so only
 * `instanceof AppError`, which the error handler uses, holds.
 *
 * Harness: the REAL adapter classes, driven directly and through the REAL routes (the factory is stubbed only to hand
 * them out). Each adapter's hard-coded provider URL is rewritten to one local stub server, and the real `fetch` is
 * called with the adapter's own init, signal included, so a hang and a stalled body are real network behaviour.
 * `AbortSignal.timeout` is spied: each call records the bound asked for (that pins 10 000 ms) and gets a 200 ms signal
 * back. At the base nothing calls it, the exchange never settles, and the cell gives up after 4 s, which is the red.
 * =============================================================================
 */

async function listen(server: http.Server): Promise<string> {
  await new Promise<void>((resolve) => server.listen(0, '127.0.0.1', () => resolve()));
  return `http://127.0.0.1:${(server.address() as AddressInfo).port}`;
}
async function closeServer(server: http.Server | undefined): Promise<void> {
  if (!server) return;
  server.closeAllConnections();
  await new Promise<void>((r) => server.close(() => r()));
}

type Outcome = { gaveUp: true } | { gaveUp: false; resolved: boolean; status?: unknown; code?: unknown; name?: unknown; message: string; ms: number };
async function exchange(p: SocialProvider): Promise<Outcome> {
  const t0 = Date.now();
  let timer: NodeJS.Timeout | undefined;
  const gaveUp = new Promise<Outcome>((r) => { timer = setTimeout(() => r({ gaveUp: true }), GIVE_UP_MS); });
  const run = p.exchangeCode('code-ks1240b').then(
    (): Outcome => ({ gaveUp: false, resolved: true, message: 'resolved', ms: Date.now() - t0 }),
    (e: { statusCode?: unknown; code?: unknown; name?: unknown; message?: unknown }): Outcome =>
      ({ gaveUp: false, resolved: false, status: e?.statusCode, code: e?.code, name: e?.name, message: String(e?.message), ms: Date.now() - t0 }),
  );
  const out = await Promise.race([run, gaveUp]);
  clearTimeout(timer);
  return out;
}

type Res = { status: number; code: string | null; message: string | null; raw: string; ms: number };
function call(method: 'GET' | 'POST', p: string, body?: unknown): Promise<Res> {
  return new Promise((resolve, reject) => {
    const target = new URL(appUrl);
    const payload = body === undefined ? undefined : JSON.stringify(body);
    const t0 = Date.now();
    let settled = false;
    const req = http.request({ hostname: target.hostname, port: target.port, path: p, method, agent: false,
      headers: payload ? { 'content-type': 'application/json', 'content-length': String(Buffer.byteLength(payload)), authorization: 'Bearer ks1240b' } : { authorization: 'Bearer ks1240b' } }, (res) => {
      const c: Buffer[] = [];
      res.on('data', (d) => c.push(d));
      res.on('end', () => {
        if (settled) return;
        settled = true;
        clearTimeout(giveUp);
        let code: string | null = null; let message: string | null = null;
        try { const e = JSON.parse(Buffer.concat(c).toString())?.error; code = e?.code ?? null; message = e?.message ?? null; } catch { /* not JSON */ }
        resolve({ status: res.statusCode ?? 0, code, message, raw: Buffer.concat(c).toString(), ms: Date.now() - t0 });
      });
    });
    const giveUp = setTimeout(() => { if (settled) return; settled = true; req.destroy(); resolve({ status: 0, code: 'CLIENT_GAVE_UP', message: null, raw: '', ms: Date.now() - t0 }); }, GIVE_UP_MS);
    req.on('error', (err) => { if (!settled) { settled = true; clearTimeout(giveUp); reject(err); } });
    if (payload) req.write(payload);
    req.end();
  });
}

const SOCIAL = path.resolve(__dirname, '../providers/social');
function boundedness(file: string): boolean[] {
  const sf = ts.createSourceFile(file, readFileSync(path.join(SOCIAL, file), 'utf8'), ts.ScriptTarget.Latest, true, ts.ScriptKind.TS);
  const out: boolean[] = [];
  const visit = (n: ts.Node): void => {
    if (ts.isCallExpression(n) && ts.isIdentifier(n.expression) && n.expression.text === 'fetch') {
      const init = n.arguments[1];
      out.push(!!init && ts.isObjectLiteralExpression(init) && init.properties.some((p) =>
        (ts.isPropertyAssignment(p) || ts.isShorthandPropertyAssignment(p)) && p.name.getText(sf) === 'signal'));
    }
    ts.forEachChild(n, visit);
  };
  visit(sf);
  return out;
}

  it('🔴 12 fetch( calls across the six files, every one bounded', () => {
    expect(Object.fromEntries(FAMILIES.map((f) => [f.file, boundedness(f.file)]))).toEqual(
      Object.fromEntries(FAMILIES.map((f) => [f.file, f.sites.map(() => true)])));
  });
});

async function freshState(provider: string): Promise<string> {
  const r = await new Promise<string>((resolve, reject) => {
    const target = new URL(appUrl);
    http.get({ hostname: target.hostname, port: target.port, path: `/api/auth/social/${provider}`, agent: false }, (res) => {
      const c: Buffer[] = []; res.on('data', (d) => c.push(d)); res.on('end', () => resolve(Buffer.concat(c).toString()));
    }).on('error', reject);
  });
  return String(JSON.parse(r).state);
}

    const r = await call('POST', '/api/auth/social/link', { provider: 'github', code: 'c6' });
    expect([r.status, r.code, r.message]).toEqual([503, 'SERVICE_UNAVAILABLE', unreachable('GitHub')]);
  }, 15000);

  it('🔴 login callback, GitHub refuses the CONNECTION: 400 with the classified text, not "fetch failed"', async () => {
    const state = await freshState('github');
    mode['github.com/login/oauth/access_token'] = 'deadport';
    const r = await call('GET', `/api/auth/social/github/callback?code=c7&state=${encodeURIComponent(state)}`);
    expect([r.status, r.code, r.message]).toEqual([400, 'BAD_REQUEST', `OAuth authentication failed: ${unreachable('GitHub')}`]);
  }, 15000);

  it('🟢 control: a refused exchange on POST /social/link still answers 500 INTERNAL_ERROR, as before', async () => {
    mode['github.com/login/oauth/access_token'] = 'refuse';
    const r = await call('POST', '/api/auth/social/link', { provider: 'github', code: 'c5' });
    expect([r.status, r.code]).toEqual([500, 'INTERNAL_ERROR']);
  }, 15000);
});
