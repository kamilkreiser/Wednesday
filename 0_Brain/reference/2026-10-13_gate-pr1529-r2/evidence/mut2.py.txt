def sha(p): return hashlib.sha256(open(p, "rb").read()).hexdigest()

def run(cmd, tag, extra=None):
    e = dict(env); e.update(extra or {})
    r = subprocess.run(cmd, cwd=SVC, capture_output=True, text=True, env=e)
    out = re.sub(r"\x1b\[[0-9;]*m", "", r.stdout + r.stderr).replace(PW, "<REDACTED>")
    open(f"{EV}/mut2_{tag}.txt", "w").write(out + f"\n# rc={r.returncode}\n")
    m = re.search(r"Tests\s+(?:(\d+) failed \| )?(\d+) passed", out)
    failed = int(m.group(1) or 0) if m else 0
    passed = int(m.group(2)) if m else 0
    if not m:
        mf = re.search(r"Tests\s+(\d+) failed", out)
        failed = int(mf.group(1)) if mf else 0
    cells = sorted({re.sub(r"\s+\d+ms$", "", re.sub(r"^.*> ", "", l)).strip() for l in out.splitlines() if re.match(r"\s*×", l)})
    return {"passed": passed, "failed": failed, "cells": cells, "loadfail": passed == 0 and failed == 0, "rc": r.returncode}

PIN_MUTANTS = {"i1_pin_removed", "i3_pin_repeatable_read", "i4_pin_serializable", "g3_pin_skips_grace", "M5_pin_after_guc", "M6_pin_inside_savepoint", "lock_removed", "M2_lock_tenant_only"}

def apply(text, edits):
    for find, repl in edits:
        if text.count(find) != 1:
            return None, f"anchor count {text.count(find)}: {find[:70]!r}"
        text = text.replace(find, repl, 1)
    return text, None

def one(mid, why, edits, marker, origin):
    base_text = open(SRC, encoding="utf8").read(); base_sha = sha(SRC)
    text, err = apply(base_text, edits)
    if text is None:
        return {"id": mid, "origin": origin, "status": "ANCHOR_FAIL", "err": err}
    open(SRC, "w", encoding="utf8").write(text)
    try:
        hits = open(SRC, encoding="utf8").read().count(marker)
        assert hits >= 1, f"{mid}: marker {marker} absent"
        res = {"id": mid, "origin": origin, "why": why, "marker_hits": hits}
        if mid.startswith("W2"):
            res["harness_rc"] = run(HARN, f"{mid}_harn", {"GATE_TAG": f"mut_{mid}", "GATE_ONLY": "W2"})
            return res
        res["unit"] = run(UNIT, f"{mid}_unit")
        res["author_integ"] = run(INTEG, f"{mid}_integ")
        res["harness_rc"] = run(HARN, f"{mid}_harn_rc", {"GATE_TAG": f"mut_{mid}_rc"})
        if mid in PIN_MUTANTS:
            res["harness_rr"] = run(HARN, f"{mid}_harn_rr", {"GATE_TAG": f"mut_{mid}_rr", "GATE_ISO": "repeatable read"})
        suites = [res[k] for k in ("unit", "author_integ", "harness_rc", "harness_rr") if k in res]
        if any(s["loadfail"] for s in suites):
            res["status"] = "LOADFAIL"
        else:
            res["status"] = "KILLED" if any(s["cells"] for s in suites) else "SURVIVED"
        return res
    finally:
        open(SRC, "w", encoding="utf8").write(base_text)
        assert sha(SRC) == base_sha, f"{mid}: RESTORE FAILED"

def main():
    sel = sys.argv[1]
    table = []
    if sel in ("all", "author"):
        table += [(m, *AUTHOR_TABLE[m], "author") for m in AUTHOR_TABLE]
    if sel in ("all", "gate"):
        table += [(m, *GATE_TABLE[m], "gate") for m in GATE_TABLE if not m.startswith("W2")]
    if sel not in ("all", "author", "gate"):
        for m in sel.split(","):
            table.append((m, *(AUTHOR_TABLE.get(m) or GATE_TABLE[m]), "author" if m in AUTHOR_TABLE else "gate"))
    base_sha = sha(SRC)
    print("baseline index.ts sha256/16", base_sha[:16], flush=True)
    for mid, why, edits, marker, origin in table:
        r = one(mid, why, edits, marker, origin)
        with open(f"{EV}/mut2_results.jsonl", "a") as f:
            f.write(json.dumps(r) + "\n")
        def s(k):
            x = r.get(k)
            return f"{x['passed']}p/{x['failed']}f" if x else "-"
        print(f"{r.get('status','?'):9} {mid:36} unit {s('unit')} author {s('author_integ')} harnRC {s('harness_rc')} harnRR {s('harness_rr')}", flush=True)
    assert sha(SRC) == base_sha
    print("final sha ok", sha(SRC)[:16], flush=True)

main()
