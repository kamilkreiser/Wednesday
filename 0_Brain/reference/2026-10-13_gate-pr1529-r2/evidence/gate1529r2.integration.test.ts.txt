const PRIVATE_PEM = privateKey.export({ type: 'pkcs8', format: 'pem' }).toString();
const PUBLIC_PEM = publicKey.export({ type: 'spki', format: 'pem' }).toString();
const b64url = (b: Buffer): string => b.toString('base64').replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
function token(): string {
  const header = b64url(Buffer.from(JSON.stringify({ alg: 'RS256', typ: 'JWT' })));
  const now = Math.floor(Date.now() / 1000);
  const payload = b64url(Buffer.from(JSON.stringify({ sub: 'g1529r2', iat: now, exp: now + 600, role: 'super_admin' })));
  const sig = b64url(crypto.sign('RSA-SHA256', Buffer.from(`${header}.${payload}`), PRIVATE_PEM));
  return `${header}.${payload}.${sig}`;
}
const H = () => ({ 'Content-Type': 'application/json', Authorization: `Bearer ${token()}` });

type Minted = { status: number; id: string | null; key: string | null; priorKeysRevoked: unknown; ms: number; t0: number; t1: number };
async function mint(c: string, rotate: boolean, name: string, tenantId = TENANT): Promise<Minted> {
  const t0 = Date.now();
  const res = await fetch(`${base}/api/keys`, { method: 'POST', headers: H(), body: JSON.stringify({ name, scopes: ['documents:read'], tenantId, organizationId: ORG, connectorId: c, rotate }) });
  const text = await res.text(); let d: any = {};
  try { d = JSON.parse(text).data ?? {}; } catch { /* */ }
  const t1 = Date.now();
  return { status: res.status, id: d.id ?? null, key: d.key ?? null, priorKeysRevoked: d.priorKeysRevoked, ms: t1 - t0, t0, t1 };
}
async function validate(key: string): Promise<{ status: number; valid: unknown; reason: unknown; ms: number }> {
  const t0 = Date.now();
  const res = await fetch(`${base}/api/keys/validate`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ key }) });
  const b: any = await res.json().catch(() => ({}));
  return { status: res.status, valid: b.data?.valid, reason: b.data?.reason, ms: Date.now() - t0 };
}
async function revoke(id: string): Promise<{ status: number }> {
  const res = await fetch(`${base}/api/keys/${id}`, { method: 'DELETE', headers: H() });
  await res.text();
  return { status: res.status };
}
async function listed(id: string): Promise<unknown> {
  const res = await fetch(`${base}/api/keys?organizationId=${ORG}`, { headers: H() });
  const b: any = await res.json().catch(() => ({}));
  const k = (b.data ?? []).find((x: any) => x.id === id);
  return k ? k.isActive : 'absent';
}
type Row = { id: string; is_active: boolean; expires_at: Date | null; name: string; tenant_id: string };
async function rowsOf(c: string, tenantId?: string): Promise<Row[]> {
  const q = tenantId
    ? superPool.query<Row>('SELECT id, is_active, expires_at, name, tenant_id FROM svc_api_keys WHERE connector_id = $1 AND tenant_id = $2::uuid ORDER BY created_at', [c, tenantId])
    : superPool.query<Row>('SELECT id, is_active, expires_at, name, tenant_id FROM svc_api_keys WHERE connector_id = $1 ORDER BY created_at', [c]);
  return (await q).rows;
}

const memActive = (c: string) => [...mem.values()].filter((k) => k.connectorId === c && k.isActive).map((k) => k.id).sort();

// deferred COMMIT trigger: rows whose name starts with 'gfail' sleep 2 s then raise; 'gslow' sleep 2 s and commit
async function armCommit(): Promise<void> {
  await superPool.query(`CREATE OR REPLACE FUNCTION g2_commit() RETURNS trigger LANGUAGE plpgsql AS $$
    BEGIN
      IF NEW.connector_id LIKE '${PREFIX}%' AND NEW.name LIKE 'gfail%' THEN
        PERFORM pg_sleep(2);
        RAISE EXCEPTION 'g2 injected commit failure';
      END IF;
      IF NEW.connector_id LIKE '${PREFIX}%' AND NEW.name LIKE 'gslow%' THEN
        PERFORM pg_sleep(2);
      END IF;
      RETURN NULL;
    END $$`);
  await superPool.query('DROP TRIGGER IF EXISTS g2_commit ON svc_api_keys');
  await superPool.query('CREATE CONSTRAINT TRIGGER g2_commit AFTER INSERT ON svc_api_keys DEFERRABLE INITIALLY DEFERRED FOR EACH ROW EXECUTE FUNCTION g2_commit()');
}
async function disarmCommit(): Promise<void> {
  await superPool.query('DROP TRIGGER IF EXISTS g2_commit ON svc_api_keys');
  await superPool.query('DROP FUNCTION IF EXISTS g2_commit()');
}
async function armRetireFailure(): Promise<void> {
  await superPool.query(`CREATE OR REPLACE FUNCTION g2_fail_retire() RETURNS trigger LANGUAGE plpgsql AS $$
    BEGIN
      IF NEW.connector_id LIKE '${PREFIX}%:k577%' AND (OLD.is_active AND NOT NEW.is_active OR NEW.expires_at IS DISTINCT FROM OLD.expires_at) THEN
        RAISE EXCEPTION 'g2 injected retire failure';
      END IF;
      RETURN NEW;
    END $$`);
  await superPool.query('DROP TRIGGER IF EXISTS g2_fail_retire ON svc_api_keys');
  await superPool.query('CREATE TRIGGER g2_fail_retire BEFORE UPDATE ON svc_api_keys FOR EACH ROW EXECUTE FUNCTION g2_fail_retire()');
}
async function disarmRetireFailure(): Promise<void> {
  await superPool.query('DROP TRIGGER IF EXISTS g2_fail_retire ON svc_api_keys');
  await superPool.query('DROP FUNCTION IF EXISTS g2_fail_retire()');
}

      await sleep(800);
      const hadIt = mem.has(s.id!); mem.delete(s.id!);
      const r = await p;
      const afterFail = mem.has(s.id!);
      const v = await validate(s.key!);
      out.push({ hadIt, st: r.status, resurrected: afterFail, v: v.valid, memAfterValidate: mem.get(s.id!)?.isActive });
    } finally { await disarmCommit(); }
    results.A2_deleted = out;
    expect(out.filter((o) => !o.hadIt || o.st < 500 || o.resurrected !== false || o.v !== true)).toEqual([]);
  });

  // A3 / A3b (round-1 shapes) plus my own: the rotate SUCCEEDS while the revoke waits (A3c), grace 0 and 60, revoke save ok / fails.
  cell('A3_family', async () => {
    const out: any[] = [];
    await armCommit();
    try {
      for (const g of ['0', '60']) {
        process.env.API_KEY_ROTATION_GRACE_SECONDS = g;
        for (const rotateFails of [true, false]) {
          for (const revokeFails of [false, true]) {
            for (let t = 0; t < 2; t += 1) {
              const c = conn(`a3-g${g}-rf${rotateFails}-vf${revokeFails}-${t}`);
              const s = await mint(c, false, 's');
              let settled = false;
              const p = mint(c, true, rotateFails ? 'gfail rot' : 'gslow rot').finally(() => { settled = true; });
              await sleep(700);
              if (revokeFails) await flagOn('failrevoke');
              const pendingAtIssue = !settled;
              const rv = revoke(s.id!);
              await sleep(200);
              const waiting = (await superPool.query("SELECT count(*)::int AS n FROM pg_stat_activity WHERE wait_event_type = 'Lock' AND wait_event <> 'advisory' AND query ILIKE 'INSERT INTO svc_api_keys%'")).rows[0].n;
              const r = await p; const rvr = await rv;
              await flagsOff();
              const rows = await rowsOf(c);
              const seedRow = rows.find((x) => x.id === s.id)!;
              const memSeedBefore = mem.get(s.id!)?.isActive; const listedBefore = await listed(s.id!); // read BEFORE any validate
              const v1 = await validate(s.key!); const v2 = await validate(s.key!);
              const vNew = r.key ? (await validate(r.key)).valid : null;
              out.push({ g, rotateFails, revokeFails, t, pendingAtIssue, waitingOnRowLock: waiting, rotate: r.status, revoke: rvr.status, dbSeedActive: seedRow.is_active, dbSeedExp: seedRow.expires_at, memSeedActive: memSeedBefore, listed: listedBefore, v1: v1.valid, v2: v2.valid, vNew });
            }
          }
        }
      }
    } finally { delete process.env.API_KEY_ROTATION_GRACE_SECONDS; await disarmCommit(); await flagsOff(); }
    results.A3_family = out;
    expect(out.filter((o) => !o.pendingAtIssue || o.waitingOnRowLock < 1)).toEqual([]); // instrument: the revoke landed in the window
    // the revoke must hold everywhere it is read: validate, memory and the operator list
    expect(out.filter((o) => o.v1 !== false || o.v2 !== false || o.memSeedActive !== false || o.listed !== false)).toEqual([]);
    // the failing-save trigger fires only on an active->inactive change, so after a SUCCEEDING grace-0 rotate (which already set
    // is_active=false) the revoke's save has nothing to flip and stores fine: expected 200 there.
    expect(out.filter((o) => o.revoke !== (o.revokeFails && (o.rotateFails || o.g === '60') ? 500 : 200))).toEqual([]);
    // DB: a successful revoke save stores is_active=false; a successful rotate retires/expires it; a failing rotate leaves it
    expect(out.filter((o) => !o.revokeFails && o.dbSeedActive !== false)).toEqual([]);
    expect(out.filter((o) => !o.rotateFails && (o.rotate !== 201 || o.vNew !== true))).toEqual([]);
    expect(out.filter((o) => o.rotateFails && o.rotate < 500)).toEqual([]);
  });

  // A3 CONTROL: the failing revoke with no rotate in flight.
  cell('A3_control', async () => {
    const c = conn('a3-ctl');
    const s = await mint(c, false, 's');
    await flagOn('failrevoke');
    const rv = await revoke(s.id!);
    await flagsOff();
    const out = { revoke: rv.status, dbSeedActive: (await rowsOf(c))[0].is_active, memSeedActive: mem.get(s.id!)?.isActive, v: (await validate(s.key!)).valid };
    results.A3_control = out;
    expect(out.revoke).toBe(500); expect(out.dbSeedActive).toBe(true); expect(out.v).toBe(false);
  });

  // A6 (mine): a validate of the NEW key issued inside the commit window of a SUCCEEDING rotate: not found until commit,
  // valid after; no stale entry left. And the prior key at +800: valid (pre-commit), then refused after commit.
  cell('A6_success_window', async () => {
    const out: any[] = [];
    await armCommit();
    try {
      for (const g of ['0', '60']) {
        process.env.API_KEY_ROTATION_GRACE_SECONDS = g;
        const c = conn(`a6-g${g}`);
        const s = await mint(c, false, 's');
        let settled = false;
        const p = mint(c, true, 'gslow rot').finally(() => { settled = true; });
        await sleep(800);
        const pending = !settled;
        const memMid = mem.get(s.id!)?.isActive; const memExpMid = mem.get(s.id!)?.expiresAt ?? null;
        const dbMid = activeIds(await rowsOf(c));
        const chainMid = chains ? chains.has(`${TENANT}:${c}`) : 'n/a';
        const mid = await validate(s.key!);
        const r = await p;
        const rows = await rowsOf(c);
        const newInMemBefore = mem.get(r.id!)?.isActive; const seedActBefore = mem.get(s.id!)?.isActive; const seedExpBefore = !!mem.get(s.id!)?.expiresAt; // BEFORE any validate
        const vSeed = await validate(s.key!);
        out.push({ g, pending, memMid, memExpMid, dbMid: JSON.stringify(dbMid) === JSON.stringify([s.id]), chainMid, mid: mid.reason ?? mid.valid, st: r.status, newInMem: newInMemBefore, memSeedAfter: seedActBefore, memSeedExpAfter: seedExpBefore, dbSeed: rows.find((x) => x.id === s.id), vSeed: vSeed.reason ?? vSeed.valid, vNew: (await validate(r.key!)).valid, chainAfter: chains ? chains.has(`${TENANT}:${c}`) : 'n/a' });
      }
    } finally { delete process.env.API_KEY_ROTATION_GRACE_SECONDS; await disarmCommit(); }
    results.A6_success_window = out;
    expect(out.filter((o) => !o.pending || o.memMid !== true || o.memExpMid !== null || !o.dbMid || o.mid !== true)).toEqual([]);
    expect(out.filter((o) => o.st !== 201 || o.newInMem !== true || o.vNew !== true || o.chainAfter === true)).toEqual([]);
    expect(out.filter((o) => (o.g === '0' ? (o.vSeed !== 'Key revoked' || o.memSeedAfter !== false) : (o.vSeed !== true || o.memSeedExpAfter !== true)))).toEqual([]);
  });

  // CRASH: COMMIT happened, memory never updated. (a) the author's simulation; (b) a real reload (loadFromDb) afterwards.
  cell('CRASH', async () => {
    const out: any[] = [];
    for (const g of ['0', '60']) {
      process.env.API_KEY_ROTATION_GRACE_SECONDS = g;
      try {
        const c = conn(`crash-g${g}`);
        const s = await mint(c, false, 's');
        const r = await mint(c, true, 'rot');
        const seedObj = mem.get(s.id!); seedObj.isActive = true; seedObj.expiresAt = undefined; mem.delete(r.id!);
        const listedStale = await listed(s.id!);
        const vSeed = await validate(s.key!); const vNew = await validate(r.key!);
        await mod.loadFromDb();
        out.push({ g, st: r.status, listedStale, vSeed: vSeed.reason ?? vSeed.valid, vNew: vNew.valid, afterReload: { seed: mem.get(s.id!)?.isActive, seedExp: !!mem.get(s.id!)?.expiresAt, nw: mem.get(r.id!)?.isActive } });
      } finally { delete process.env.API_KEY_ROTATION_GRACE_SECONDS; }
    }
    results.CRASH = out;
    expect(out.filter((o) => o.vNew !== true || (o.g === '0' ? o.vSeed === true : o.vSeed !== true))).toEqual([]);
  });

  // ---- the chain ----
  cell('CH1_timeout', async () => {
    // A: a SUCCEEDING slow commit (2 s); B queued behind it with ROTATE_CHAIN_WAIT_MS=300. B proceeds after ~300 ms, takes a client,
    // and the DB lock holds it until A commits. Afterwards: DB = [B], memory = [B].
    const out: any[] = [];
    const lines: string[] = []; const orig = console.log;
    console.log = (...a: unknown[]) => { const s = a.map(String).join(' '); if (s.includes('KS-1556')) lines.push(s); orig(...a); };
    await armCommit();
    process.env.ROTATE_CHAIN_WAIT_MS = '300';
    try {
      for (let t = 0; t < 3; t += 1) {
        const c = conn(`ch1-${t}`);
        const s = await mint(c, false, 's');
        const pa = mint(c, true, 'gslow A');
        await sleep(100);
        const pb = mint(c, true, 'B');
        await sleep(600);
        const waitingAdvisory = (await superPool.query("SELECT count(*)::int AS n FROM pg_stat_activity WHERE wait_event_type = 'Lock' AND wait_event = 'advisory'")).rows[0].n;
        const [a, b] = await Promise.all([pa, pb]);
        const rows = await rowsOf(c);
        out.push({ t, a: a.status, b: b.status, bAfterA: b.t1 >= a.t1, waitingAdvisory, db: JSON.stringify(activeIds(rows)) === JSON.stringify([b.id]), mem: JSON.stringify(memActive(c)) === JSON.stringify([b.id]), seedMem: mem.get(s.id!)?.isActive, aMem: mem.get(a.id!)?.isActive });
      }
    } finally { delete process.env.ROTATE_CHAIN_WAIT_MS; await disarmCommit(); console.log = orig; }
    results.CH1_timeout = { out, timeoutLogLines: lines.filter((l) => l.includes('waited the whole chain timeout')).length };
    expect(out.filter((o) => o.a !== 201 || o.b !== 201 || !o.db || !o.mem)).toEqual([]);
    expect(out.filter((o) => o.waitingAdvisory < 1)).toEqual([]); // instrument: B really proceeded past the chain and waited in the DB
  });

  cell('CH2_pred_throws', async () => {
    // A fails at COMMIT (2 s); B queued (default 10 s wait). B must start as soon as A settles, not after 10 s, and succeed.
    const out: any[] = [];
    await armCommit();
    try {
      for (let t = 0; t < 3; t += 1) {
        const c = conn(`ch2-${t}`);
        const s = await mint(c, false, 's');
        const pa = mint(c, true, 'gfail A');
        await sleep(100);
        const pb = mint(c, true, 'B');
        await sleep(600);
        const waitingAdvisory = (await superPool.query("SELECT count(*)::int AS n FROM pg_stat_activity WHERE wait_event_type = 'Lock' AND wait_event = 'advisory'")).rows[0].n;
        const [a, b] = await Promise.all([pa, pb]);
        const rows = await rowsOf(c);
        out.push({ t, a: a.status, b: b.status, gapMs: b.t1 - a.t1, waitingAdvisory, db: JSON.stringify(activeIds(rows)) === JSON.stringify([b.id]), mem: JSON.stringify(memActive(c)) === JSON.stringify([b.id]), seedMem: mem.get(s.id!)?.isActive, chainLeft: chains?.has(`${TENANT}:${c}`) });
      }
    } finally { await disarmCommit(); }
    results.CH2_pred_throws = out;
    expect(out.filter((o) => o.a < 500 || o.b !== 201 || o.gapMs > 1500 || !o.db || !o.mem || o.chainLeft)).toEqual([]);
    expect(out.filter((o) => o.waitingAdvisory !== 0)).toEqual([]); // with the chain, B waits in-process, not on the DB lock
  });

  cell('CH3_never_settles', async () => {
    // a predecessor that never settles (test seam: withConnectorChain on the rotate's own key) holds a real rotate only for
    // ROTATE_CHAIN_WAIT_MS, then the rotate succeeds; the map entry is gone afterwards.
    if (!mod.withConnectorChain) throw new Error('seam absent');
    process.env.ROTATE_CHAIN_WAIT_MS = '400';
    try {
      const c = conn('ch3');
      const s = await mint(c, false, 's');
      void mod.withConnectorChain(`${TENANT}:${c}`, () => new Promise<never>(() => { /* never */ }));
      const r = await mint(c, true, 'rot');
      const r2 = await mint(c, true, 'rot2');
      results.CH3_never_settles = { st: r.status, ms: r.ms, st2: r2.status, ms2: r2.ms, db: JSON.stringify(activeIds(await rowsOf(c))) === JSON.stringify([r2.id]), chainLeft: chains!.has(`${TENANT}:${c}`), seed: s.status };
      const o: any = results.CH3_never_settles;
      expect(o.st).toBe(201); expect(o.ms).toBeGreaterThanOrEqual(350); expect(o.ms).toBeLessThan(3000);
      expect(o.st2).toBe(201); expect(o.ms2).toBeLessThan(350); expect(o.db).toBe(true); expect(o.chainLeft).toBe(false);
    } finally { delete process.env.ROTATE_CHAIN_WAIT_MS; }
  });

  cell('CH4_entries', async () => {
    const before = chains!.size;
    const cs = Array.from({ length: 25 }, (_, i) => conn(`ch4-${i}`));
    for (const c of cs) await mint(c, false, 's');
    const p = Promise.all(cs.flatMap((c) => [mint(c, true, 'a'), mint(c, true, 'b')]));
    await sleep(5);
    const mid = chains!.size;
    const all = await p;
    const after = chains!.size;
    const bad = [] as string[];
    for (const c of cs) if (activeIds(await rowsOf(c)).length !== 1) bad.push(c);
    results.CH4_entries = { before, mid, after, non201: all.filter((x) => x.status !== 201).length, bad: bad.length, keysLeft: [...chains!.keys()].filter((k) => k.includes(PREFIX)).length };
    expect(after).toBe(before); expect(bad).toEqual([]); expect(all.filter((x) => x.status !== 201)).toEqual([]);
  });

  cell('CH5_two_tenants', async () => {
    // same connector id in TWO tenants: a slow rotate in T1 must NOT hold a rotate in T2 (chain keyed by tenant+connector);
    // CONTROL: the same in T1 does wait.
    const out: any = {};
    await armCommit();
    try {
      const c = conn('ch5');
      const s1 = await mint(c, false, 's1', TENANT); const s2 = await mint(c, false, 's2', TENANT2);
      const pa = mint(c, true, 'gslow T1', TENANT);
      await sleep(200);
      const keysMid = [...chains!.keys()].filter((k) => k.endsWith(c));
      const b = await mint(c, true, 'T2', TENANT2);
      const a = await pa;
      out.other = { s1: s1.status, s2: s2.status, a: a.status, b: b.status, bMs: b.ms, bFinishedBeforeA: b.t1 < a.t1, keysMid, t1: activeIds(await rowsOf(c, TENANT)), t2: activeIds(await rowsOf(c, TENANT2)), aId: a.id, bId: b.id };
      // control: same tenant
      const pc = mint(c, true, 'gslow T1 again', TENANT);
      await sleep(200);
      const d = await mint(c, true, 'T1 second', TENANT);
      const cc = await pc;
      out.same = { c: cc.status, d: d.status, dMs: d.ms, dAfterC: d.t1 >= cc.t1 };
    } finally { await disarmCommit(); }
    results.CH5_two_tenants = out;
    expect(out.other.a).toBe(201); expect(out.other.b).toBe(201);
    expect(out.other.bFinishedBeforeA).toBe(true); expect(out.other.bMs).toBeLessThan(1000);
    expect(out.other.t1).toEqual([out.other.aId]); expect(out.other.t2).toEqual([out.other.bId]);
    expect(out.same.dAfterC).toBe(true); expect(out.same.dMs).toBeGreaterThan(1200); // control: the chain does serialise one tenant
  });

  // W2: OPT-IN widening probe (run only with GATE_ONLY=W2 against the W2 probe patch, which delays the post-COMMIT memory apply of
  // a rotate named 'gdelay*' by 1500 ms). Bounded wait 300 ms vs the default 10 s: does memory end in DB commit order?
  (ONLY.split(',').includes('W2') ? it : it.skip)('W2', async () => {
    const out: any[] = [];
    for (const waitMs of ['300', '']) {
      if (waitMs) process.env.ROTATE_CHAIN_WAIT_MS = waitMs; else delete process.env.ROTATE_CHAIN_WAIT_MS;
      try {
        for (let t = 0; t < 3; t += 1) {
          const c = conn(`w2-${waitMs || 'default'}-${t}`);
          const s = await mint(c, false, 's');
          const pa = mint(c, true, 'gdelay A');
          await sleep(100);
          const pb = mint(c, true, 'B');
          const [a, b] = await Promise.all([pa, pb]);
          const rows = await rowsOf(c);
          // memory and the list are read BEFORE any validate: a validate re-reads the stored row and would correct memory
          const memNow = memActive(c).map((x) => (x === a.id ? 'A' : x === b.id ? 'B' : 'seed'));
          const listedA = await listed(a.id!);
          const vA = await validate(a.key!); const vB = await validate(b.key!);
          const memAfterV = memActive(c).map((x) => (x === a.id ? 'A' : x === b.id ? 'B' : 'seed'));
          out.push({ waitMs: waitMs || 'default', t, a: a.status, b: b.status, bBeforeA: b.t1 < a.t1, db: activeIds(rows).map((x) => (x === a.id ? 'A' : x === b.id ? 'B' : 'seed')), mem: memNow, listedA, vA: vA.reason ?? vA.valid, vB: vB.valid, memAfterV, seed: s.status });
        }
      } finally { delete process.env.ROTATE_CHAIN_WAIT_MS; }
    }
    results.W2 = out;
    expect(out.length).toBe(6);
  }, 180_000);

  cell('CH6_pool', async () => {
    // pool pressure: hold 8 of the service pool's 10 clients, so 2 remain. A slow rotate (2 s commit) holds one; three more rotates of
    // the SAME connector queue. A validate of an unrelated key at +400 ms must get the last client at once. Without the in-process
    // chain the queued rotates take that client and wait on the advisory lock.
    const out: any[] = [];
    await armCommit();
    const held: any[] = [];
    try {
      const other = conn('ch6-other'); const o = await mint(other, false, 'o');
      expect((await validate(o.key!)).valid).toBe(true);
      for (let t = 0; t < 3; t += 1) {
        const c = conn(`ch6-${t}`); await mint(c, false, 's');
        for (let i = 0; i < 8; i += 1) held.push(await db.getClient());
        const pa = mint(c, true, 'gslow A');
        await sleep(100);
        const qs = [mint(c, true, 'B'), mint(c, true, 'C'), mint(c, true, 'D')];
        await sleep(300);
        const svcBackends = (await superPool.query("SELECT count(*)::int AS n FROM pg_stat_activity WHERE usename = 'secuura_app' AND wait_event = 'advisory'")).rows[0].n;
        const v = await validate(o.key!);
        while (held.length) held.pop().release();
        const all = await Promise.all([pa, ...qs]);
        out.push({ t, vMs: v.ms, v: v.valid, vStatus: v.status, waitingOnAdvisory: svcBackends, st: all.map((x) => x.status), oneActive: activeIds(await rowsOf(c)).length });
      }
    } finally { while (held.length) held.pop().release(); await disarmCommit(); }
    results.CH6_pool = out;
    expect(out.filter((o) => o.v !== true || o.vMs > 800 || o.waitingOnAdvisory !== 0 || o.st.some((s: number) => s !== 201) || o.oneActive !== 1)).toEqual([]);
  });
