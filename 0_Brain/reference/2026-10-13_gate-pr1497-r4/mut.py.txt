def reds(t):

    for l in t.splitlines():
        m = re.match(r"\s+✕\s+(.*?)(?:\s+\(\d+ ms\))?$", l)
        if m: out.append(re.sub(r"^🔴\s*", "", m.group(1))[:58])
    return out
