def main(argv):
    want = set(argv)
    origs = {V: open(V, "rb").read(), W: open(W, "rb").read()}; h0 = {V: sha(V), W: sha(W)}
    print("baseline sha256/16", h0[V][:16], h0[W][:16])
    results = []
    for mid, target, fn, req in MUTS:
        if want and mid not in want: continue
        m = fn(origs[target].decode())
        if m is None: print(f"[{mid}] ANCHOR MISSING - mutant not applied (runner bug)"); results.append((mid, "ANCHOR-MISSING")); continue
        try:
            open(target, "w").write(m)
            tsc = run(["npx", "tsc", "--noEmit", "-p", "tsconfig.json"], cwd=O)
            if tsc.returncode != 0:
                print(f"[{mid}] DID NOT COMPILE - discarded (not a kill): " + (tsc.stdout.splitlines() or ["?"])[0][:160]); results.append((mid, "NO-COMPILE")); continue
            u = run(["npx", "jest", "--verbose"] + UNITS, cwd=O); open(f"{SP}/mut57_{mid}_unit.log", "w").write(u.stdout + u.stderr)
            layers = {"unit": (bool(re.search(r"^Tests:", u.stdout + u.stderr, re.M)), reds(f"{SP}/mut57_{mid}_unit.log"))}
            for mode, flag in (("F", "false"), ("T", "true")):
                r = run(["env", "WT=" + SP + "/wt1557", "TZ=UTC", "zsh", SP + "/integ5.sh", f"mut57_{mid}_{mode}", "app", flag, INTEG], cwd=None)
                layers[mode] = ("NO 'Tests:' LINE" not in r.stdout, reds(f"{SP}/integ_mut57_{mid}_{mode}.log"))
            verdict = []
            for layer, (has, rd) in layers.items():
                if not has: verdict.append(f"{layer}=LOADFAIL"); continue
                need = (req or {}).get(layer, [])
                hit = [n for n in need if any(n in r for r in rd)]
                if req is None: verdict.append(f"{layer}=reds:{len(rd)}")
                elif need and not hit: verdict.append(f"{layer}=SURVIVED(need one of {need}; reds={len(rd)})")
                else: verdict.append(f"{layer}=killed({len(rd)} red)" if need else f"{layer}=n/a({len(rd)} red)")
            print(f"[{mid}] " + " | ".join(verdict)); results.append((mid, " | ".join(verdict)))
        finally:
            open(V, "wb").write(origs[V]); open(W, "wb").write(origs[W])
            if sha(V) != h0[V] or sha(W) != h0[W]: print(f"[{mid}] !!! RESTORE NOT BYTE-IDENTICAL"); sys.exit(9)
    print("restored byte-identical after every mutant:", sha(V) == h0[V] and sha(W) == h0[W])
    bad = [r for r in results if "SURVIVED" in r[1] or "LOADFAIL" in r[1] or r[1] in ("ANCHOR-MISSING", "NO-COMPILE")]
    print("PROBLEMS:", bad if bad else "none")
main(sys.argv[1:])
