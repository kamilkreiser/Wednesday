def sha(p): return hashlib.sha256(open(p, 'rb').read()).hexdigest()
def reds(text):
    out = []
    for l in text.splitlines():
        m = re.match(r"\s+✕\s+(.*?)(?:\s+\(\d+ ms\))?$", l)
        if m: out.append(m.group(1)[:160])
    tl = re.findall(r"^Tests:.*$", text, re.M)
    fail_suite = 'Test suite failed to run' in text
    return {'reds': out, 'tests': (tl[-1] if tl else 'NO TESTS LINE'), 'suite_failed': fail_suite}
