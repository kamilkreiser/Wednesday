const CLEAN = 'Gate205-MARKER-Clean!Zv3$Rk9w';

const sha1 = (s: string) => createHash('sha1').update(s).digest('hex').toUpperCase();
const realFetch = globalThis.fetch;
const savedHibp = process.env.HIBP_ENABLED;
let server: Server;
let base = '';
let token = '';
const thrownMessages: string[] = [];

function hibpStub(url: string, init: { signal?: AbortSignal } | undefined): Promise<Response> | Response {
  ev.seq.push('hibp.fetch');
  const full = sha1(BREACHED);
  if (ev.mode === 'throw') {
    // hostile error text: carries the password, the full hash and the suffix; none may escape
    const e = new Error(`gate205 network down ${BREACHED} ${CLEAN} ${full} ${full.slice(5)} ${sha1(CLEAN)}`);
    thrownMessages.push(e.message);
    throw e;
  }
  if (ev.mode === 'non200') return new Response('unavailable', { status: 503 });
  if (ev.mode === 'non200_429') return new Response('slow down', { status: 429 });
  if (ev.mode === 'non200_500') return new Response('boom', { status: 500 });
  if (ev.mode === 'timeout') {
    return new Promise<Response>((_resolve, reject) => {
      const s = init?.signal;
      if (!s) return; // never resolves: would hang the cell, which the cell timeout reports
      s.addEventListener('abort', () => reject(s.reason ?? new Error('aborted')));
    });
  }
  const lines = ['00000000000000000000000000000000001:0', '0000000000000000000000000000000000F:2'];
  if (url.endsWith('/' + full.slice(0, 5))) lines.push(`${full.slice(5)}:1234`);
  if (ev.mode === 'badbody') return new Response('<<not a range body>>', { status: 200 });
  return new Response(lines.join('\r\n'), { status: 200 });
}

beforeAll(async () => {

  const origHash = await (await vi.importActual<typeof import('../services/password')>('../services/password')).hashPassword(CURRENT);
  ev.storedHash = origHash;

    if (url.startsWith('https://api.pwnedpasswords.com/')) { ev.calls.push({ url, init }); return hibpStub(url, init as never); }
    if (base && url.startsWith(base)) return realFetch(input as never, init as never);
    throw new Error(`gate205: unexpected outbound request to ${url}`);
  }) as typeof fetch;
  const mod = SIDE === 'base' ? await import('../routes/users.gatebase') : await import('../routes/users');

  app.use(express.json());

  await initJwtKeys();
  token = generateAccessToken({ id: USER_ID, email: 'g205@example.test', role: 'OWNER', verificationLevel: 'BASIC' } as never, 'gate205-session');
});

const n = (fn: unknown) => (fn as ReturnType<typeof vi.fn>).mock.calls.length;
async function change(currentPassword: string, newPassword: string) {
  const t0 = Date.now();
  const res = await realFetch(base + '/api/users/me/change-password', {
    method: 'POST', headers: { 'content-type': 'application/json', authorization: 'Bearer ' + token },
    body: JSON.stringify({ currentPassword, newPassword }),
  });
  const text = await res.text();
  return {
    status: res.status, text, ms: Date.now() - t0,
    lookups: ev.calls.length, hash: n(password.hashPassword), write: n(userRepo.updateUserOrThrow),
    revoke: n(session.revokeAllUserSessions), store: n(session.isRedisAvailable), seq: ev.seq.join('>'),
    urls: ev.calls.map((c) => c.url).join(','),
  };
}
function record(cell: string, r: Awaited<ReturnType<typeof change>>) {
  if (!OUT) return;
  let body = r.text;
  try { const j = JSON.parse(r.text); body = JSON.stringify(j); } catch { /* raw */ }
  fs.appendFileSync(OUT, [SIDE, cell, r.status, r.lookups, r.hash, r.write, r.revoke, r.store, r.seq, body].join('\t') + '\n');
}
function setSwitch(v: 'unset' | 'true' | 'false' | string) {
  if (v === 'unset') delete process.env.HIBP_ENABLED; else process.env.HIBP_ENABLED = v;
}

describe(`gate205 decision table (${SIDE})`, () => {
  for (const sw of ['unset', 'true', 'false'] as const) {
    for (const mode of ['ok', 'throw', 'non200', 'timeout'] as const) {
      for (const pw of ['breached', 'clean'] as const) {
        it(`cell sw=${sw} svc=${mode} pw=${pw}`, async () => {
          setSwitch(sw); ev.mode = mode;
          const r = await change(CURRENT, pw === 'breached' ? BREACHED : CLEAN);
          record(`sw=${sw}|svc=${mode}|pw=${pw}`, r);
          expect([200, 422]).toContain(r.status);
        }, 20_000);
      }
    }
  }
  // extra rows: other non-200s, an unreadable 200 body, other switch spellings
  for (const [sw, mode] of [['unset', 'non200_429'], ['unset', 'non200_500'], ['unset', 'badbody'], ['FALSE', 'ok'], ['0', 'ok'], ['off', 'ok']] as const) {
    it(`extra sw=${sw} svc=${mode} pw=breached`, async () => {
      setSwitch(sw); ev.mode = mode as never;
      const r = await change(CURRENT, BREACHED);
      record(`extra|sw=${sw}|svc=${mode}|pw=breached`, r);
      expect(r.status).toBeGreaterThan(0);
    }, 20_000);
  }
});

describe(`gate205 order and no-side-effect (${SIDE})`, () => {
  it('wrong current password + breached new: 400, no lookup', async () => {
    setSwitch('unset');
    const r = await change('Gate205-WRONG-current!9', BREACHED);
    record('order|wrong-current|breached', r);
    expect(r.status).toBe(400);
    expect(r.lookups).toBe(0);
    expect(r.write).toBe(0);
  });
  it('weak new (breached-listed weak string): 422 weak, no lookup', async () => {
    setSwitch('unset');
    const r = await change(CURRENT, 'password1');
    record('order|weak', r);
    expect(r.status).toBe(422);
    expect(r.text).toContain('New password is too weak');
    expect(r.lookups).toBe(0);
  });
  it('zod-short new password: 400, no lookup', async () => {
    setSwitch('unset');
    const r = await change(CURRENT, 'short');
    record('order|zod-short', r);
    expect(r.status).toBe(400);
    expect(r.lookups).toBe(0);
  });
  it('breached: refusal, and NO hash, NO store check, NO write, NO revoke (head) / sequence recorded', async () => {
    setSwitch('unset');
    const r = await change(CURRENT, BREACHED);
    record('order|breached', r);
    if (SIDE === 'head') {
      expect(r.status).toBe(422);
      expect(r.seq).toBe('repo.getUser>pw.verify>hibp.fetch');
      expect([r.hash, r.store, r.write, r.revoke]).toEqual([0, 0, 0, 0]);
      const j = JSON.parse(r.text);
      expect(j).toEqual({ success: false, error: { code: 'VALIDATION_ERROR', message: MSG, details: [{ code: 'custom', path: ['newPassword'], message: MSG }] } });
    } else {
      expect(r.status).toBe(200);
    }
  });
  it('clean: the lookup sits between verify and hash; write and revoke follow', async () => {
    setSwitch('unset');
    const r = await change(CURRENT, CLEAN);
    record('order|clean', r);
    expect(r.status).toBe(200);
    if (SIDE === 'head') expect(r.seq).toBe('repo.getUser>pw.verify>hibp.fetch>pw.hash>session.isRedisAvailable>repo.write>session.revokeAll>session.isRedisAvailable');
  });
});

describe(`gate205 privacy (${SIDE})`, () => {
  it('only the 5-char prefix leaves; no marker, full hash or suffix in any url, init, body, log line or thrown text', async () => {
    const outputs: string[] = [];
    for (const mode of ['ok', 'throw', 'non200', 'timeout', 'badbody'] as const) {
      for (const pw of [BREACHED, CLEAN]) {
        setSwitch('unset'); ev.mode = mode; ev.calls = []; ev.logs.length = 0;
        const r = await change(CURRENT, pw);
        const sent = ev.calls.map((c) => c.url + ' ' + JSON.stringify(c.init ?? null, (_k, v) => (v instanceof AbortSignal ? '[signal]' : v)));
        for (const c of ev.calls) expect(c.url).toBe('https://api.pwnedpasswords.com/range/' + sha1(pw).slice(0, 5));
        const logs = ev.logs.join('\n');
        outputs.push(`--- mode=${mode} pw=${pw === BREACHED ? 'BREACHED' : 'CLEAN'} status=${r.status}\nSENT ${sent.join(' | ')}\nBODY ${r.text}\nLOGS ${logs}`);
        const full = sha1(pw);
        for (const leak of [pw, full, full.slice(5), full.toLowerCase(), full.slice(5).toLowerCase(), CURRENT, sha1(CURRENT)]) {
          for (const hay of [...sent, r.text, logs]) expect(hay.includes(leak), `${leak.slice(0, 12)}… found in output (mode=${mode})`).toBe(false);
        }
      }
    }
    if (OUT) fs.writeFileSync(OUT.replace(/\.tsv$/, '') + `.privacy_${SIDE}.txt`, outputs.join('\n') + `\nTHROWN_PLANTED ${thrownMessages.length}\n`);
    expect(thrownMessages.length).toBeGreaterThan(0); // the hostile throw text really was produced (control)
  }, 60_000);
  it('control: the capture transport sees an ordinary log line', async () => {
    ev.logs.length = 0;
    authLogger.warn('gate205 capture control');
    await new Promise((r) => setTimeout(r, 20));
    expect(ev.logs.join('\n')).toContain('gate205 capture control');
  });
});
