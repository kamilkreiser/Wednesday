import { describe, it, expect, vi, afterEach } from 'vitest';
import { enableWallet } from '../cardano/wallet';

// KS-1868: enableWallet (src/cardano/wallet.ts) wraps a wallet's `enable()` failure as
// `Failed to enable wallet <name>: <reason>` where the reason is read as `error instanceof Error ? error.message : 'User rejected'`,
// unguarded. When the thrown value cannot be read (a revoked Proxy throws on `instanceof`; an Error whose `message` getter throws)
// the catch itself throws, so the call rejects with the engine error instead of the documented Error. The wallet is a plain
// stand-in on a stubbed `window` whose `enable` rejects with the cell's value; no browser, no network.
async function failure(value: unknown): Promise<string> {
  vi.stubGlobal('window', { cardano: { nami: { enable: () => Promise.reject(value) } } });
  try {
    await enableWallet('nami' as never);
    return 'resolved';
  } catch (e) {
    return (e as Error).message;
  }
}

const revokedProxy = (): unknown => {
  const { proxy, revoke } = Proxy.revocable({}, {});
  revoke();
  return proxy;
};

const getterThrows = (): unknown =>
  Object.create(Error.prototype, {
    message: {
      get() {
        throw new Error('getter boom');
      },
    },
  });

describe('KS-1868 enableWallet reads a wallet failure safely', () => {
  afterEach(() => {
    vi.unstubAllGlobals();
  });

  it('control ks1868 C1 an ordinary Error is wrapped with its message', async () => {
    expect(await failure(new Error('user declined'))).toBe('Failed to enable wallet nami: user declined');
  });

  it('control ks1868 C2 a thrown plain string is wrapped with the generic text', async () => {
    expect(await failure('plain text failure')).toBe('Failed to enable wallet nami: User rejected');
  });

  it('RED ks1868 R1 a revoked Proxy is wrapped with a constant reason', async () => {
    expect(await failure(revokedProxy())).toBe('Failed to enable wallet nami: unreadable thrown value');
  });

  it('RED ks1868 R2 an Error whose message getter throws is wrapped with a constant reason', async () => {
    expect(await failure(getterThrows())).toBe('Failed to enable wallet nami: unreadable thrown value');
  });
});
