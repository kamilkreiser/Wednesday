import { describe, it, expect, vi } from 'vitest';

// KS-1868: safeOutboundRequest (src/security/ssrf-guard.ts) promises to RETURN the guard's error rather than throw it; when
// the synchronous `https.request(...)` call throws, its catch answers `{ ok: false, reason: 'request_failed', error: (err as Error).message }`.
// The `.message` read is unguarded. When the value cannot be read (a revoked Proxy throws on any property read; an Error whose
// `message` getter throws) the catch itself throws out of the Promise executor, so the call REJECTS where the contract says
// it returns a tagged failure. The URL is a TEST-NET-3 IP literal (203.0.113.7, no DNS); `https.request` is a plain function
// that throws the cell's value; no network.
const h = vi.hoisted(() => ({ toThrow: undefined as unknown }));

vi.mock('https', async (importOriginal) => {
  const actual = await importOriginal<typeof import('https')>();
  return {
    ...actual,
    default: actual,
    request: () => {
      throw h.toThrow;
    },
  };
});

const { safeOutboundRequest } = await import('../security/ssrf-guard');

type Outcome = { rejected: boolean; result: unknown };

async function run(value: unknown): Promise<Outcome> {
  h.toThrow = value;
  try {
    const result = await safeOutboundRequest('https://203.0.113.7/', { timeoutMs: 2000 });
    return { rejected: false, result };
  } catch {
    return { rejected: true, result: undefined };
  }
}

const revokedProxy = (): unknown => {
  const { proxy, revoke } = Proxy.revocable({}, {});
  revoke();
  return proxy;
};

const getterThrows = (): unknown =>
  Object.create(Error.prototype, {
    message: {
      get() {
        throw new Error('getter boom');
      },
    },
  });

describe('KS-1868 safeOutboundRequest reads a request construction failure safely', () => {
  it('control ks1868 C1 an ordinary Error is returned as a request_failed result with its message', async () => {
    const out = await run(new Error('bad header'));
    expect(out.rejected).toBe(false);
    expect(out.result).toEqual({ ok: false, reason: 'request_failed', error: 'bad header' });
  });

  it('control ks1868 C2 a thrown plain string is returned as a request_failed result without a message', async () => {
    const out = await run('plain text failure');
    expect(out.rejected).toBe(false);
    expect(out.result).toEqual({ ok: false, reason: 'request_failed', error: undefined });
  });

  it('RED ks1868 R1 a revoked Proxy is returned as a request_failed result with a constant reason', async () => {
    const out = await run(revokedProxy());
    expect(out.rejected).toBe(false);
    expect(out.result).toEqual({ ok: false, reason: 'request_failed', error: 'unreadable thrown value' });
  });

  it('RED ks1868 R2 an Error whose message getter throws is returned as a request_failed result with a constant reason', async () => {
    const out = await run(getterThrows());
    expect(out.rejected).toBe(false);
    expect(out.result).toEqual({ ok: false, reason: 'request_failed', error: 'unreadable thrown value' });
  });
});
