def one(s, a, b, n=1):
    assert s.count(a) == n, (a[:60], s.count(a))
    return s.replace(a, b)

    'CTL':  lambda s: one(s, "// 3b. The guard again (m-1)", "// 3b. The guard again (m-1) [gate r6 comment-only control]"),
    'N1':   lambda s: one(s, RESOLVER_REFUSAL, ''),

    'N4':   lambda s: one(s, PRED, "return connectorId.trim() === '';"),

def run(cmd):
    p = subprocess.run(cmd, shell=True, capture_output=True, text=True, executable='/bin/zsh')
    return p.returncode, p.stdout + p.stderr

def reds(o):
    return [l.strip()[2:90] for l in o.splitlines() if l.strip().startswith('✕')]

    names = sys.argv[1].split(',')
    only = os.environ.get('GATE_ONLY_ATTACK', 'SETUP,K1,V1,D1,N4,R3,R2,R4')
    for name in names:
        s = MUTANTS[name](ORIG)
        assert s != ORIG, name + ': no change'
        open(SVC, 'w').write(s)
        try:
            out = [f'== {name} ==']
            rc, o = run(f'builtin cd {OD} && npx tsc --noEmit'); out.append(f'  tsc rc={rc}')

            sha = hashlib.sha256(open(SVC).read().encode()).hexdigest()

            print(f'   restored sha256={sha[:12]} ok', flush=True)

if __name__ == '__main__':
