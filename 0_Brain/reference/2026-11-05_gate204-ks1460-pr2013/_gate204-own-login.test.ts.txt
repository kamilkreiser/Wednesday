// KS-1460 row 1 (Wednesday's ruling of 2026-11-05: "make the 'rehashed' info line truthful in the same PR, log only,
// no behaviour change, one cell"). POST /api/auth/login's UPGRADE-ON-VERIFY step rehashes a bcrypt hash to Argon2id
// with `userRepo.updateUser`, which answers `null` when its UPDATE matched no row. The route discarded that answer
// and logged "UPGRADE-ON-VERIFY: rehashed bcrypt -> argon2id" whether or not the write landed, so the log recorded a
// rehash that may not have happened. It now logs the info line only when the write landed, and a warn naming the
// write that did not land otherwise. The step stays non-fatal: the login answers 200 with tokens either way.
// Harness: the ks1868 rehash test's (real router and error handler; repository, password, session, jwt, lockout,
// mail and event bus mocked), with the logger's info AND warn captured.

}));

const urls: Record<string, string> = {};
const servers: Server[] = [];
beforeAll(async () => {
  const { errorHandler } = await import('../middleware/errorHandler');
  for (const [k, p] of [['base', '../routes/auth.gatebase'], ['head', '../routes/auth']] as const) {
    const mod = await import(p);
    const app = express(); app.use(express.json()); app.use('/api/auth', mod.authRoutes); app.use(errorHandler);
    const s = app.listen(0, '127.0.0.1'); await new Promise<void>((r) => s.once('listening', () => r())); servers.push(s);
    urls[k] = 'http://127.0.0.1:' + String((s.address() as AddressInfo).port);
  }
}, 120000);
afterAll(async () => { for (const s of servers) await new Promise<void>((r) => s.close(() => r())); });

const PW = 'PLANT_PASSWORD_G204';
async function run(side: 'base' | 'head', mode: typeof h.mode) {
  h.mode = mode; h.info = []; h.warn = []; h.writes = []; h.sessions = 0;
  const res = await fetch(urls[side] + '/api/auth/login', { method: 'POST', headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ email: LOGIN_EMAIL, password: PW }) });
  const body = (await res.text()).replace(/"requestId":"[^"]*"|"timestamp":"[^"]*"/g, '');
  const hdr = [...res.headers.entries()].filter(([k]) => !['date', 'keep-alive', 'connection'].includes(k));
  return { status: res.status, body, hdr, writes: JSON.stringify(h.writes), sessions: h.sessions,
    rehashLog: [...h.info, ...h.warn].filter((c) => String(c[0]).includes('UPGRADE-ON-VERIFY') || String(c[1]).includes('UPGRADE-ON-VERIFY')),
    allLogs: JSON.stringify([...h.info, ...h.warn]) };
}
const out: string[] = [];
afterAll(async () => { const fs = await import('node:fs'); fs.writeFileSync(process.env.G204_LOGIN_OUT || '/dev/null', out.join('\n') + '\n'); });
for (const mode of ['lands', 'zero', 'throw', 'throwToken'] as const) {
  it('login base vs head, rehash ' + mode, async () => {
    const b = await run('base', mode);
    const hd = await run('head', mode);
    out.push(mode + '\tBASE ' + b.status + ' ' + JSON.stringify(b.rehashLog) + '\n\tHEAD ' + hd.status + ' ' + JSON.stringify(hd.rehashLog)
      + '\n\tbody-equal=' + (b.body === hd.body) + ' hdr-equal=' + (JSON.stringify(b.hdr) === JSON.stringify(hd.hdr))
      + ' writes-equal=' + (b.writes === hd.writes) + ' sessions ' + b.sessions + '/' + hd.sessions
      + '\n\thead-log-plants: ' + ['PLANT_PASSWORD', 'PLANT_OLDHASH', 'PLANT_NEWHASH', 'PLANT_TOKEN', 'MINTED_1460_LOGIN'].map((m) => m + '=' + hd.allLogs.includes(m)).join(' ')
      + '\n\tbase-log-plants: ' + ['PLANT_PASSWORD', 'PLANT_OLDHASH', 'PLANT_NEWHASH', 'PLANT_TOKEN', 'MINTED_1460_LOGIN'].map((m) => m + '=' + b.allLogs.includes(m)).join(' '));
    expect(hd.status).toBe(b.status);
    expect(hd.body).toBe(b.body);
    expect(hd.hdr).toEqual(b.hdr);
    expect(hd.writes).toBe(b.writes);
    expect(hd.sessions).toBe(b.sessions);
    if (mode === 'zero') {
      const warnLine = h.warn.find((c) => String(c[0]).includes('did not take effect'));
      expect(JSON.stringify(warnLine)).not.toMatch(/PLANT_|MINTED/);
    }
  });
}
