type Obs = { status: number; body: string; cookie: string | null; sessions: number; minted: number; levelUpdates: number; levelAfter: string };
async function run(side: 'base' | 'head', route: 'callback' | 'link', mode: typeof st.mode, level = 'basic'): Promise<Obs> {
  st.level = level; st.mode = mode; st.updates = []; st.sessions = 0; st.minted = 0; st.logs = [];
  let res: Response;
  if (route === 'callback') {
    const { storeOAuthState } = await import('../services/oauthState');
    const s = `g204-${randomUUID()}`; await storeOAuthState(s, 'google');
    res = await fetch(`${urls[side]}/api/auth/social/google/callback?code=c&state=${encodeURIComponent(s)}`, { redirect: 'manual' });
  } else {
    res = await fetch(`${urls[side]}/api/auth/social/link`, { method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ provider: 'google', code: 'c' }) });
  }
  const body = await res.text();
  return { status: res.status, body: body.replace(/"requestId":"[^"]*"|"timestamp":"[^"]*"/g, ''), cookie: res.headers.get('set-cookie'),
    sessions: st.sessions, minted: st.minted, levelUpdates: st.updates.filter((u) => u.sql.includes('verification_level')).length, levelAfter: st.level };
}

const table: string[] = [];
afterAll(async () => { const fs = await import('node:fs'); fs.writeFileSync(process.env.G204_OUT || '/dev/null', table.join('\n') + '\n'); });

for (const route of ['callback', 'link'] as const) {
  for (const level of ['basic', 'none', 'social', 'standard', 'high']) {
    for (const mode of ['lands', 'zero', 'throw', 'infra', 'blind'] as const) {
      it(`${route} level=${level} mode=${mode}`, async () => {
        const b = await run('base', route, mode, level);
        const h = await run('head', route, mode, level);
        table.push(`${route}\t${level}\t${mode}\tBASE ${b.status} sess=${b.sessions} mint=${b.minted} cookie=${b.cookie ? 'Y' : 'N'} upd=${b.levelUpdates} body=${b.body.slice(0, 90)}\tHEAD ${h.status} sess=${h.sessions} mint=${h.minted} cookie=${h.cookie ? 'Y' : 'N'} upd=${h.levelUpdates} body=${h.body.slice(0, 140)}`);
        const raises = level === 'basic' || level === 'none';
        // Q3 direction: head never issues what base did not
        expect(h.sessions).toBeLessThanOrEqual(b.sessions);
        expect(h.minted).toBeLessThanOrEqual(b.minted);
        if (!raises || mode === 'lands') {
          expect(h).toEqual(b); // identical to base
        } else if (mode === 'zero') {
          expect(h.status).toBe(503);
          expect(h.body).toContain('SERVICE_UNAVAILABLE');
          expect(h.body).not.toContain('"success":true');
          expect(h.body).not.toContain('G204_');
          expect(h.cookie).toBeNull();
          expect(h.sessions).toBe(0); expect(h.minted).toBe(0);
        }
      });
    }
  }
}
