import { describe, it, expect, vi, beforeEach } from 'vitest';

// KS-1868: `getWalletNFTCertificates` must still log its line and resolves an empty list when the thrown value
// itself cannot be read (a revoked Proxy throws on `instanceof`; an Error whose `message` getter throws,
// throws on `.message`).
// The db stand-in is a plain function, not a vi.fn(): a spy inspects what it rejects with and
// could itself fail on a revoked Proxy before the code under test saw it.
const h = vi.hoisted(() => ({
  toThrow: undefined as unknown,
  warn: vi.fn(),
  error: vi.fn(),
}));

vi.mock('../db', () => ({
  isDbAvailable: () => true,
  query: () => Promise.reject(h.toThrow),
}));

vi.mock('../cardano', () => ({
  isCardanoNFTReady: () => false,
  mintNFTOnChain: vi.fn(),
  getPolicyId: () => null,
  waitForNFTConfirmation: vi.fn(),
}));

vi.mock('../utils/logger', () => ({
  logger: { info: vi.fn(), warn: h.warn, error: h.error, debug: vi.fn() },
}));

import { nftMintingService } from '../services/minting.service';

const call = () => nftMintingService.getWalletNFTCertificates('ks1868-wallet');

const LINES = () => h.error.mock.calls.filter((c) => String(c[0]).includes('DB get wallet certificates failed'));

describe('KS-1868 `getWalletNFTCertificates` reads the thrown value safely', () => {
  beforeEach(() => {
    h.toThrow = undefined;
    h.warn.mockReset();
    h.error.mockReset();
  });

  it('control ks1868 C1 an ordinary Error still logs its message and resolves an empty list', async () => {
    h.toThrow = new Error('db down');
    await expect(call()).resolves.toEqual([]);
    expect(LINES()).toHaveLength(1);
    expect(LINES()[0][1]).toEqual({ error: 'db down' });
  });

  it('RED ks1868 R1 a revoked Proxy thrown value logs a constant reason and resolves an empty list', async () => {
    const { proxy, revoke } = Proxy.revocable({}, {});
    revoke();
    h.toThrow = proxy;
    await call().catch(() => undefined);
    expect(LINES()).toHaveLength(1);
    expect(LINES()[0][1]).toEqual({ error: 'unreadable thrown value' });
  });

  it('RED ks1868 R2 an Error whose message getter throws logs a constant reason and resolves an empty list', async () => {
    h.toThrow = Object.create(Error.prototype, {
      message: {
        get() {
          throw new Error('getter boom');
        },
      },
    });
    await call().catch(() => undefined);
    expect(LINES()).toHaveLength(1);
    expect(LINES()[0][1]).toEqual({ error: 'unreadable thrown value' });
  });
});
