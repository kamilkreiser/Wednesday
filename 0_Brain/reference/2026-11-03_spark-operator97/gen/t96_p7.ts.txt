import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest';

// KS-1868: SecretsManager.getSecret (src/security/secrets-manager.ts) tries Vault first; when the Vault read fails its
// catch logs a warning with `error instanceof Error ? error.message : String(error)` and falls through to the environment
// fallback. The read is unguarded. When the value cannot be read (a revoked Proxy throws on `instanceof`; an Error whose
// `message` getter throws, throws on `.message`) the catch itself throws, so getSecret REJECTS and the environment fallback
// is never tried. The global `fetch` is a plain function rejecting with the cell's value (stubbed with `vi.stubGlobal`,
// removed after each cell); the logger is a plain recorder; no network.
const h = vi.hoisted(() => ({
  toThrow: undefined as unknown,
  errors: [] as unknown[][],
  warns: [] as unknown[][],
}));

vi.mock('../logger', () => ({
  logger: {
    debug: () => undefined,
    info: () => undefined,
    warn: (...args: unknown[]) => {
      h.warns.push(args);
    },
    error: (...args: unknown[]) => {
      h.errors.push(args);
    },
  },
}));

import { SecretsManager } from '../security/secrets-manager';

const ENV_KEY = 'KS1868_VAULT_SECRET';

/** The getSecret result, or 'rejected' when it rejects (an assertion can then fail on it). */
async function getIt(): Promise<unknown> {
  vi.stubGlobal('fetch', () => Promise.reject(h.toThrow));
  const sm = new SecretsManager({
    vault: { address: 'http://vault.invalid', token: 't', engine: 'secret', basePath: 'p' },
    useEnvironment: true,
  });
  try {
    return await sm.getSecret('ks1868.vault.secret');
  } catch {
    return 'rejected';
  }
}

const logged = (): unknown[][] => h.warns.filter((a) => String(a[0]).includes('Vault read failed'));

const revokedProxy = (): unknown => {
  const { proxy, revoke } = Proxy.revocable({}, {});
  revoke();
  return proxy;
};

const getterThrows = (): unknown =>
  Object.create(Error.prototype, {
    message: {
      get() {
        throw new Error('getter boom');
      },
    },
  });

describe('KS-1868 SecretsManager.getSecret reads a Vault failure safely', () => {
  beforeEach(() => {
    h.toThrow = undefined;
    h.errors = [];
    h.warns = [];
    process.env[ENV_KEY] = 'env-value';
  });

  afterEach(() => {
    vi.unstubAllGlobals();
    delete process.env[ENV_KEY];
  });

  it('control ks1868 C1 an ordinary Error logs its message and falls back to the environment', async () => {
    h.toThrow = new Error('vault down');
    expect(await getIt()).toBe('env-value');
    expect(logged()[0][1]).toEqual({ key: 'ks1868.vault.secret', error: 'vault down' });
  });

  it('control ks1868 C2 a thrown plain string logs that text and falls back to the environment', async () => {
    h.toThrow = 'plain text failure';
    expect(await getIt()).toBe('env-value');
    expect(logged()[0][1]).toEqual({ key: 'ks1868.vault.secret', error: 'plain text failure' });
  });

  it('RED ks1868 R1 a revoked Proxy logs a constant reason and falls back to the environment', async () => {
    h.toThrow = revokedProxy();
    expect(await getIt()).toBe('env-value');
    expect(logged()[0][1]).toEqual({ key: 'ks1868.vault.secret', error: 'unreadable thrown value' });
  });

  it('RED ks1868 R2 an Error whose message getter throws logs a constant reason and falls back to the environment', async () => {
    h.toThrow = getterThrows();
    expect(await getIt()).toBe('env-value');
    expect(logged()[0][1]).toEqual({ key: 'ks1868.vault.secret', error: 'unreadable thrown value' });
  });
});
