import { describe, it, expect, beforeEach } from 'vitest';
import { vi } from 'vitest';

// KS-1868 (operator 88 screen, auth event bus): the Redis 'error' listener (line 95) of services/auth/src/events.ts passes
// `err.message` straight into the logger's meta object, unguarded. A thrown value whose read throws (a revoked Proxy, an
// Error whose message getter throws) makes the listener throw: an exception thrown by a listener of the client's 'error'
// event escapes the EventEmitter and is an uncaught exception; an Error whose message is not a string (a Symbol) is passed
// to the logger uncoerced. The cells emit such a value to the captured listener and assert the listener returns and the
// logged value is text (the constant when the value cannot be turned into text). Harness: the real module with ioredis
// replaced by a plain stand-in class (it hands over the 'error' listener) and the logger by a plain recording object.
const h = vi.hoisted(() => ({
  listener: undefined as undefined | ((e: unknown) => void),
  ctorFailure: undefined as undefined | { value: unknown },
  logs: [] as { level: string; message: string; meta: Record<string, unknown> | undefined }[],
}));

vi.mock('ioredis', () => ({
  default: function RedisMock() {
    if (h.ctorFailure) throw h.ctorFailure.value;
    return {
      status: 'wait',
      on: (event: string, fn: (e: unknown) => void) => {
        if (event === 'error') h.listener = fn;
      },
      connect: async () => undefined,
      disconnect: () => undefined,
      xadd: async () => '1-0',
    };
  },
}));
vi.mock('../utils/logger', () => {
  const rec = (level: string) => (message: string, meta?: Record<string, unknown>) => {
    h.logs.push({ level, message, meta });
  };
  return { createLogger: () => ({ debug: rec('debug'), info: rec('info'), warn: rec('warn'), error: rec('error') }) };
});

type Ev = typeof import('../events');
beforeEach(() => {
  h.listener = undefined;
  h.ctorFailure = undefined;
  h.logs.length = 0;
});

async function freshEvents(): Promise<Ev> {
  vi.resetModules();
  return import('../events');
}
async function emit(value: unknown): Promise<{ threw: boolean }> {
  const ev = await freshEvents();
  ev.initEventBus('redis://localhost:6379');
  try {
    (h.listener as (e: unknown) => void)(value);
    return { threw: false };
  } catch {
    return { threw: true };
  }
}
const LINE = (message: string): { level: string; message: string; meta: Record<string, unknown> | undefined }[] =>
  h.logs.filter((l) => l.message === message);
const LSN = 'EventBus Redis error (non-fatal)';
/** The logged `error` field as a string; the type name when it is not one (a hostile value never reaches an assertion). */
const reasonOf = (l: { meta: Record<string, unknown> | undefined } | undefined): string =>
  typeof l?.meta?.error === 'string' ? (l.meta.error as string) : '<' + typeof l?.meta?.error + '>';

const revokedProxy = (): unknown => {
  const rp = Proxy.revocable({}, {});
  rp.revoke();
  return rp.proxy;
};
/** An Error whose `message` getter succeeds and returns the value m. */
const withMessage = (m: unknown): unknown => Object.create(Error.prototype, { message: { get() { return m; } } });
const getterThrows = (): unknown =>
  Object.create(Error.prototype, {
    message: {
      get() {
        throw new Error('getter boom');
      },
    },
  });

describe('KS-1868 the event bus Redis error listener survives and logs text for a hostile thrown value', () => {
  it('control ks1868 C1: an ordinary Error reaching the listener does not throw and is logged with its message', async () => {
    expect(await emit(new Error('listener boom'))).toEqual({ threw: false });
    expect(LINE(LSN)).toHaveLength(1);
    expect(reasonOf(LINE(LSN)[0])).toBe('listener boom');
  });
  it('control ks1868 C2: a plain string reaching the listener does not throw and is logged without a message', async () => {
    expect(await emit('plain boom')).toEqual({ threw: false });
    expect(LINE(LSN)).toHaveLength(1);
    expect(reasonOf(LINE(LSN)[0])).toBe('<undefined>');
  });
  it('RED ks1868 R1: a revoked Proxy reaching the listener does not throw and is logged as text, as for an ordinary Error', async () => {
    expect(await emit(revokedProxy())).toEqual({ threw: false });
    expect(LINE(LSN)).toHaveLength(1);
    expect(reasonOf(LINE(LSN)[0])).toBe('unreadable thrown value');
  });
  it('RED ks1868 R2: an Error whose message getter throws reaching the listener does not throw and is logged as text, as for an ordinary Error', async () => {
    expect(await emit(getterThrows())).toEqual({ threw: false });
    expect(LINE(LSN)).toHaveLength(1);
    expect(reasonOf(LINE(LSN)[0])).toBe('unreadable thrown value');
  });
  it('RED ks1868 R3: an Error whose message is a Symbol reaching the listener does not throw and is logged as text, as for an ordinary Error', async () => {
    expect(await emit(withMessage(Symbol('ks1868')))).toEqual({ threw: false });
    expect(LINE(LSN)).toHaveLength(1);
    expect(reasonOf(LINE(LSN)[0])).toBe('Symbol(ks1868)');
  });
});
