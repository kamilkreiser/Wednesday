// Gate 253 probe (QA only, never committed). FAIL condition stated first: for a served, refused POST on an /api route,
// rows-for-429 != refused count. Composition in index.ts's order: real versioningMiddleware, the /api/v1 rewrite copied
// verbatim from index.ts:550-553, real createAuditMiddleware, a machine principal, real enforceClientRateLimit,
// then a case-insensitive express router mounted at /api/documents (express default; index.ts:548 says routing is case insensitive).
import { describe, it, expect } from 'vitest';
import express from 'express';
import type { Server } from 'node:http';
import type { AddressInfo } from 'node:net';
import { createAuditMiddleware } from '../middleware/audit';
import { enforceClientRateLimit } from '../middleware/rateLimitEnforce';
import { versioningMiddleware } from '../routes/versioning';
import { foldAsciiCase } from '../middleware/pathFold';

const fakeRedis = () => { const c = new Map<string, number>(); return { status: 'ready', incr: async (k: string) => { const n = (c.get(k) ?? 0) + 1; c.set(k, n); return n; }, pexpire: async () => 1, pttl: async () => 60_000 }; };

async function boot() {
  const rows: { params: unknown[] }[] = [];
  const app = express();
  app.use(versioningMiddleware());
  app.use((req, _res, next) => { if (foldAsciiCase(req.path).startsWith('/api/v1/')) req.url = req.url.replace(/\/api\/v1\//i, '/api/'); next(); });
  app.use(createAuditMiddleware({ query: async (_s, params) => { rows.push({ params }); return { rows: [] }; }, isDbAvailable: () => true, log: () => {} }));
  app.use((req, _res, next) => { (req as any).user = { authMethod: 'api_key', userId: 'connector:k', organizationId: 'c1205000-0000-4000-8000-0000000000c1', tenantId: 'b1205000-0000-4000-8000-0000000000a1', rateLimit: 1, rateLimitWindow: 60 }; next(); });
  app.use(enforceClientRateLimit(fakeRedis() as any));
  const r = express.Router();
  r.all('/', (_q, s) => { s.status(200).json({ ok: true }); });
  app.use('/api/documents', r);
  const server: Server = await new Promise((res) => { const s = app.listen(0, '127.0.0.1', () => res(s)); });
  const base = `http://127.0.0.1:${(server.address() as AddressInfo).port}`;
  const send = async (m: string, p: string) => (await fetch(base + p, { method: m, redirect: 'manual', headers: { 'content-type': 'application/json' }, body: '{}' })).status;
  const settle = async () => { await new Promise((x) => setTimeout(x, 400)); return rows.length; };
  return { rows, send, settle, close: () => new Promise<void>((x) => server.close(() => x())) };
}

describe('g253 probe: path spelling', () => {
  for (const p of ['/api/documents', '/API/documents', '/Api/documents', '/API/v1/documents', '/api/documents/']) {
    it(`NODE_ENV=${process.env.NODE_ENV} POST ${p}`, async () => {
      const g = await boot();
      try {
        const st = [await g.send('POST', p), await g.send('POST', p)];
        const n = await g.settle();
        // eslint-disable-next-line no-console
        console.log(`G253 env=${process.env.NODE_ENV} path=${p} statuses=${st.join(',')} rows=${n} rows429=${g.rows.filter((r) => JSON.parse(String(r.params[8])).status === 429).length}`);
        expect(st).toBeDefined();
      } finally { await g.close(); }
    }, 30_000);
  }
});
