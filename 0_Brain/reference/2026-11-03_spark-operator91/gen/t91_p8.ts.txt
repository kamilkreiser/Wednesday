import { describe, it, expect, beforeEach, afterEach, vi } from 'vitest';

// KS-1868 (operator 91 screen, nft-certificate cardano module): the initialisation catch of initCardanoNFT (line 58) of
// services/nft-certificate/src/cardano/index.ts interpolates `err?.message` into the console.error template, unguarded.
// A thrown value whose read throws (a revoked Proxy, an Error whose message getter throws) makes the catch itself throw:
// initCardanoNFT REJECTS to its boot caller (server.ts, `await initCardanoNFT(...)`) instead of logging and answering false
// (its contract: "Returns true if real minting is available", mock mode otherwise); an Error whose message is readable but not a
// string is interpolated by the template literal, which throws for a Symbol. The cells make the Blockfrost provider
// initialisation throw such a value and assert initCardanoNFT resolves false and the logged line carries text (the constant when
// the value cannot be turned into text). Harness: the real module with ./provider, ./wallet, ./mintPolicy and ./mintTransaction
// replaced by plain stand-ins and console.error / console.log by plain recording functions (no Blockfrost, no wallet).
const h = vi.hoisted(() => ({
  toThrow: undefined as unknown,
  errors: [] as unknown[][],
}));

vi.mock('../cardano/provider', () => ({
  initProvider: () => {
    throw h.toThrow;
  },
  fetchProtocolParams: async () => ({}),
  submitTransaction: async () => 'hash',
  getTransaction: async () => null,
}));
vi.mock('../cardano/wallet', () => ({ initWallet: () => ({ address: 'addr_test1stub' }) }));
vi.mock('../cardano/mintPolicy', () => ({ createSignaturePolicy: () => ({ policyId: 'policy' }) }));
vi.mock('../cardano/mintTransaction', () => ({ buildMintTransaction: async () => ({}) }));

import { initCardanoNFT } from '../cardano';

const realError = console.error;
const realLog = console.log;
const realKey = process.env.BLOCKFROST_API_KEY;
const realMnemonic = process.env.PLATFORM_WALLET_MNEMONIC;

beforeEach(() => {
  h.toThrow = undefined;
  h.errors.length = 0;
  process.env.BLOCKFROST_API_KEY = 'preprodStubKey';
  process.env.PLATFORM_WALLET_MNEMONIC = 'stub mnemonic words';
  console.error = (...args: unknown[]) => {
    h.errors.push(args);
  };
  console.log = () => undefined;
});
afterEach(() => {
  console.error = realError;
  console.log = realLog;
  if (realKey === undefined) delete process.env.BLOCKFROST_API_KEY;
  else process.env.BLOCKFROST_API_KEY = realKey;
  if (realMnemonic === undefined) delete process.env.PLATFORM_WALLET_MNEMONIC;
  else process.env.PLATFORM_WALLET_MNEMONIC = realMnemonic;
});

async function run(value: unknown): Promise<{ ok: boolean; value: boolean | undefined }> {
  h.toThrow = value;
  try {
    return { ok: true, value: await initCardanoNFT('preprod') };
  } catch {
    return { ok: false, value: undefined };
  }
}
const LINES = () => h.errors.filter((a) => typeof a[0] === 'string' && (a[0] as string).startsWith('[Cardano-NFT] Initialisation failed: '));
/** The text after the prefix of the logged line. */
const reasonOf = (a: unknown[] | undefined): string => (a ? (a[0] as string).slice('[Cardano-NFT] Initialisation failed: '.length) : '<none>');

const revokedProxy = (): unknown => {
  const rp = Proxy.revocable({}, {});
  rp.revoke();
  return rp.proxy;
};
/** An Error whose `message` getter succeeds and returns the value m. */
const withMessage = (m: unknown): unknown => Object.create(Error.prototype, { message: { get() { return m; } } });
const getterThrows = (): unknown =>
  Object.create(Error.prototype, {
    message: {
      get() {
        throw new Error('getter boom');
      },
    },
  });

describe('KS-1868 the initCardanoNFT catch answers false and logs text for a hostile thrown value', () => {
  it('control ks1868 C1: an ordinary Error from the provider init is logged with its message and initCardanoNFT answers false', async () => {
    expect(await run(new Error('provider boom'))).toEqual({ ok: true, value: false });
    expect(LINES()).toHaveLength(1);
    expect(reasonOf(LINES()[0])).toBe('provider boom');
  });
  it('control ks1868 C2: a thrown plain string is logged as undefined text and initCardanoNFT answers false', async () => {
    expect(await run('plain boom')).toEqual({ ok: true, value: false });
    expect(LINES()).toHaveLength(1);
    expect(reasonOf(LINES()[0])).toBe('undefined');
  });
  it('control ks1868 C3: an Error with an empty message is logged with the empty text and initCardanoNFT answers false', async () => {
    expect(await run(new Error(''))).toEqual({ ok: true, value: false });
    expect(LINES()).toHaveLength(1);
    expect(reasonOf(LINES()[0])).toBe('');
  });
  it('RED ks1868 R1: a provider init failure that is a revoked Proxy is logged as text and initCardanoNFT answers false, as for an ordinary Error', async () => {
    expect(await run(revokedProxy())).toEqual({ ok: true, value: false });
    expect(LINES()).toHaveLength(1);
    expect(reasonOf(LINES()[0])).toBe('unreadable thrown value');
  });
  it('RED ks1868 R2: a provider init failure that is an Error whose message getter throws is logged as text and initCardanoNFT answers false, as for an ordinary Error', async () => {
    expect(await run(getterThrows())).toEqual({ ok: true, value: false });
    expect(LINES()).toHaveLength(1);
    expect(reasonOf(LINES()[0])).toBe('unreadable thrown value');
  });
  it('RED ks1868 R3: a provider init failure that is an Error whose message is a Symbol is logged as text and initCardanoNFT answers false, as for an ordinary Error', async () => {
    expect(await run(withMessage(Symbol('ks1868')))).toEqual({ ok: true, value: false });
    expect(LINES()).toHaveLength(1);
    expect(reasonOf(LINES()[0])).toBe('Symbol(ks1868)');
  });
});
